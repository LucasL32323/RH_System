## Suposições
- Colunas opcionais no SQL mas obrigatórias no negócio (nome, e-mail, ativo, datas de admissão...) ficam `NOT NULL` no EF; tabelas, colunas, tipos, PKs e FKs seguem o SQL.
- Três índices únicos acrescentados: e-mail do usuário (sem diferenciar maiúsculas), ponto por funcionário por dia (RNF09, no lugar do `idx_ponto_func_data`) e código do certificado (RF39).
- `status`/`tipo` continuam texto livre; os valores permitidos entram como enums na Fase 3.
- Colunas `data_*` de momento exato (envio, solicitação, análise, publicação de aviso, resposta, criação do formulário, último login, visualização) são `DateTime` (`AAAA-MM-DDTHH:MM:SS`, Brasília); as de dia são `DateOnly`.

````markdown
## Banco de dados (Fase 2)

O modelo oficial é `database/modelo_fisico_sqlite.sql` (25 tabelas). O back-end o mapeia com EF Core
(`HrAutoDbContext`) e o banco é criado **só pelas migrations** (RNF15).

### Primeira vez: gerar a migration inicial (uma única vez)

```bash
cd backend
dotnet tool install --global dotnet-ef --version 8.0.10   # se ainda não tiver
dotnet restore
dotnet ef migrations add ModeloFisicoInicial --project src/HRAuto.Infrastructure --output-dir Persistencia/Migrations
```

Faça o commit da pasta `Persistencia/Migrations`. Depois disso, a API aplica as migrations sozinha ao subir.
Para recriar o banco do zero: apague `hrauto.db` (em `backend/src/HRAuto.Api`) e suba a API de novo.

Os testes `EsquemaDoBancoTests` comparam o banco gerado pelas migrations com o SQL oficial
(tabelas, colunas, tipos, chaves primárias e estrangeiras). Se algum falhar, o mapeamento divergiu do modelo.

### Dados iniciais (seed)

Rodam a cada subida da API e são idempotentes:

- Perfis fixos: `1 = Colaborador`, `2 = RH`, `3 = Admin` (iguais ao enum `PerfilAcesso`).
- 5 cargos e 5 departamentos de exemplo.
- **Somente em Development**: administrador de TESTE `admin@hrauto.local` / `Admin@123` (código FUNC-0001).
  Controlado por `Seed:CriarAdminDeTeste` (desligado no `appsettings.json`). Nunca ligue em produção.

### Alterações em relação ao SQL oficial

Não há tabelas nem colunas novas. Mudanças de integridade, todas em índices e nulidade:

- Índices únicos novos: `USUARIO.email` (sem diferenciar maiúsculas), `PONTO(id_funcionario, data)` (RNF09; substitui o
  `idx_ponto_func_data`) e `CERTIFICADO.codigo_validacao` (RF39).
- Colunas que o SQL deixava opcionais mas o negócio exige (ex.: `nome`, `email`, `ativo`, datas de admissão) são `NOT NULL` no EF.
- Datas `DateOnly` = `AAAA-MM-DD`; horários do ponto = `HH:MM:SS`; data e hora = `AAAA-MM-DDTHH:MM:SS` (Brasília).
````

```csharp
using System.Globalization;
using Microsoft.EntityFrameworkCore.Storage.ValueConversion;

namespace HRAuto.Infrastructure.Persistencia;

/// <summary>
/// Grava DateTime como texto ISO 8601 sem fração (yyyy-MM-ddTHH:mm:ss), no horário de Brasília (RNF11/RNF15).
/// O formato permite ordenar e filtrar por intervalo direto no SQLite (comparação de texto).
/// </summary>
public sealed class ConversorDataHora : ValueConverter<DateTime, string>
{
    private const string Formato = "yyyy-MM-dd'T'HH:mm:ss";

    public ConversorDataHora()
        : base(
            valor => valor.ToString(Formato, CultureInfo.InvariantCulture),
            texto => DateTime.ParseExact(texto, Formato, CultureInfo.InvariantCulture, DateTimeStyles.None))
    {
    }
}
```

### backend/src/HRAuto.Infrastructure/Persistencia/HrAutoDbContext.cs

```csharp
using System.Text.RegularExpressions;
using HRAuto.Domain.Entities;
using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Metadata;

namespace HRAuto.Infrastructure.Persistencia;

/// <summary>
/// Contexto do EF Core. Mapeia as 25 tabelas do modelo físico oficial (database/modelo_fisico_sqlite.sql).
/// Convenção de nomes: classe LogAuditoria -> tabela LOG_AUDITORIA; propriedade DataNascimento -> coluna data_nascimento;
/// Id -> id_&lt;tabela&gt;. Exceções ficam nas classes de Configuracoes.
/// </summary>
public class HrAutoDbContext : DbContext
{
    public HrAutoDbContext(DbContextOptions<HrAutoDbContext> opcoes) : base(opcoes) { }

    // --- Acesso e estrutura (M1/M2) ---
    public DbSet<Perfil> Perfis => Set<Perfil>();
    public DbSet<Cargo> Cargos => Set<Cargo>();
    public DbSet<Departamento> Departamentos => Set<Departamento>();
    public DbSet<Funcionario> Funcionarios => Set<Funcionario>();
    public DbSet<Usuario> Usuarios => Set<Usuario>();
    public DbSet<LogAuditoria> LogsAuditoria => Set<LogAuditoria>();
    public DbSet<FuncionarioDetalhes> FuncionariosDetalhes => Set<FuncionarioDetalhes>();

    // --- Ponto, férias e holerites (M3/M4) ---
    public DbSet<Ponto> Pontos => Set<Ponto>();
    public DbSet<ManutencaoPonto> ManutencoesPonto => Set<ManutencaoPonto>();
    public DbSet<BancoHoras> BancoHoras => Set<BancoHoras>();
    public DbSet<Feriado> Feriados => Set<Feriado>();
    public DbSet<Ferias> Ferias => Set<Ferias>();
    public DbSet<Holerite> Holerites => Set<Holerite>();

    // --- Comunicação e disciplina (M5) + benefícios ---
    public DbSet<Notificacao> Notificacoes => Set<Notificacao>();
    public DbSet<Aviso> Avisos => Set<Aviso>();
    public DbSet<AvisoFuncionario> AvisosFuncionarios => Set<AvisoFuncionario>();
    public DbSet<Advertencia> Advertencias => Set<Advertencia>();
    public DbSet<Beneficio> Beneficios => Set<Beneficio>();
    public DbSet<FuncionarioBeneficio> FuncionariosBeneficios => Set<FuncionarioBeneficio>();

    // --- Treinamentos e formulários (M6) ---
    public DbSet<Treinamento> Treinamentos => Set<Treinamento>();
    public DbSet<TreinamentoFuncionario> TreinamentosFuncionarios => Set<TreinamentoFuncionario>();
    public DbSet<Certificado> Certificados => Set<Certificado>();
    public DbSet<Formulario> Formularios => Set<Formulario>();
    public DbSet<Pergunta> Perguntas => Set<Pergunta>();
    public DbSet<Resposta> Respostas => Set<Resposta>();

    // Chaves primárias cujo nome não segue "id_<tabela em minúsculas>".
    private static readonly Dictionary<string, string> ChavesEspeciais = new()
    {
        ["LOG_AUDITORIA"] = "id_log",
        ["MANUTENCAO_PONTO"] = "id_manutencao"
    };

    protected override void ConfigureConventions(ModelConfigurationBuilder configuracao)
    {
        // Datas/horas (DateTime) vão como texto ISO 8601; DateOnly e TimeOnly já são ISO no provider SQLite.
        configuracao.Properties<DateTime>().HaveConversion<ConversorDataHora>();
    }

    protected override void OnModelCreating(ModelBuilder modelo)
    {
        // 1) Convenção de nomes (tabelas e colunas em snake_case)...
        foreach (var entidade in modelo.Model.GetEntityTypes())
        {
            var tabela = ParaSnake(entidade.ClrType.Name).ToUpperInvariant();
            entidade.SetTableName(tabela);

            foreach (var propriedade in entidade.GetProperties())
            {
                var coluna = propriedade.Name == "Id"
                    ? ChavesEspeciais.GetValueOrDefault(tabela, "id_" + tabela.ToLowerInvariant())
                    : ParaSnake(propriedade.Name);
                propriedade.SetColumnName(coluna);
            }
        }

        // 2) ...depois as configurações explícitas, que vencem a convenção quando houver conflito.
        modelo.ApplyConfigurationsFromAssembly(typeof(HrAutoDbContext).Assembly);

        // 3) Nenhuma exclusão em cascata: o modelo físico não tem ON DELETE e batidas de ponto nunca são excluídas (RNF07).
        foreach (var fk in modelo.Model.GetEntityTypes().SelectMany(e => e.GetForeignKeys()))
            fk.DeleteBehavior = DeleteBehavior.Restrict;
    }

    private static string ParaSnake(string nome) =>
        Regex.Replace(nome, "(?<=[a-z0-9])(?=[A-Z])", "_").ToLowerInvariant();
}
```

### backend/src/HRAuto.Infrastructure/Persistencia/Seed/InicializadorBanco.cs

```csharp
using HRAuto.Application.Abstractions;
using HRAuto.Domain.Entities;
using HRAuto.Domain.Enums;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Logging;
using Microsoft.Extensions.Options;

namespace HRAuto.Infrastructure.Persistencia.Seed;

/// <summary>
/// Aplica as migrations (o banco é recriado do zero só com elas, RNF15) e grava os dados iniciais.
/// É idempotente: pode rodar a cada subida da API sem duplicar registros.
/// </summary>
public sealed class InicializadorBanco : IInicializadorBanco
{
    private static readonly (PerfilAcesso Perfil, string Descricao)[] Perfis =
    {
        (PerfilAcesso.Colaborador, "Acessa as próprias informações: ponto, férias, holerites, advertências e treinamentos."),
        (PerfilAcesso.RH, "Gerencia funcionários, ponto, férias, advertências, avisos, treinamentos e formulários."),
        (PerfilAcesso.Admin, "Acesso total, inclusive usuários, perfis e auditoria.")
    };

    private static readonly string[] Cargos =
    {
        "Administrador do Sistema", "Analista de RH", "Gerente de Equipe", "Analista de Sistemas", "Assistente Administrativo"
    };

    private static readonly string[] Departamentos =
    {
        "Recursos Humanos", "Tecnologia da Informação", "Administrativo", "Financeiro", "Operações"
    };

    private readonly HrAutoDbContext _db;
    private readonly IHashSenha _hash;
    private readonly IRelogio _relogio;
    private readonly OpcoesSeed _opcoes;
    private readonly ILogger<InicializadorBanco> _log;

    public InicializadorBanco(
        HrAutoDbContext db,
        IHashSenha hash,
        IRelogio relogio,
        IOptions<OpcoesSeed> opcoes,
        ILogger<InicializadorBanco> log)
    {
        _db = db;
        _hash = hash;
        _relogio = relogio;
        _opcoes = opcoes.Value;
        _log = log;
    }

    public async Task ExecutarAsync(CancellationToken cancelamento = default)
    {
        // Sem migrations o banco não teria tabelas: avisa com o comando exato em vez de falhar com "no such table".
        if (!_db.Database.GetMigrations().Any())
            throw new InvalidOperationException(
                "Nenhuma migration encontrada. Gere a inicial (uma única vez) na pasta backend: " +
                "dotnet ef migrations add ModeloFisicoInicial --project src/HRAuto.Infrastructure --output-dir Persistencia/Migrations");

        await _db.Database.MigrateAsync(cancelamento);

        await SemearPerfisAsync(cancelamento);
        await SemearCargosAsync(cancelamento);
        await SemearDepartamentosAsync(cancelamento);

        if (_opcoes.CriarAdminDeTeste)
            await SemearAdminDeTesteAsync(cancelamento);
    }

    // Os ids dos perfis são fixos (1 = Colaborador, 2 = RH, 3 = Admin), iguais ao enum PerfilAcesso.
    private async Task SemearPerfisAsync(CancellationToken ct)
    {
        foreach (var (perfil, descricao) in Perfis)
        {
            var id = (int)perfil;
            if (await _db.Perfis.AnyAsync(p => p.Id == id, ct)) continue;

            var entidade = new Perfil { Nome = NomesPerfil.De(perfil), Descricao = descricao };
            _db.Perfis.Add(entidade);
            _db.Entry(entidade).Property(p => p.Id).CurrentValue = id;
        }
        await _db.SaveChangesAsync(ct);
    }

    private async Task SemearCargosAsync(CancellationToken ct)
    {
        foreach (var nome in Cargos)
            if (!await _db.Cargos.AnyAsync(c => c.NomeCargo == nome, ct))
                _db.Cargos.Add(new Cargo { NomeCargo = nome });
        await _db.SaveChangesAsync(ct);
    }

    private async Task SemearDepartamentosAsync(CancellationToken ct)
    {
        foreach (var nome in Departamentos)
            if (!await _db.Departamentos.AnyAsync(d => d.Nome == nome, ct))
                _db.Departamentos.Add(new Departamento { Nome = nome });
        await _db.SaveChangesAsync(ct);
    }

    private async Task SemearAdminDeTesteAsync(CancellationToken ct)
    {
        if (string.IsNullOrWhiteSpace(_opcoes.AdminSenha))
```

### backend/src/HRAuto.Infrastructure/Seguranca/HashSenhaBCrypt.cs

```csharp
using BCryptNet = BCrypt.Net.BCrypt;
using HRAuto.Application.Abstractions;

namespace HRAuto.Infrastructure.Seguranca;

/// <summary>
/// Hash de senha com BCrypt (RNF05). O salt vai embutido no hash; o custo (work factor) é 11, acima do mínimo 10.
/// </summary>
public sealed class HashSenhaBCrypt : IHashSenha
{
    public const int CustoMinimo = 10;
    private readonly int _custo;

    public HashSenhaBCrypt() : this(11) { }

    public HashSenhaBCrypt(int custo)
    {
        if (custo < CustoMinimo)
            throw new ArgumentOutOfRangeException(nameof(custo), custo, $"O custo do BCrypt deve ser pelo menos {CustoMinimo} (RNF05).");
        _custo = custo;
    }

    public string Gerar(string senha)
    {
        if (string.IsNullOrEmpty(senha))
            throw new ArgumentException("A senha é obrigatória.", nameof(senha));
        return BCryptNet.HashPassword(senha, _custo);
    }

    public bool Verificar(string senha, string hash)
    {
        if (string.IsNullOrEmpty(senha) || string.IsNullOrEmpty(hash)) return false;
        try
        {
            return BCryptNet.Verify(senha, hash);
        }
        catch (Exception)
        {
            return false; // hash em formato inválido: falha fechada, nunca autentica
        }
    }
}
```

```csharp
using Microsoft.Data.Sqlite;
using Microsoft.EntityFrameworkCore;

namespace HRAuto.Tests.Persistencia;

/// <summary>
/// RNF15: "Banco recriado do zero só com as migrations". Estes testes montam o banco pelas migrations e o
/// comparam com o modelo físico oficial (modelo_fisico_sqlite.sql): tabelas, colunas, tipos, chaves primárias e FKs.
/// Pré-requisito: a migration inicial gerada (ver README, "Banco de dados").
/// </summary>
public class EsquemaDoBancoTests
{
    private sealed record Esquema(
        SortedSet<string> Tabelas,
        SortedSet<string> Colunas,      // "TABELA.coluna:TIPO"
        SortedSet<string> Chaves,       // "TABELA.coluna#posição"
        SortedSet<string> Estrangeiras, // "TABELA.coluna->TABELA_PAI.coluna_pai"
        SortedSet<string> Indices);     // nomes dos índices criados explicitamente

    private static Esquema Ler(SqliteConnection conexao)
    {
        var e = new Esquema(new(), new(), new(), new(), new());

        using (var cmd = conexao.CreateCommand())
        {
            cmd.CommandText =
                "SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%' AND name <> '__EFMigrationsHistory'";
            using var r = cmd.ExecuteReader();
            while (r.Read()) e.Tabelas.Add(r.GetString(0));
        }

        foreach (var tabela in e.Tabelas)
        {
            using (var cmd = conexao.CreateCommand())
            {
                cmd.CommandText = $"PRAGMA table_info('{tabela}')";
                using var r = cmd.ExecuteReader();
                while (r.Read())
                {
                    e.Colunas.Add($"{tabela}.{r.GetString(1)}:{r.GetString(2).ToUpperInvariant()}");
                    if (r.GetInt32(5) > 0) e.Chaves.Add($"{tabela}.{r.GetString(1)}#{r.GetInt32(5)}");
                }
            }

            using (var cmd = conexao.CreateCommand())
            {
                cmd.CommandText = $"PRAGMA foreign_key_list('{tabela}')";
                using var r = cmd.ExecuteReader();
                while (r.Read())
                    e.Estrangeiras.Add($"{tabela}.{r.GetString(3)}->{r.GetString(2)}.{r.GetString(4)}");
            }
        }

        using (var cmd = conexao.CreateCommand())
        {
            cmd.CommandText = "SELECT name FROM sqlite_master WHERE type='index' AND name NOT LIKE 'sqlite_%' AND name NOT LIKE 'IX_%'";
            using var r = cmd.ExecuteReader();
            while (r.Read()) e.Indices.Add(r.GetString(0));
        }

        return e;
    }

    private static (Esquema Oficial, Esquema Migrations) Montar()
    {
        // 1) Banco criado pelo SQL oficial.
        var oficial = new SqliteConnection("Data Source=:memory:");
        oficial.Open();
        using (var cmd = oficial.CreateCommand())
        {
            cmd.CommandText = File.ReadAllText(Path.Combine(AppContext.BaseDirectory, "modelo_fisico_sqlite.sql"));
            cmd.ExecuteNonQuery();
        }

        // 2) Banco criado só pelas migrations.
        using var banco = new BancoDeTeste();
        using var contexto = banco.NovoContexto();
        Assert.True(contexto.Database.GetMigrations().Any(),
            "Nenhuma migration encontrada. Gere a inicial: dotnet ef migrations add ModeloFisicoInicial " +
            "--project src/HRAuto.Infrastructure --output-dir Persistencia/Migrations");
        contexto.Database.Migrate();

        var conexaoMigrations = (SqliteConnection)contexto.Database.GetDbConnection();
        return (Ler(oficial), Ler(conexaoMigrations));
    }

    [Fact]
    public void Migrations_CriamAsMesmas25Tabelas_DoModeloFisico()
    {
        var (oficial, migrations) = Montar();

        Assert.Equal(25, oficial.Tabelas.Count);
        Assert.Equal(oficial.Tabelas, migrations.Tabelas);
    }

    [Fact]
    public void Migrations_CriamAsMesmasColunasETipos_DoModeloFisico()
    {
        var (oficial, migrations) = Montar();

        Assert.Equal(oficial.Colunas, migrations.Colunas);
    }

    [Fact]
    public void Migrations_CriamAsMesmasChavesPrimariasEEstrangeiras_DoModeloFisico()
    {
        var (oficial, migrations) = Montar();

        Assert.Equal(oficial.Chaves, migrations.Chaves);
        Assert.Equal(oficial.Estrangeiras, migrations.Estrangeiras);
    }

    [Fact]
    public void Migrations_CriamOsIndicesDefinidos()
    {
        var (_, migrations) = Montar();

        // Índices do modelo físico (menos idx_ponto_func_data, trocado por um índice único com as mesmas colunas)
        // + índices únicos acrescentados: e-mail do usuário, ponto por dia e código do certificado.
        var esperados = new[]
        {
            "idx_ferias_func", "idx_holerite_func", "idx_notificacao_func", "idx_log_usuario",
            "ux_ponto_func_data", "ux_usuario_email", "ux_certificado_codigo"
        };
        foreach (var indice in esperados)
            Assert.Contains(indice, migrations.Indices);
    }
}
```

```csharp
using Microsoft.AspNetCore.Hosting;
using Microsoft.AspNetCore.Mvc.Testing;
using Microsoft.Extensions.Configuration;

namespace HRAuto.Tests.Api;

/// <summary>
/// Sobe a API em memória com um banco SQLite temporário (arquivo apagado ao final),
/// para os testes de integração não tocarem no hrauto.db de desenvolvimento.
/// </summary>
public sealed class ApiFabrica : WebApplicationFactory<Program>
{
    private readonly string _arquivoBanco = Path.Combine(Path.GetTempPath(), $"hrauto-teste-{Guid.NewGuid():N}.db");

    protected override void ConfigureWebHost(IWebHostBuilder construtor)
    {
        construtor.ConfigureAppConfiguration((_, configuracao) =>
        {
            configuracao.AddInMemoryCollection(new Dictionary<string, string?>
            {
                ["Banco:ConnectionString"] = $"Data Source={_arquivoBanco};Pooling=False",
                ["Seed:CriarAdminDeTeste"] = "true",
                ["Seed:AdminEmail"] = "admin@hrauto.local",
                ["Seed:AdminSenha"] = "Admin@123"
            });
        });
    }

    protected override void Dispose(bool disposing)
    {
        base.Dispose(disposing);
        try { File.Delete(_arquivoBanco); } catch (IOException) { /* arquivo ainda em uso: o SO limpa a pasta temp */ }
    }
}
```
