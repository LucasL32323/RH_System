```csharp
using HRAuto.Domain.Common;
using HRAuto.Domain.Enums;

namespace HRAuto.Domain.Entities;

/// <summary>
/// Tabela PONTO (RF17, RF18, RF19, RNF09): um registro por funcionário por dia, com as quatro batidas.
/// Batidas nunca são excluídas (RNF07). Campos de horário nulos = batida ainda não feita.
/// As regras de sequência (RF18) e de sincronização offline (RF19) ficam aqui, no Domain.
/// Jornadas que viram a meia-noite não são tratadas (fora do escopo do TCC).
/// </summary>
public class Ponto : EntidadeBase
{
    /// <summary>Intervalo mínimo entre duas batidas do mesmo dia; abaixo disso a batida é considerada repetida.</summary>
    public static readonly TimeSpan IntervaloMinimoEntreBatidas = TimeSpan.FromMinutes(1);

    // Ordem obrigatória das batidas do dia.
    private static readonly TipoBatida[] Sequencia =
    {
        TipoBatida.Entrada, TipoBatida.SaidaIntervalo, TipoBatida.RetornoIntervalo, TipoBatida.Saida
    };

    public int IdFuncionario { get; set; }
    public DateOnly Data { get; set; }
    public TimeOnly? Entrada { get; set; }
    public TimeOnly? SaidaIntervalo { get; set; }
    public TimeOnly? RetornoIntervalo { get; set; }
    public TimeOnly? Saida { get; set; }

    /// <summary>Cria o registro (vazio) do dia para o funcionário.</summary>
    public static Ponto Criar(int idFuncionario, DateOnly data)
    {
        if (idFuncionario <= 0) throw new DominioException("O funcionário do ponto é obrigatório.");
        return new Ponto { IdFuncionario = idFuncionario, Data = data };
    }

    /// <summary>Horário gravado para o tipo de batida, ou nulo se ainda não foi feita.</summary>
    public TimeOnly? Obter(TipoBatida tipo) => tipo switch
    {
        TipoBatida.Entrada => Entrada,
        TipoBatida.SaidaIntervalo => SaidaIntervalo,
        TipoBatida.RetornoIntervalo => RetornoIntervalo,
        TipoBatida.Saida => Saida,
        _ => throw new ArgumentOutOfRangeException(nameof(tipo), tipo, "Tipo de batida desconhecido.")
    };

    /// <summary>Quantas das quatro batidas já foram feitas.</summary>
    public int TotalBatidas() => Sequencia.Count(t => Obter(t) is not null);

    /// <summary>Próxima batida esperada (a primeira ainda vazia), ou nulo se o dia já está completo.</summary>
    public TipoBatida? ProximaBatida()
    {
        foreach (var tipo in Sequencia)
            if (Obter(tipo) is null) return tipo;
        return null;
    }

    /// <summary>
    /// Registra a próxima batida do dia (RF17/RF18). O horário vem do servidor (relógio de Brasília).
    /// Bloqueia batida fora de ordem, repetida ou além da quarta batida.
    /// </summary>
    /// <returns>O tipo de batida que foi gravado.</returns>
    public TipoBatida RegistrarBatida(TimeOnly hora)
    {
        var tipo = ProximaBatida()
                   ?? throw new DominioException("As quatro batidas do dia já foram registradas.");

        ValidarOrdem(tipo, hora);
        Definir(tipo, hora);
        return tipo;
    }

    /// <summary>
    /// Aplica uma batida guardada offline no aparelho (RF19). É idempotente (RNF09): reenviar a mesma batida
    /// (mesmo tipo e mesmo horário) não duplica nada e devolve <see cref="ResultadoSincronizacao.JaRegistrada"/>.
    /// Se o tipo já tiver outro horário gravado, a batida é recusada: a correção passa por manutenção de ponto (RF22).
    /// </summary>
    public ResultadoSincronizacao AplicarBatidaSincronizada(TipoBatida tipo, TimeOnly hora)
    {
        var atual = Obter(tipo);
        if (atual is not null)
        {
            if (atual.Value == hora) return ResultadoSincronizacao.JaRegistrada;
            throw new DominioException(
                $"Já existe a batida de {Nome(tipo)} às {atual.Value:HH\\:mm\\:ss}. " +
                "Para corrigir, solicite a manutenção do ponto.");
        }

        // Todas as batidas anteriores precisam existir: a fila offline é enviada em ordem.
        foreach (var anterior in Sequencia.TakeWhile(t => t != tipo))
        {
            if (Obter(anterior) is null)
                throw new DominioException($"Falta a batida de {Nome(anterior)} antes da batida de {Nome(tipo)}.");
        }

        ValidarOrdem(tipo, hora);
        Definir(tipo, hora);
        return ResultadoSincronizacao.Aplicada;
    }

    /// <summary>Nome da batida em português, para mensagens.</summary>
    public static string Nome(TipoBatida tipo) => tipo switch
    {
        TipoBatida.Entrada => "entrada",
        TipoBatida.SaidaIntervalo => "saída para o intervalo",
        TipoBatida.RetornoIntervalo => "retorno do intervalo",
        TipoBatida.Saida => "saída",
        _ => tipo.ToString()
    };

    private void Definir(TipoBatida tipo, TimeOnly hora)
    {
        switch (tipo)
        {
            case TipoBatida.Entrada: Entrada = hora; break;
            case TipoBatida.SaidaIntervalo: SaidaIntervalo = hora; break;
            case TipoBatida.RetornoIntervalo: RetornoIntervalo = hora; break;
            case TipoBatida.Saida: Saida = hora; break;
            default: throw new ArgumentOutOfRangeException(nameof(tipo), tipo, "Tipo de batida desconhecido.");
        }
    }

    // A nova batida precisa ser posterior à batida anterior (com folga mínima) e anterior a uma eventual seguinte.
    private void ValidarOrdem(TipoBatida tipo, TimeOnly hora)
    {
        var indice = Array.IndexOf(Sequencia, tipo);

        for (var i = indice - 1; i >= 0; i--)
        {
            var anterior = Obter(Sequencia[i]);
            if (anterior is null) continue;

            var diferenca = hora.ToTimeSpan() - anterior.Value.ToTimeSpan();
            if (diferenca < TimeSpan.Zero)
                throw new DominioException(
                    $"Batida fora de ordem: a {Nome(tipo)} ({hora:HH\\:mm}) é anterior à {Nome(Sequencia[i])} ({anterior.Value:HH\\:mm}).");
            if (diferenca < IntervaloMinimoEntreBatidas)
                throw new DominioException(
                    $"Batida repetida: já houve a {Nome(Sequencia[i])} às {anterior.Value:HH\\:mm\\:ss}.");
            break;
        }

        for (var i = indice + 1; i < Sequencia.Length; i++)
        {
            var seguinte = Obter(Sequencia[i]);
            if (seguinte is null) continue;

            if (hora.ToTimeSpan() >= seguinte.Value.ToTimeSpan())
                throw new DominioException(
                    $"Batida fora de ordem: a {Nome(tipo)} ({hora:HH\\:mm}) não pode ser depois da {Nome(Sequencia[i])} ({seguinte.Value:HH\\:mm}).");
            break;
        }
    }
}
```

```csharp
/// Jornada padrão usada no cálculo de horas (RF20). Valores iniciais: segunda a sexta, entrada às 08:00,

    {
        DayOfWeek.Monday, DayOfWeek.Tuesday, DayOfWeek.Wednesday, DayOfWeek.Thursday, DayOfWeek.Friday
    };

    /// <summary>Jornada padrão da empresa.</summary>
    public static JornadaPadrao Padrao => new();

    /// <summary>Verdadeiro se o dia da semana tem expediente (não olha feriados nem férias).</summary>
```

```csharp
using HRAuto.Domain.Entities;
using HRAuto.Domain.Enums;

namespace HRAuto.Domain.Jornada;

/// <summary>
/// Calcula horas trabalhadas, extras e atrasos do dia pela jornada padrão (RF20).
/// Regras:
/// - Feriado, férias aprovadas e descanso semanal não têm carga: não geram atraso nem hora faltante;
///   se a pessoa trabalhar nesses dias, todo o tempo trabalhado vira hora extra.
/// - A tolerância diária (padrão 10 min) vale para hora extra, hora faltante e atraso. Passou dela, conta tudo.
/// - Só dia COMPLETO (4 batidas) gera horas extras/faltantes. Dia aberto fica "em andamento"; dia encerrado
///   com batida faltando fica "inconsistente" até haver manutenção de ponto (RF22).
/// - Falta (dia útil encerrado sem batida) é contada à parte e NÃO debita o banco de horas.
/// </summary>
public sealed class CalculadoraJornada
{
    private readonly JornadaPadrao _jornada;

    public CalculadoraJornada() : this(JornadaPadrao.Padrao) { }

    public CalculadoraJornada(JornadaPadrao jornada) => _jornada = jornada;

    /// <param name="ponto">Registro do dia (nulo se não há nenhuma batida).</param>
    /// <param name="data">Dia calculado.</param>
    /// <param name="contexto">Natureza do dia (ver <see cref="ResolvedorContextoDia"/>).</param>
    /// <param name="diaEncerrado">Verdadeiro se o dia já acabou (data anterior a hoje no horário de Brasília).</param>
    public ResumoDia Calcular(Ponto? ponto, DateOnly data, ContextoDia contexto, bool diaEncerrado)
    {
        var util = contexto == ContextoDia.DiaUtil;
        var carga = util ? _jornada.CargaDiaria : TimeSpan.Zero;
        var batidas = ponto?.TotalBatidas() ?? 0;
        var trabalhado = Trabalhado(ponto);
        var atraso = util ? Atraso(ponto) : TimeSpan.Zero;

        var situacao = batidas switch
        {
            0 when !util => SituacaoDia.SemExpediente,
            0 => diaEncerrado ? SituacaoDia.Falta : SituacaoDia.EmAndamento,
            4 => SituacaoDia.Completo,
            _ => diaEncerrado ? SituacaoDia.Inconsistente : SituacaoDia.EmAndamento
        };

        var extras = TimeSpan.Zero;
        var faltantes = TimeSpan.Zero;

        if (situacao == SituacaoDia.Completo)
        {
            if (!util)
            {
                extras = trabalhado; // trabalho em feriado, férias ou descanso: tudo é extra
            }
            else
            {
                var diferenca = trabalhado - carga;
                if (diferenca.Duration() > _jornada.ToleranciaDiaria)
                {
                    if (diferenca > TimeSpan.Zero) extras = diferenca;
                    else faltantes = -diferenca;
                }
            }
        }

        return new ResumoDia(data, contexto, situacao, trabalhado, carga, extras, faltantes, atraso);
    }

    /// <summary>RF25: o dia passou do limite de horas extras que obriga a notificar o gestor?</summary>
    public bool ExcedeLimiteHoraExtra(ResumoDia resumo) => resumo.HorasExtras > _jornada.LimiteHoraExtraDiaria;

    /// <summary>Soma os dias de um período (espelho de ponto e indicadores).</summary>
    public static ResumoPeriodo Totalizar(IEnumerable<ResumoDia> dias)
    {
        var trabalhado = TimeSpan.Zero;
        var extras = TimeSpan.Zero;
        var faltantes = TimeSpan.Zero;
        var atrasos = TimeSpan.Zero;
        var diasTrabalhados = 0;
        var faltas = 0;
        var inconsistentes = 0;

        foreach (var dia in dias)
        {
            trabalhado += dia.Trabalhado;
            extras += dia.HorasExtras;
            faltantes += dia.HorasFaltantes;
            atrasos += dia.Atraso;
            if (dia.Trabalhado > TimeSpan.Zero) diasTrabalhados++;
            if (dia.Situacao == SituacaoDia.Falta) faltas++;
            if (dia.Situacao == SituacaoDia.Inconsistente) inconsistentes++;
        }

        return new ResumoPeriodo(trabalhado, extras, faltantes, atrasos, diasTrabalhados, faltas, inconsistentes);
    }

    // Soma os dois períodos do dia que estiverem fechados (entrada->saída p/ intervalo e retorno->saída).
    private static TimeSpan Trabalhado(Ponto? ponto)
    {
        if (ponto is null) return TimeSpan.Zero;
        return Periodo(ponto.Entrada, ponto.SaidaIntervalo) + Periodo(ponto.RetornoIntervalo, ponto.Saida);
    }

    private static TimeSpan Periodo(TimeOnly? inicio, TimeOnly? fim)
    {
        if (inicio is null || fim is null) return TimeSpan.Zero;
        var duracao = fim.Value.ToTimeSpan() - inicio.Value.ToTimeSpan();
        return duracao > TimeSpan.Zero ? duracao : TimeSpan.Zero; // dado inconsistente nunca gera horas negativas
    }

    private TimeSpan Atraso(Ponto? ponto)
    {
        if (ponto?.Entrada is null) return TimeSpan.Zero;
        var atraso = ponto.Entrada.Value.ToTimeSpan() - _jornada.HoraEntrada.ToTimeSpan();
        return atraso > _jornada.ToleranciaDiaria ? atraso : TimeSpan.Zero;
    }
}
```

```csharp
/// <summary>
/// Decide se um dia é útil, descanso semanal, feriado ou férias (RF20, RF28).
/// Prioridade quando mais de um se aplica: férias aprovadas, feriado, descanso semanal.
/// </summary>
public static class ResolvedorContextoDia
{
    /// <summary>
    /// Datas de feriado que valem para o funcionário: nacionais valem para todos; estaduais, só para quem mora no
    /// mesmo estado; municipais, só para quem mora na mesma cidade e estado. Cidade e estado vêm de
    /// FUNCIONARIO_DETALHES. Feriado com tipo desconhecido é ignorado.
    /// </summary>
    public static HashSet<DateOnly> DatasDeFeriadoAplicaveis(IEnumerable<Feriado> feriados, string? cidade, string? estado)
    {
        var datas = new HashSet<DateOnly>();
        foreach (var feriado in feriados)
        {
            var vale = feriado.Tipo switch
            {
                TiposFeriado.Nacional => true,
                TiposFeriado.Estadual => Igual(feriado.Estado, estado),
                TiposFeriado.Municipal => Igual(feriado.Estado, estado) && Igual(feriado.Cidade, cidade),
                _ => false
            };
            if (vale) datas.Add(feriado.Data);
        }
        return datas;
    }

    /// <param name="data">Dia a classificar.</param>
    /// <param name="jornada">Jornada que define os dias úteis da semana.</param>
    /// <param name="feriasDoFuncionario">Férias do funcionário (só as aprovadas contam).</param>
    /// <param name="datasDeFeriado">Resultado de <see cref="DatasDeFeriadoAplicaveis"/>.</param>
    public static ContextoDia Resolver(
        DateOnly data,
        JornadaPadrao jornada,
        IEnumerable<Ferias> feriasDoFuncionario,
        IReadOnlySet<DateOnly> datasDeFeriado)
    {
        if (feriasDoFuncionario.Any(f => RegrasFerias.CobreDia(f, data))) return ContextoDia.Ferias;
        if (datasDeFeriado.Contains(data)) return ContextoDia.Feriado;
        return jornada.TemExpediente(data) ? ContextoDia.DiaUtil : ContextoDia.DescansoSemanal;
    }

    private static bool Igual(string? a, string? b) =>
        !string.IsNullOrWhiteSpace(a) && !string.IsNullOrWhiteSpace(b) &&
        string.Equals(a.Trim(), b.Trim(), StringComparison.OrdinalIgnoreCase);
}
```

```csharp
using HRAuto.Domain.Common;
using HRAuto.Domain.Entities;
using HRAuto.Domain.Enums;

namespace HRAuto.Domain.Jornada;

/// <summary>
/// Banco de horas (RF24): créditos e débitos diários gerados pelo ponto e ajustes manuais do RH.
/// Quantidade é sempre positiva e em horas decimais (4 casas); o sinal vem do tipo (Credito/Debito).
/// </summary>
public static class RegrasBancoHoras
{
    private const int CasasDecimais = 4;
    public const int TamanhoMinimoObservacao = 5;

    /// <summary>
    /// Gera o lançamento do fechamento do dia, ou nulo quando não há o que lançar:
    /// dia que não está completo (em andamento, inconsistente, falta, sem expediente), saldo zero,
    /// ou dia que já tem lançamento de origem "Ponto" (idempotente: rodar o fechamento duas vezes não duplica).
    /// </summary>
    public static BancoHoras? LancamentoDoDia(int idFuncionario, ResumoDia resumo, IEnumerable<BancoHoras> lancamentosExistentes)
    {
        if (idFuncionario <= 0) throw new DominioException("O funcionário do lançamento é obrigatório.");
        if (resumo.Situacao != SituacaoDia.Completo) return null;

        var jaLancado = lancamentosExistentes.Any(l =>
            l.IdFuncionario == idFuncionario && l.Data == resumo.Data && l.Origem == OrigensBancoHoras.Ponto);
        if (jaLancado) return null;

        var horas = Math.Round(resumo.Saldo.TotalHours, CasasDecimais, MidpointRounding.AwayFromZero);
        if (horas == 0) return null;

        return new BancoHoras
        {
            IdFuncionario = idFuncionario,
            Data = resumo.Data,
            Tipo = horas > 0 ? TiposBancoHoras.Credito : TiposBancoHoras.Debito,
            Quantidade = Math.Abs(horas),
            Origem = OrigensBancoHoras.Ponto,
            Observacao = $"Fechamento do ponto de {resumo.Data:dd/MM/yyyy}"
        };
    }

    /// <summary>
    /// Ajuste manual do RH. Exige observação (rastreabilidade, RNF07) e não aceita data futura.
    /// </summary>
    public static BancoHoras AjusteManual(
        int idFuncionario, DateOnly data, string tipo, double horas, string? observacao, DateOnly hoje)
    {
        if (idFuncionario <= 0) throw new DominioException("O funcionário do ajuste é obrigatório.");
        if (tipo != TiposBancoHoras.Credito && tipo != TiposBancoHoras.Debito)
            throw new DominioException("O tipo do ajuste deve ser Credito ou Debito.");
        if (double.IsNaN(horas) || double.IsInfinity(horas) || horas <= 0)
            throw new DominioException("A quantidade de horas do ajuste deve ser maior que zero.");
        if (data > hoje) throw new DominioException("O ajuste não pode ter data futura.");

        var texto = observacao?.Trim() ?? string.Empty;
        if (texto.Length < TamanhoMinimoObservacao)
            throw new DominioException($"Informe a observação do ajuste (mínimo de {TamanhoMinimoObservacao} caracteres).");

        return new BancoHoras
        {
            IdFuncionario = idFuncionario,
            Data = data,
            Tipo = tipo,
            Quantidade = Math.Round(horas, CasasDecimais, MidpointRounding.AwayFromZero),
            Origem = OrigensBancoHoras.AjusteRH,
            Observacao = texto
        };
    }

    /// <summary>Saldo em horas: soma dos créditos menos a soma dos débitos (pode ser negativo).</summary>
    public static decimal Saldo(IEnumerable<BancoHoras> lancamentos)
    {
        decimal saldo = 0;
        foreach (var l in lancamentos)
        {
            var quantidade = (decimal)l.Quantidade;
            saldo += l.Tipo switch
            {
                TiposBancoHoras.Credito => quantidade,
                TiposBancoHoras.Debito => -quantidade,
                _ => throw new DominioException($"Tipo de lançamento desconhecido no banco de horas: '{l.Tipo}'.")
            };
        }
        return Math.Round(saldo, CasasDecimais, MidpointRounding.AwayFromZero);
    }
}
```

```csharp
/// <summary>

    public const int TamanhoMinimoObservacaoRejeicao = 5;

    /// <param name="ponto">Registro de ponto que será ajustado (já salvo, com Id).</param>
    /// <param name="idSolicitante">Funcionário que está pedindo; só pode pedir do próprio ponto.</param>
    /// <param name="justificativa">Obrigatória.</param>
    /// <param name="agora">Data e hora de Brasília (do relógio do servidor).</param>

        if (existentes.Any(m => m.IdPonto == ponto.Id && m.Status == StatusSolicitacao.Pendente))
            throw new DominioException("Já existe um pedido de manutenção pendente para este dia.");

        return new ManutencaoPonto
        {
            IdPonto = ponto.Id,
            IdFuncionario = idSolicitante,
            Justificativa = texto,
            DataSolicitacao = agora,
            Status = StatusSolicitacao.Pendente
        };
    }

        pedido.IdResponsavel = idResponsavel;
        pedido.DataAnalise = agora;
        pedido.Observacao = string.IsNullOrEmpty(texto) ? null : texto;
    }
```

```csharp
using HRAuto.Domain.Common;
using HRAuto.Domain.Entities;
using HRAuto.Domain.Enums;
using HRAuto.Domain.ValueObjects;

namespace HRAuto.Domain.Afastamentos;

/// <summary>Dados mínimos do funcionário que as regras de férias precisam (evita acoplar a entidade inteira).</summary>
public sealed record DadosFuncionario(int Id, DateOnly DataAdmissao, bool Ativo);

/// <summary>
/// Regras de férias (RF26, RF27), baseadas na CLT (arts. 130 e 134):
/// - O direito nasce após 12 meses de casa; cada ano completo de casa forma um período aquisitivo de 30 dias.
/// - Pode-se dividir em até 3 frações; cada fração tem no mínimo 5 dias e uma delas, no mínimo 14 dias.
/// - O início não pode cair em domingo, em feriado, nem nos 2 dias que antecedem domingo ou feriado.
/// Simplificações assumidas (a tabela FERIAS não tem coluna de período aquisitivo):
/// - Cada solicitação pertence ao período aquisitivo identificado pela DATA DE INÍCIO (anos completos de casa
///   nessa data). Férias vencidas de anos anteriores e abono pecuniário não são controlados.
/// - Solicitações Pendentes e Aprovadas consomem saldo; Rejeitadas não.
/// </summary>
public static class RegrasFerias
{
    public const int DiasPorPeriodo = 30;
    public const int MinimoDiasPorFracao = 5;
    public const int MinimoDiasFracaoPrincipal = 14;
    public const int MaximoFracoes = 3;
    public const int TamanhoMinimoMotivoRejeicao = 5;

    /// <summary>As férias estão aprovadas e cobrem o dia? (usado pelo cálculo de horas, RF20)</summary>
    public static bool CobreDia(Ferias ferias, DateOnly dia) =>
        ferias.Status == StatusSolicitacao.Aprovada && dia >= ferias.DataInicio && dia <= ferias.DataFim;

    /// <summary>Anos completos de casa na data (0 = ainda no primeiro ano, sem direito).</summary>
    public static int AnosCompletosDeCasa(DateOnly admissao, DateOnly data)
    {
        if (data < admissao) return 0;
        var anos = data.Year - admissao.Year;
        if (data < admissao.AddYears(anos)) anos--;
        return anos;
    }

    /// <summary>
    /// Verdadeiro se o início é permitido: o dia de início e os 2 dias seguintes não podem ser domingo nem feriado
    /// (CLT, art. 134 §3º: vedado iniciar nos 2 dias que antecedem feriado ou dia de repouso semanal).
    /// </summary>
    public static bool InicioPermitido(DateOnly inicio, IReadOnlySet<DateOnly> datasDeFeriado)
    {
        for (var i = 0; i <= 2; i++)
        {
            var dia = inicio.AddDays(i);
            if (dia.DayOfWeek == DayOfWeek.Sunday || datasDeFeriado.Contains(dia)) return false;
        }
        return true;
    }

    /// <summary>Cria a solicitação (status Pendente) ou lança <see cref="DominioException"/> com o motivo.</summary>
    /// <param name="dias">Quantidade informada pelo funcionário; precisa bater com o período (RF26).</param>
    /// <param name="hoje">Data de hoje em Brasília.</param>
    /// <param name="feriasDoFuncionario">Férias já cadastradas do funcionário (outros funcionários são ignorados).</param>
    /// <param name="datasDeFeriado">Feriados que valem para o funcionário (<see cref="Jornada.ResolvedorContextoDia"/>).</param>
    public static Ferias Solicitar(
        DadosFuncionario funcionario,
        DateOnly inicio,
        DateOnly fim,
        int dias,
        string? observacao,
        DateOnly hoje,
        IEnumerable<Ferias> feriasDoFuncionario,
        IReadOnlySet<DateOnly> datasDeFeriado)
    {
        if (!funcionario.Ativo) throw new DominioException("Funcionário inativo não pode solicitar férias.");
        if (fim < inicio) throw new DominioException("A data final não pode ser anterior à data inicial.");

        var diasDoPeriodo = fim.DayNumber - inicio.DayNumber + 1;
        if (dias != diasDoPeriodo)
            throw new DominioException($"A quantidade de dias ({dias}) não corresponde ao período informado ({diasDoPeriodo} dias).");
        if (inicio < hoje) throw new DominioException("O início das férias não pode ser no passado.");

        var periodo = AnosCompletosDeCasa(funcionario.DataAdmissao, inicio);
        if (periodo < 1)
            throw new DominioException("O direito a férias começa após 12 meses de trabalho; a data de início é anterior a isso.");

        if (dias < MinimoDiasPorFracao)
            throw new DominioException($"Cada período de férias deve ter no mínimo {MinimoDiasPorFracao} dias.");
        if (!InicioPermitido(inicio, datasDeFeriado))
            throw new DominioException("As férias não podem começar em domingo, feriado ou nos 2 dias que os antecedem.");

        var ativas = feriasDoFuncionario
            .Where(f => f.IdFuncionario == funcionario.Id &&
                        (f.Status == StatusSolicitacao.Pendente || f.Status == StatusSolicitacao.Aprovada))
            .ToList();

        if (ativas.Any(f => inicio <= f.DataFim && fim >= f.DataInicio))
            throw new DominioException("O período se sobrepõe a outra solicitação de férias do funcionário.");

        // Frações do mesmo período aquisitivo, já com a nova.
        var fracoes = ativas
            .Where(f => AnosCompletosDeCasa(funcionario.DataAdmissao, f.DataInicio) == periodo)
            .Select(f => f.Dias)
            .ToList();
        var jaUsados = fracoes.Sum();
        fracoes.Add(dias);

        var total = fracoes.Sum();
        if (total > DiasPorPeriodo)
            throw new DominioException($"Saldo insuficiente: restam {Math.Max(0, DiasPorPeriodo - jaUsados)} dias neste período aquisitivo.");
        if (fracoes.Count > MaximoFracoes)
            throw new DominioException($"As férias podem ser divididas em no máximo {MaximoFracoes} períodos.");

        // Uma das frações precisa ter 14 dias ou mais: já existe, ou ainda cabe uma fração com o saldo que sobra.
        var temFracaoPrincipal = fracoes.Max() >= MinimoDiasFracaoPrincipal;
        var aindaPossivel = fracoes.Count < MaximoFracoes && DiasPorPeriodo - total >= MinimoDiasFracaoPrincipal;
        if (!temFracaoPrincipal && !aindaPossivel)
            throw new DominioException(
                $"Ao dividir as férias, um dos períodos deve ter no mínimo {MinimoDiasFracaoPrincipal} dias.");

        return new Ferias
        {
            IdFuncionario = funcionario.Id,
            DataInicio = inicio,
            DataFim = fim,
            Dias = dias,
            Status = StatusSolicitacao.Pendente,
            DataSolicitacao = hoje,
            Observacao = string.IsNullOrWhiteSpace(observacao) ? null : observacao.Trim()
        };
    }

    /// <summary>
    /// Aprova ou rejeita a solicitação (RF27). Quem pode decidir (gestor do funcionário ou RH) é checado na API.
    /// Como a tabela FERIAS não guarda o decisor, ele é registrado no texto da observação (e no log de auditoria).
    /// </summary>
    public static void Decidir(Ferias ferias, bool aprovar, int idDecisor, string? observacao)
    {
        if (ferias.Status != StatusSolicitacao.Pendente)
            throw new DominioException("Esta solicitação de férias já foi decidida.");
        if (idDecisor <= 0) throw new DominioException("O responsável pela decisão é obrigatório.");
        if (idDecisor == ferias.IdFuncionario)
            throw new DominioException("Ninguém pode decidir a própria solicitação de férias.");

        var motivo = observacao?.Trim();
        if (!aprovar && (motivo is null || motivo.Length < TamanhoMinimoMotivoRejeicao))
            throw new DominioException("Informe o motivo da rejeição.");

        ferias.Status = aprovar ? StatusSolicitacao.Aprovada : StatusSolicitacao.Rejeitada;

        var registro = $"[{(aprovar ? "Aprovada" : "Rejeitada")} por {CodigoFuncionario.Formatar(idDecisor)}]" +
                       (string.IsNullOrEmpty(motivo) ? string.Empty : $" {motivo}");
        ferias.Observacao = string.IsNullOrEmpty(ferias.Observacao) ? registro : $"{ferias.Observacao}\n{registro}";
    }
}
```

### backend/src/HRAuto.Domain/Disciplina/ReguaConsequencias.cs

```csharp
using HRAuto.Domain.Entities;
using HRAuto.Domain.Enums;

namespace HRAuto.Domain.Disciplina;

/// <summary>Resultado da régua: o que o sistema sugere ao RH, com os números que levaram a isso.</summary>
public sealed record SugestaoConsequencia(
    AcaoSugerida Acao,
    int LevesDesdeUltimaSuspensao,
    int SuspensoesAtivas,
    string Mensagem);

/// <summary>
/// Régua de consequências automatizada (RF34):
/// - 3 advertências leves ativas desde a última suspensão = sugerir 1 suspensão;
/// - 2 suspensões ativas = sugerir justa causa (tem prioridade sobre a suspensão).
/// O sistema apenas SUGERE; o RH decide e registra a suspensão ou a justa causa como nova advertência.
/// Uma suspensão "consome" as leves anteriores a ela (a contagem das leves recomeça depois dela), por isso a
/// sugestão não se repete depois que o RH registra a suspensão. Advertências canceladas não contam.
/// A ordem é a da data da advertência (empate: ordem de cadastro).
/// </summary>
public static class ReguaConsequencias
{
    public const int LimiteLeves = 3;
    public const int LimiteSuspensoes = 2;

    public static SugestaoConsequencia Avaliar(IEnumerable<Advertencia> advertenciasDoFuncionario)
    {
        var leves = 0;
        var suspensoes = 0;
        var justaCausa = false;

        foreach (var advertencia in RegrasAdvertencia.Ativas(advertenciasDoFuncionario))
        {
            switch (RegrasAdvertencia.LerTipo(advertencia.Tipo))
            {
                case TipoAdvertencia.Leve:
                    leves++;
                    break;
                case TipoAdvertencia.Suspensao:
                    suspensoes++;
                    leves = 0; // a suspensão consome as leves anteriores
                    break;
                case TipoAdvertencia.JustaCausa:
                    justaCausa = true;
                    break;
            }
        }

        if (justaCausa)
            return new(AcaoSugerida.Nenhuma, leves, suspensoes, "Justa causa já aplicada: nenhuma nova consequência a sugerir.");

        if (suspensoes >= LimiteSuspensoes)
            return new(AcaoSugerida.JustaCausa, leves, suspensoes,
                $"Sugestão: avaliar justa causa ({suspensoes} suspensões ativas).");

        if (leves >= LimiteLeves)
            return new(AcaoSugerida.Suspensao, leves, suspensoes,
                $"Sugestão: aplicar suspensão ({leves} advertências leves desde a última suspensão).");

        return new(AcaoSugerida.Nenhuma, leves, suspensoes, "Nenhuma consequência a sugerir.");
    }
}
```

```csharp
/// <summary>
/// Cria as notificações geradas pelas regras de negócio (RF25, RF36). A API as grava; o app as lista.
/// A data/hora recebida deve vir do relógio do servidor (Brasília).
/// </summary>
public static class FabricaNotificacoes

    public static Notificacao HoraExtra(int idGestor, int idFuncionario, TimeSpan horasExtras, DateOnly dia, DateTime agora) =>

    public static Notificacao? SugestaoDaRegua(int idRh, int idFuncionario, SugestaoConsequencia sugestao, DateTime agora)
    {
        if (sugestao.Acao == AcaoSugerida.Nenhuma) return null;

        var titulo = sugestao.Acao == AcaoSugerida.Suspensao ? "Sugestão de suspensão" : "Sugestão de justa causa";
        return Criar(idRh, TiposNotificacao.Advertencia, titulo,
            $"{CodigoFuncionario.Formatar(idFuncionario)}: {sugestao.Mensagem}", agora);
    }

    /// <summary>RF36: informa o funcionário da decisão sobre a solicitação de férias.</summary>

        return Criar(ferias.IdFuncionario, TiposNotificacao.Ferias, $"Férias {resultado}",
            $"Sua solicitação de férias de {ferias.DataInicio:dd/MM/yyyy} a {ferias.DataFim:dd/MM/yyyy} foi {resultado}.", agora);
    }

    /// <summary>RF36: informa o funcionário da decisão sobre o pedido de manutenção de ponto.</summary>
    public static Notificacao DecisaoDeManutencao(ManutencaoPonto pedido, DateTime agora)
    {
        var resultado = ResultadoDaDecisao(pedido.Status);
        return Criar(pedido.IdFuncionario, TiposNotificacao.ManutencaoPonto, $"Manutenção de ponto {resultado}",
            $"Seu pedido de manutenção de ponto foi {resultado}.", agora);
    }

    // Aprovada -> "aprovada", Rejeitada -> "rejeitada". Pedido ainda pendente não gera notificação.
    private static string ResultadoDaDecisao(string status) => status switch
    {
        StatusSolicitacao.Aprovada => "aprovada",
        StatusSolicitacao.Rejeitada => "rejeitada",

            Tipo = tipo,
            Titulo = titulo,
            Mensagem = mensagem,
            DataEnvio = agora,
            Lida = false
        };
    }

    {
        var minutos = (int)Math.Round(duracao.TotalMinutes, MidpointRounding.AwayFromZero);
        return string.Create(CultureInfo.InvariantCulture, $"{minutos / 60}h{minutos % 60:D2}min");
    }
```
