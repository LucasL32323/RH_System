## Regras de segurança implementadas (revisar com o grupo)

- Hierarquia de perfis (`RegrasAcesso`): Admin gerencia/concede qualquer perfil; RH só gerencia contas Colaborador e só concede Colaborador. Impede o RH de trocar o e-mail de um Admin e tomar a conta pelo "esqueci minha senha".
- Ninguém altera o próprio acesso nem se desliga. O último Admin ativo não pode ser desativado, rebaixado nem desligado.
- Não se desliga quem ainda tem subordinados ativos (409).
- Conta criada pelo RH nasce com senha aleatória que ninguém conhece; o funcionário define a senha pelo link do convite (mesmo token do "esqueci minha senha", 30 min, uso único). Falha no e-mail não desfaz o cadastro; o RH reenvia.
- LGPD: CPF salvo só com dígitos, nunca aparece em lista nem em log; logs de edição trazem só nomes de campos.
- 403 antes de 404 para quem não tem relação com o funcionário (não revela quais códigos existem).
- Perfil trocado vale no próximo login (JWT já emitido dura até 8 h).

```csharp
namespace HRAuto.Domain.ValueObjects;

/// <summary>
/// CPF (RF10, RNF08). Dado pessoal: no banco fica só com os 11 dígitos; nas listagens aparece mascarado.
/// A validação confere o tamanho e os dois dígitos verificadores (módulo 11).
/// </summary>
public static class Cpf
{
    public const int Tamanho = 11;

    /// <summary>Aceita com ou sem máscara (123.456.789-09). Devolve só os 11 dígitos. Retorna false se for inválido.</summary>
    public static bool TentarNormalizar(string? texto, out string digitos)
    {
        digitos = string.Empty;
        if (string.IsNullOrWhiteSpace(texto)) return false;

        var limpo = new System.Text.StringBuilder(Tamanho);
        foreach (var c in texto.Trim())
        {
            if (c is >= '0' and <= '9') limpo.Append(c);
            else if (c is '.' or '-' or ' ') continue; // só estes símbolos de máscara são tolerados
            else return false;
        }

        var d = limpo.ToString();
        if (d.Length != Tamanho) return false;
        if (d.All(c => c == d[0])) return false; // 111.111.111-11 passa na conta, mas não existe

        if (DigitoVerificador(d, 9) != d[9] - '0') return false;
        if (DigitoVerificador(d, 10) != d[10] - '0') return false;

        digitos = d;
        return true;
    }

    /// <summary>12345678909 -> 123.456.789-09. Entrada deve ter 11 dígitos.</summary>
    public static string Formatar(string digitos) =>
        digitos.Length == Tamanho
            ? $"{digitos[..3]}.{digitos[3..6]}.{digitos[6..9]}-{digitos[9..]}"
            : digitos;

    /// <summary>12345678909 -> ***.456.789-** (LGPD: listagens não mostram o CPF inteiro).</summary>
    public static string Mascarar(string digitos) =>
        digitos.Length == Tamanho ? $"***.{digitos[3..6]}.{digitos[6..9]}-**" : "***";

    // Soma ponderada dos 'quantos' primeiros dígitos, pesos de quantos+1 até 2.
    private static int DigitoVerificador(string d, int quantos)
    {
        var soma = 0;
        for (var i = 0; i < quantos; i++) soma += (d[i] - '0') * (quantos + 1 - i);
        var resto = soma * 10 % 11;
        return resto == 10 ? 0 : resto;
    }
}
```

### backend/src/HRAuto.Domain/Seguranca/GeradorSenhaTemporaria.cs

```csharp
using System.Security.Cryptography;

namespace HRAuto.Domain.Seguranca;

/// <summary>
/// Senha aleatória para contas criadas pelo RH (RF06). Ninguém a conhece nem a recebe: ela só existe para a conta
/// nunca ficar sem senha; o funcionário define a própria senha pelo link de convite (mesmo fluxo do "esqueci minha senha").
/// Atende a <see cref="PoliticaSenha"/> e usa gerador criptográfico.
/// </summary>
public static class GeradorSenhaTemporaria
{
    private const string Maiusculas = "ABCDEFGHJKLMNPQRSTUVWXYZ";
    private const string Minusculas = "abcdefghijkmnpqrstuvwxyz";
    private const string Numeros = "23456789";
    private const string Simbolos = "!@#$%&*?";
    public const int Tamanho = 24;

    public static string Gerar()
    {
        var todos = Maiusculas + Minusculas + Numeros + Simbolos;
        var caracteres = new List<char>(Tamanho)
        {
            Sortear(Maiusculas), Sortear(Minusculas), Sortear(Numeros), Sortear(Simbolos)
        };
        while (caracteres.Count < Tamanho) caracteres.Add(Sortear(todos));

        // Embaralha (Fisher-Yates) para os quatro primeiros não ficarem sempre no começo.
        for (var i = caracteres.Count - 1; i > 0; i--)
        {
            var j = RandomNumberGenerator.GetInt32(i + 1);
            (caracteres[i], caracteres[j]) = (caracteres[j], caracteres[i]);
        }
        return new string(caracteres.ToArray());
    }

    private static char Sortear(string alfabeto) => alfabeto[RandomNumberGenerator.GetInt32(alfabeto.Length)];
}
```

```csharp
        if (nome.Any(char.IsControl)) throw new DominioException("O nome contém caracteres inválidos.");

    private static string Texto(string? valor, string campo, int minimo, int maximo)
    {
        var texto = string.Join(' ', (valor ?? string.Empty).Split((char[]?)null, StringSplitOptions.RemoveEmptyEntries));
        if (texto.Length < minimo)
            throw new DominioException(minimo <= 1 ? $"{campo} é obrigatório." : $"{campo} é obrigatório (mínimo de {minimo} caracteres).");

    private static string SomenteDigitos(string? valor) =>
        new((valor ?? string.Empty).Where(c => c is >= '0' and <= '9').ToArray());
```

```csharp
/// <summary>Pedido de página de uma listagem. Padrão: página 1 com 20 itens; máximo de 100 por página.</summary>
public sealed record Paginacao
{
    public const int TamanhoPadrao = 20;
    public const int TamanhoMaximo = 100;

    public int Pagina { get; }
    public int Tamanho { get; }

    /// <summary>Quantos itens pular antes da página pedida.</summary>
    public int Pular => (Pagina - 1) * Tamanho;

    private Paginacao(int pagina, int tamanho)
    {
        Pagina = pagina;
        Tamanho = tamanho;
    }

    /// <summary>Corrige valores ausentes ou fora da faixa em vez de falhar (página mínima 1, tamanho entre 1 e 100).</summary>
    public static Paginacao De(int? pagina, int? tamanho) =>
        new(Math.Max(1, pagina ?? 1), Math.Clamp(tamanho ?? TamanhoPadrao, 1, TamanhoMaximo));
}
```

### backend/src/HRAuto.Application/Cadastro/CadastrarFuncionario.cs

```csharp
using HRAuto.Application.Abstractions;
using HRAuto.Application.Common;
using HRAuto.Domain.Cadastro;
using HRAuto.Domain.Common;
using HRAuto.Domain.Entities;
using HRAuto.Domain.Enums;
using HRAuto.Domain.Seguranca;
using HRAuto.Domain.ValueObjects;

namespace HRAuto.Application.Cadastro;

/// <summary>
/// RF09, RF10, RF16: o RH (ou Admin) cadastra o funcionário com CPF, RG e residência, define cargo, departamento e gestor
/// e, por padrão, cria o acesso. O código FUNC-0001 nasce do id gerado pelo banco.
/// Tudo é gravado numa transação: funcionário (gera o id), detalhes, usuário e log de auditoria.
/// </summary>
public sealed class CadastrarFuncionario
{
    private readonly IFuncionarioCadastroRepositorio _funcionarios;
    private readonly IUsuarioGestaoRepositorio _usuarios;
    private readonly ICargoRepositorio _cargos;
    private readonly IDepartamentoRepositorio _departamentos;
    private readonly IHashSenha _hash;
    private readonly EnviadorDeConvite _convite;
    private readonly IAuditoria _auditoria;
    private readonly IUnidadeDeTrabalho _unidade;
    private readonly IRelogio _relogio;

    public CadastrarFuncionario(
        IFuncionarioCadastroRepositorio funcionarios,
        IUsuarioGestaoRepositorio usuarios,
        ICargoRepositorio cargos,
        IDepartamentoRepositorio departamentos,
        IHashSenha hash,
        EnviadorDeConvite convite,
        IAuditoria auditoria,
        IUnidadeDeTrabalho unidade,
        IRelogio relogio)
    {
        _funcionarios = funcionarios;
        _usuarios = usuarios;
        _cargos = cargos;
        _departamentos = departamentos;
        _hash = hash;
        _convite = convite;
        _auditoria = auditoria;
        _unidade = unidade;
        _relogio = relogio;
    }

    public async Task<Resultado<FuncionarioCadastradoDto>> ExecutarAsync(
        Solicitante solicitante, CadastrarFuncionarioDto? dto, CancellationToken cancelamento = default)
    {
        if (!solicitante.Perfil.EhRhOuAdmin())
            return Resultado<FuncionarioCadastradoDto>.Falha(TipoErro.Proibido, "Apenas o RH pode cadastrar funcionários.");
        if (dto is null)
            return Resultado<FuncionarioCadastradoDto>.Falha("Envie os dados do funcionário.");

        // Perfil e acesso
        var criarAcesso = dto.CriarAcesso ?? true;
        if (!MapeadorCadastro.TentarLerPerfil(dto.Perfil, out var perfil))
            return Resultado<FuncionarioCadastradoDto>.Falha("Perfil inválido. Use Colaborador, RH ou Admin.");
        if (criarAcesso && !RegrasAcesso.PodeConceder(solicitante.Perfil, perfil))
            return Resultado<FuncionarioCadastradoDto>.Falha(TipoErro.Proibido, $"Você não pode conceder o perfil {NomesPerfil.De(perfil)}.");

        if (!MapeadorCadastro.TentarLerGestor(dto.CodigoGestor, out var idGestor))
            return Resultado<FuncionarioCadastradoDto>.Falha("Código do gestor inválido. Use o formato FUNC-0001.");

        // Validação do Domain (CPF, datas, e-mail, endereço...)
        DadosCadastroFuncionario dados;
        try
        {
            var entrada = MapeadorCadastro.Entrada(
                dto.Nome, dto.Cpf, dto.Rg, dto.Pis, dto.Ctps, dto.DataNascimento, dto.DataAdmissao,
                dto.Email, dto.Telefone, dto.IdCargo, dto.IdDepartamento, idGestor, dto.Endereco);
            dados = RegrasFuncionario.Normalizar(entrada, _relogio.Hoje());
        }
        catch (DominioException e)
        {
            return Resultado<FuncionarioCadastradoDto>.Falha(e.Message);
        }

        // Existência dos vínculos
        var cargo = await _cargos.ObterAsync(dados.IdCargo, cancelamento);
        if (cargo is null) return Resultado<FuncionarioCadastradoDto>.Falha("O cargo informado não existe.");
        var departamento = await _departamentos.ObterAsync(dados.IdDepartamento, cancelamento);
        if (departamento is null) return Resultado<FuncionarioCadastradoDto>.Falha("O departamento informado não existe.");

        if (dados.IdGestor is { } idDoGestor)
        {
            var gestor = await _funcionarios.ObterAsync(idDoGestor, cancelamento);
            if (gestor is null) return Resultado<FuncionarioCadastradoDto>.Falha("O gestor informado não existe.");
            try
            {
                RegrasFuncionario.ValidarGestor(0, gestor, Array.Empty<int>());
            }
            catch (DominioException e)
            {
                return Resultado<FuncionarioCadastradoDto>.Falha(e.Message);
            }
        }

        // Unicidade (a checagem no banco existe só para o e-mail do usuário; o resto é regra desta camada)
        if (await _funcionarios.CpfJaCadastradoAsync(dados.Cpf, null, cancelamento))
            return Resultado<FuncionarioCadastradoDto>.Falha(TipoErro.Conflito, "Já existe um funcionário com este CPF.");
        if (await _funcionarios.EmailJaCadastradoAsync(dados.Email, null, cancelamento)
            || await _usuarios.EmailEmUsoAsync(dados.Email, null, cancelamento))
            return Resultado<FuncionarioCadastradoDto>.Falha(TipoErro.Conflito, "Já existe um funcionário ou usuário com este e-mail.");

        var funcionario = RegrasFuncionario.Criar(dados);
        _funcionarios.Adicionar(funcionario);

        Usuario? usuario = null;
        try
        {
            await using var transacao = await _unidade.IniciarTransacaoAsync(cancelamento);

            await _unidade.SalvarAsync(cancelamento); // gera o id do funcionário

            _funcionarios.AdicionarDetalhes(RegrasFuncionario.CriarDetalhes(funcionario.Id, dados));

            if (criarAcesso)
            {
                usuario = new Usuario
                {
                    IdFuncionario = funcionario.Id,
                    IdPerfil = (int)perfil,
                    Email = dados.Email,
                    Senha = _hash.Gerar(GeradorSenhaTemporaria.Gerar()), // ninguém conhece: o funcionário define a dele pelo convite
                    Ativo = true
                };
                _usuarios.Adicionar(usuario);
            }

            // Sem CPF, RG nem endereço no log: LGPD (RNF08).
            _auditoria.Registrar(
                solicitante.IdUsuario, "CadastrarFuncionario", "FUNCIONARIO", funcionario.Id,
                $"{funcionario.Codigo} cadastrado" + (criarAcesso ? $" com acesso {NomesPerfil.De(perfil)}" : " sem acesso"));
            await _unidade.SalvarAsync(cancelamento);

            await transacao.ConfirmarAsync(cancelamento);
        }
        catch (ConflitoDePersistenciaException)
        {
            return Resultado<FuncionarioCadastradoDto>.Falha(
                TipoErro.Conflito, "Já existe um usuário com este e-mail. Confira os dados e tente novamente.");
        }

        // O convite sai depois da gravação: se o e-mail falhar, o cadastro já está salvo e o RH pode reenviar.
        var conviteEnviado = false;
        if (usuario is not null)
        {
            try
            {
                await _convite.EnviarAsync(usuario, funcionario.Nome, cancelamento);
                conviteEnviado = true;
            }
            catch (Exception) when (!cancelamento.IsCancellationRequested)
            {
                conviteEnviado = false;
            }
        }

        return Resultado<FuncionarioCadastradoDto>.Ok(new FuncionarioCadastradoDto(
            funcionario.Codigo, funcionario.Nome, usuario is not null,
            usuario is null ? null : NomesPerfil.De(perfil), conviteEnviado));
    }
}
```

```csharp
{

    private readonly IUsuarioGestaoRepositorio _usuarios;
```

```csharp
        Solicitante solicitante, string? codigoTexto, AtualizarFuncionarioDto? dto, CancellationToken cancelamento = default)
    {
        if (!solicitante.Perfil.EhRhOuAdmin())
            return Resultado<FuncionarioResumoDto>.Falha(TipoErro.Proibido, "Apenas o RH pode editar funcionários.");
        if (!CodigoFuncionario.TentarConverter(codigoTexto, out var codigo))

        if (dto is null)
            return Resultado<FuncionarioResumoDto>.Falha("Envie os dados do funcionário.");
        if (!MapeadorCadastro.TentarLerGestor(dto.CodigoGestor, out var idGestor))
            return Resultado<FuncionarioResumoDto>.Falha("Código do gestor inválido. Use o formato FUNC-0001.");

        if (usuario is not null && !RegrasAcesso.PodeGerenciar(solicitante.Perfil, (PerfilAcesso)usuario.IdPerfil))
            return Resultado<FuncionarioResumoDto>.Falha(TipoErro.Proibido, "Você não pode alterar os dados de um usuário RH ou Admin.");

        DadosCadastroFuncionario dados;
        try
        {
            var entrada = MapeadorCadastro.Entrada(
                dto.Nome, dto.Cpf, dto.Rg, dto.Pis, dto.Ctps, dto.DataNascimento, dto.DataAdmissao,
                dto.Email, dto.Telefone, dto.IdCargo, dto.IdDepartamento, idGestor, dto.Endereco);

                {
                    var cadeia = await _funcionarios.ObterCadeiaDeGestoresAsync(idDoGestor, cancelamento);
                    RegrasFuncionario.ValidarGestor(funcionario.Id, gestor, cadeia);
                }
                catch (DominioException e)
                {
                    return Resultado<FuncionarioResumoDto>.Falha(e.Message);
                }
            }

        if (await _funcionarios.CpfJaCadastradoAsync(dados.Cpf, funcionario.Id, cancelamento))

        var alterados = RegrasFuncionario.CamposAlterados(funcionario, detalhes, dados);

        if (alterados.Count > 0)
        {
            RegrasFuncionario.Aplicar(funcionario, dados);
            if (detalhes is null)
                _funcionarios.AdicionarDetalhes(RegrasFuncionario.CriarDetalhes(funcionario.Id, dados));
            else
                RegrasFuncionario.AplicarDetalhes(detalhes, dados);

            if (usuario is not null) usuario.Email = dados.Email; // o login acompanha o e-mail do cadastro

            _auditoria.Registrar(
                solicitante.IdUsuario, "AtualizarFuncionario", "FUNCIONARIO", funcionario.Id,
                $"{funcionario.Codigo}: {string.Join(", ", alterados)}");
            try
            {
                await _unidade.SalvarAsync(cancelamento);
            }
            catch (ConflitoDePersistenciaException)
            {
                return Resultado<FuncionarioResumoDto>.Falha(TipoErro.Conflito, "Já existe um usuário com este e-mail.");
            }
        }

        // Nomes montados a partir do que foi carregado (não das propriedades de navegação, que o EF só corrige ao gravar).
        return Resultado<FuncionarioResumoDto>.Ok(new FuncionarioResumoDto(
            funcionario.Codigo, funcionario.Nome, funcionario.Email, cargo.NomeCargo, departamento.Nome,
            funcionario.IdGestor is { } g ? CodigoFuncionario.Formatar(g) : null, gestor?.Nome,
            funcionario.Ativo, funcionario.DataAdmissao));
    }
```

```csharp
/// RF12: desligamento e reativação. O cadastro nunca é apagado (RNF07): o desligamento grava a data, inativa o funcionário
/// e bloqueia o acesso. Reativar o funcionário NÃO devolve o acesso: o RH reativa o login em separado (evita volta silenciosa).
/// Ninguém desliga a si mesmo; o RH não desliga RH/Admin; não se desliga quem ainda tem subordinados ativos
/// nem o último Admin ativo.
/// </summary>
public sealed class AlterarSituacaoFuncionario
{
    private readonly IFuncionarioCadastroRepositorio _funcionarios;
    private readonly IUsuarioGestaoRepositorio _usuarios;
    private readonly IAuditoria _auditoria;
    private readonly IUnidadeDeTrabalho _unidade;
    private readonly IRelogio _relogio;

    public AlterarSituacaoFuncionario(
        IFuncionarioCadastroRepositorio funcionarios,
        IUsuarioGestaoRepositorio usuarios,
        IAuditoria auditoria,
        IUnidadeDeTrabalho unidade,
        IRelogio relogio)

        _usuarios = usuarios;
        _auditoria = auditoria;
        _unidade = unidade;
        _relogio = relogio;
    }

    public async Task<Resultado> DesligarAsync(
        Solicitante solicitante, string? codigoTexto, DesligarFuncionarioDto? dto, CancellationToken cancelamento = default)
    {
        var preparo = await PrepararAsync(solicitante, codigoTexto, cancelamento);
        if (preparo.Falha is not null) return preparo.Falha;
        var (funcionario, usuario) = (preparo.Funcionario!, preparo.Usuario);

        if (funcionario.Ativo)
        {
            var subordinados = await _funcionarios.ContarSubordinadosAtivosAsync(funcionario.Id, cancelamento);
            if (subordinados > 0)
                return Resultado.Falha(TipoErro.Conflito,
                    $"{funcionario.Nome} ainda é gestor de {subordinados} funcionário(s) ativo(s). Troque o gestor deles antes de desligar.");

            if (usuario is { Ativo: true } && (PerfilAcesso)usuario.IdPerfil == PerfilAcesso.Admin
                && await _usuarios.ContarAdminsAtivosAsync(cancelamento) <= 1)
                return Resultado.Falha(TipoErro.Conflito, "Este é o último administrador ativo; ele não pode ser desligado.");
        }

        try
        {
            RegrasFuncionario.Desligar(funcionario, dto?.Data ?? _relogio.Hoje(), _relogio.Hoje());
        }
        catch (DominioException e)
        {
            return Resultado.Falha(e.Message);
        }

        if (usuario is not null) usuario.Ativo = false; // sem funcionário ativo não há acesso

        _auditoria.Registrar(
            solicitante.IdUsuario, "DesligarFuncionario", "FUNCIONARIO", funcionario.Id,
            $"{funcionario.Codigo} desligado em {funcionario.DataDesligamento:dd/MM/yyyy}");
        await _unidade.SalvarAsync(cancelamento);
        return Resultado.Ok();
    }

    public async Task<Resultado> ReativarAsync(
        Solicitante solicitante, string? codigoTexto, CancellationToken cancelamento = default)
    {
        var preparo = await PrepararAsync(solicitante, codigoTexto, cancelamento);
        if (preparo.Falha is not null) return preparo.Falha;
        var funcionario = preparo.Funcionario!;

        try
        {
            RegrasFuncionario.Reativar(funcionario);
        }
        catch (DominioException e)
        {
            return Resultado.Falha(TipoErro.Conflito, e.Message);
        }

        _auditoria.Registrar(
            solicitante.IdUsuario, "ReativarFuncionario", "FUNCIONARIO", funcionario.Id, $"{funcionario.Codigo} reativado");
        await _unidade.SalvarAsync(cancelamento);
        return Resultado.Ok();
    }

    private sealed record Preparo(Domain.Entities.Funcionario? Funcionario, Domain.Entities.Usuario? Usuario, Resultado? Falha);

    /// <summary>Verificações comuns: perfil, código, existência, "não em si mesmo" e hierarquia de perfis.</summary>
    private async Task<Preparo> PrepararAsync(Solicitante solicitante, string? codigoTexto, CancellationToken cancelamento)
    {
        if (!solicitante.Perfil.EhRhOuAdmin())
            return new(null, null, Resultado.Falha(TipoErro.Proibido, "Apenas o RH pode alterar a situação de funcionários."));
        if (!CodigoFuncionario.TentarConverter(codigoTexto, out var codigo))
            return new(null, null, Resultado.Falha("Código inválido. Use o formato FUNC-0001."));
        if (codigo.Id == solicitante.IdFuncionario)
            return new(null, null, Resultado.Falha(TipoErro.Proibido, "Você não pode alterar a sua própria situação."));

        var funcionario = await _funcionarios.ObterParaEdicaoAsync(codigo.Id, cancelamento);
        if (funcionario is null)
            return new(null, null, Resultado.Falha(TipoErro.NaoEncontrado, "Funcionário não encontrado."));

        var usuario = await _usuarios.ObterPorFuncionarioAsync(funcionario.Id, cancelamento);
        if (usuario is not null && !RegrasAcesso.PodeGerenciar(solicitante.Perfil, (PerfilAcesso)usuario.IdPerfil))
            return new(null, null, Resultado.Falha(TipoErro.Proibido, "Você não pode alterar um usuário RH ou Admin."));

        return new(funcionario, usuario, null);
    }
```

```csharp
            return Resultado<AcessoDto>.Falha(TipoErro.Proibido, "Apenas o RH pode consultar acessos.");
        if (!CodigoFuncionario.TentarConverter(codigoTexto, out var codigo))
            return Resultado<AcessoDto>.Falha("Código inválido. Use o formato FUNC-0001.");
        if (await _funcionarios.ObterAsync(codigo.Id, cancelamento) is null)
            return Resultado<AcessoDto>.Falha(TipoErro.NaoEncontrado, "Funcionário não encontrado.");

        return Resultado<AcessoDto>.Ok(MapeadorCadastro.Acesso(await _usuarios.ObterPorFuncionarioAsync(codigo.Id, cancelamento)));
    }

    /// <summary>Cria o login de um funcionário que ainda não tem (e-mail do cadastro) e envia o convite.</summary>
    public async Task<Resultado<AcessoCriadoDto>> CriarAsync(
        Solicitante solicitante, string? codigoTexto, CriarAcessoDto? dto, CancellationToken cancelamento = default)
    {
        if (!solicitante.Perfil.EhRhOuAdmin())
            return Resultado<AcessoCriadoDto>.Falha(TipoErro.Proibido, "Apenas o RH pode criar acessos.");
        if (!CodigoFuncionario.TentarConverter(codigoTexto, out var codigo))
            return Resultado<AcessoCriadoDto>.Falha("Código inválido. Use o formato FUNC-0001.");
        if (codigo.Id == solicitante.IdFuncionario)
            return Resultado<AcessoCriadoDto>.Falha(TipoErro.Proibido, "Você não pode alterar o seu próprio acesso.");
        if (!MapeadorCadastro.TentarLerPerfil(dto?.Perfil, out var perfil))
            return Resultado<AcessoCriadoDto>.Falha("Perfil inválido. Use Colaborador, RH ou Admin.");
        if (!RegrasAcesso.PodeConceder(solicitante.Perfil, perfil))
            return Resultado<AcessoCriadoDto>.Falha(TipoErro.Proibido, $"Você não pode conceder o perfil {NomesPerfil.De(perfil)}.");

        var funcionario = await _funcionarios.ObterAsync(codigo.Id, cancelamento);
        if (funcionario is null)
            return Resultado<AcessoCriadoDto>.Falha(TipoErro.NaoEncontrado, "Funcionário não encontrado.");
        if (!funcionario.Ativo)
            return Resultado<AcessoCriadoDto>.Falha(TipoErro.Conflito, "O funcionário está desligado.");
        if (await _usuarios.ObterPorFuncionarioAsync(funcionario.Id, cancelamento) is not null)
            return Resultado<AcessoCriadoDto>.Falha(TipoErro.Conflito, "Este funcionário já tem acesso.");
        if (await _usuarios.EmailEmUsoAsync(funcionario.Email, null, cancelamento))
            return Resultado<AcessoCriadoDto>.Falha(TipoErro.Conflito, "Já existe um usuário com o e-mail deste funcionário.");

        var usuario = new Usuario
        {
            IdFuncionario = funcionario.Id,
            IdPerfil = (int)perfil,
            Email = funcionario.Email,
            Senha = _hash.Gerar(GeradorSenhaTemporaria.Gerar()),
            Ativo = true
        };
        _usuarios.Adicionar(usuario);

        try
        {
            await _unidade.SalvarAsync(cancelamento); // gera o id do usuário
            _auditoria.Registrar(
                solicitante.IdUsuario, "CriarAcesso", "USUARIO", usuario.Id,
                $"{funcionario.Codigo}: acesso {NomesPerfil.De(perfil)} criado");
            await _unidade.SalvarAsync(cancelamento);
        }
        catch (ConflitoDePersistenciaException)
        {
            return Resultado<AcessoCriadoDto>.Falha(TipoErro.Conflito, "Já existe um usuário com o e-mail deste funcionário.");
        }

        var enviado = await TentarEnviarConviteAsync(usuario, funcionario.Nome, cancelamento);
        return Resultado<AcessoCriadoDto>.Ok(new AcessoCriadoDto(funcionario.Codigo, NomesPerfil.De(perfil), enviado));
    }

    /// <summary>Troca o perfil e/ou ativa e desativa o login. Campos nulos não mudam nada.</summary>
    public async Task<Resultado<AcessoDto>> AlterarAsync(
        Solicitante solicitante, string? codigoTexto, AlterarAcessoDto? dto, CancellationToken cancelamento = default)
    {
        var preparo = await PrepararAsync(solicitante, codigoTexto, cancelamento);
        if (preparo.Falha is not null) return Resultado<AcessoDto>.Falha(preparo.Tipo, preparo.Falha);
        var (funcionario, usuario) = (preparo.Funcionario!, preparo.Usuario!);

        if (dto is null || (dto.Perfil is null && dto.Ativo is null))
            return Resultado<AcessoDto>.Falha("Informe o perfil e/ou a situação do acesso.");

        var perfilAtual = (PerfilAcesso)usuario.IdPerfil;
        var novoPerfil = perfilAtual;
        if (dto.Perfil is not null)
        {
            if (!PerfilAcessoExtensoes.TentarLer(dto.Perfil.Trim(), out novoPerfil))
                return Resultado<AcessoDto>.Falha("Perfil inválido. Use Colaborador, RH ou Admin.");
            if (!RegrasAcesso.PodeConceder(solicitante.Perfil, novoPerfil))
                return Resultado<AcessoDto>.Falha(TipoErro.Proibido, $"Você não pode conceder o perfil {NomesPerfil.De(novoPerfil)}.");
        }

        var novoAtivo = dto.Ativo ?? usuario.Ativo;
        if (novoAtivo && !funcionario.Ativo)
            return Resultado<AcessoDto>.Falha(TipoErro.Conflito, "O funcionário está desligado: reative o funcionário antes do acesso.");

        // O último Admin ativo precisa continuar Admin e ativo.
        var deixaDeSerAdminAtivo = usuario.Ativo && perfilAtual == PerfilAcesso.Admin
                                   && (novoPerfil != PerfilAcesso.Admin || !novoAtivo);
        if (deixaDeSerAdminAtivo && await _usuarios.ContarAdminsAtivosAsync(cancelamento) <= 1)
            return Resultado<AcessoDto>.Falha(TipoErro.Conflito, "Este é o último administrador ativo; ele não pode ser rebaixado nem desativado.");

        var mudancas = new List<string>();
        if (novoPerfil != perfilAtual)
            mudancas.Add($"perfil {NomesPerfil.De(perfilAtual)} -> {NomesPerfil.De(novoPerfil)}");
        if (novoAtivo != usuario.Ativo)
            mudancas.Add(novoAtivo ? "acesso ativado" : "acesso desativado");

        if (mudancas.Count > 0)
        {
            usuario.IdPerfil = (int)novoPerfil;
            usuario.Ativo = novoAtivo;
            _auditoria.Registrar(
                solicitante.IdUsuario, "AlterarAcesso", "USUARIO", usuario.Id, $"{funcionario.Codigo}: {string.Join("; ", mudancas)}");
            await _unidade.SalvarAsync(cancelamento);
        }

        return Resultado<AcessoDto>.Ok(MapeadorCadastro.Acesso(usuario));
    }

    /// <summary>Reenvia o link para o funcionário definir (ou refazer) a senha. Só para contas que o solicitante pode gerenciar.</summary>
    public async Task<Resultado> ReenviarConviteAsync(
        Solicitante solicitante, string? codigoTexto, CancellationToken cancelamento = default)
    {
        var preparo = await PrepararAsync(solicitante, codigoTexto, cancelamento);
        if (preparo.Falha is not null) return Resultado.Falha(preparo.Tipo, preparo.Falha);
        var (funcionario, usuario) = (preparo.Funcionario!, preparo.Usuario!);

        if (!usuario.Ativo || !funcionario.Ativo)
            return Resultado.Falha(TipoErro.Conflito, "O acesso está desativado.");

        _auditoria.Registrar(solicitante.IdUsuario, "ReenviarConvite", "USUARIO", usuario.Id, funcionario.Codigo);
        await _unidade.SalvarAsync(cancelamento);

        return await TentarEnviarConviteAsync(usuario, funcionario.Nome, cancelamento)
            ? Resultado.Ok()
            : Resultado.Falha(TipoErro.Conflito, "Não foi possível enviar o e-mail agora. Tente novamente em instantes.");
    }

    private async Task<bool> TentarEnviarConviteAsync(Usuario usuario, string nome, CancellationToken cancelamento)
    {
        try
        {
            await _convite.EnviarAsync(usuario, nome, cancelamento);
            return true;
        }
        catch (Exception) when (!cancelamento.IsCancellationRequested)
        {
            return false; // o acesso já foi gravado; o RH pode reenviar
        }
    }

    private sealed record Preparo(Funcionario? Funcionario, Usuario? Usuario, string? Falha, TipoErro Tipo);

    private async Task<Preparo> PrepararAsync(Solicitante solicitante, string? codigoTexto, CancellationToken cancelamento)
    {
        if (!solicitante.Perfil.EhRhOuAdmin())
            return new(null, null, "Apenas o RH pode gerenciar acessos.", TipoErro.Proibido);
        if (!CodigoFuncionario.TentarConverter(codigoTexto, out var codigo))
            return new(null, null, "Código inválido. Use o formato FUNC-0001.", TipoErro.Validacao);
        if (codigo.Id == solicitante.IdFuncionario)
            return new(null, null, "Você não pode alterar o seu próprio acesso.", TipoErro.Proibido);

        var funcionario = await _funcionarios.ObterAsync(codigo.Id, cancelamento);
        if (funcionario is null)
            return new(null, null, "Funcionário não encontrado.", TipoErro.NaoEncontrado);

        var usuario = await _usuarios.ObterPorFuncionarioAsync(funcionario.Id, cancelamento);
        if (usuario is null)
            return new(null, null, "Este funcionário ainda não tem acesso.", TipoErro.NaoEncontrado);
        if (!RegrasAcesso.PodeGerenciar(solicitante.Perfil, (PerfilAcesso)usuario.IdPerfil))
            return new(null, null, "Você não pode gerenciar o acesso de um usuário RH ou Admin.", TipoErro.Proibido);

        return new(funcionario, usuario, null, TipoErro.Validacao);
    }
}
```

```csharp
    public static bool TentarLerTipo(string? texto, out TipoAdvertencia tipo)
    {
        tipo = default;
        var t = texto?.Trim();
        if (string.IsNullOrEmpty(t) || char.IsDigit(t[0]) || t[0] == '-') return false;
        return Enum.TryParse(t, ignoreCase: true, out tipo) && Enum.IsDefined(tipo);
    }
```

```csharp
    {
        if (!solicitante.Perfil.EhRhOuAdmin())
            return Resultado<AdvertenciaRegistradaDto>.Falha(TipoErro.Proibido, "Apenas o RH pode registrar advertências.");
        if (dto is null)
            return Resultado<AdvertenciaRegistradaDto>.Falha("Envie os dados da advertência.");
        if (!CodigoFuncionario.TentarConverter(dto.CodigoFuncionario, out var codigo))
            return Resultado<AdvertenciaRegistradaDto>.Falha("Código do funcionário inválido. Use o formato FUNC-0001.");
        if (!MapeadorDisciplina.TentarLerTipo(dto.Tipo, out var tipo))
            return Resultado<AdvertenciaRegistradaDto>.Falha("Tipo inválido. Use Leve, Suspensao ou JustaCausa.");

        var funcionario = await _funcionarios.ObterAsync(codigo.Id, cancelamento);
        if (funcionario is null)
            return Resultado<AdvertenciaRegistradaDto>.Falha(TipoErro.NaoEncontrado, "Funcionário não encontrado.");
        if (!funcionario.Ativo)
            return Resultado<AdvertenciaRegistradaDto>.Falha(TipoErro.Conflito, "O funcionário está desligado.");

        var usuarioAlvo = await _usuarios.ObterPorFuncionarioAsync(funcionario.Id, cancelamento);
        if (usuarioAlvo is not null && !RegrasAcesso.PodeGerenciar(solicitante.Perfil, (PerfilAcesso)usuarioAlvo.IdPerfil))
            return Resultado<AdvertenciaRegistradaDto>.Falha(TipoErro.Proibido, "Apenas um Admin pode advertir um usuário RH ou Admin.");

        var agora = _relogio.Agora();
        var hoje = _relogio.Hoje();

        Advertencia nova;
        IReadOnlyList<Advertencia> todas;
        SugestaoConsequencia regua;
        ResultadoSemaforo semaforo;
        try
        {
            nova = RegrasAdvertencia.Registrar(
                funcionario.Id, solicitante.IdFuncionario, tipo, dto.Motivo, dto.Descricao, dto.Data ?? hoje, hoje);

            var existentes = await _advertencias.ListarDoFuncionarioAsync(funcionario.Id, cancelamento);
            todas = existentes.Append(nova).ToList();
            regua = ReguaConsequencias.Avaliar(todas);
            semaforo = SemaforoAdvertencias.Calcular(todas);
        }
        catch (DominioException e)
        {
            return Resultado<AdvertenciaRegistradaDto>.Falha(e.Message);
        }

        _advertencias.Adicionar(nova);
        _notificacoes.Adicionar(NotificacoesDisciplina.AdvertenciaRegistrada(nova, agora.DateTime));

        // Avisa o RH quando a régua sugere suspensão ou justa causa. Ao registrar a própria justa causa não há mais o que sugerir.
        if (tipo != TipoAdvertencia.JustaCausa)
        {
            var destinatarios = (await _usuarios.ListarIdsFuncionariosRhAtivosAsync(cancelamento))
                .Append(solicitante.IdFuncionario)
                .Where(id => id != funcionario.Id)
                .Distinct();

            }

        await _unidade.SalvarAsync(cancelamento); // gera o id da advertência
        _auditoria.Registrar(
            solicitante.IdUsuario, "RegistrarAdvertencia", "ADVERTENCIA", nova.Id,
            $"{funcionario.Codigo}: {tipo} (semáforo {semaforo.Nivel})");

        await transacao.ConfirmarAsync(cancelamento);

        var nomes = new Dictionary<int, string> { [funcionario.Id] = funcionario.Nome };
        if (!nomes.ContainsKey(solicitante.IdFuncionario))
        {
            var responsavel = await _funcionarios.ObterNomesAsync(new[] { solicitante.IdFuncionario }, cancelamento);
            if (responsavel.TryGetValue(solicitante.IdFuncionario, out var nomeResponsavel))
                nomes[solicitante.IdFuncionario] = nomeResponsavel;
        }

        return Resultado<AdvertenciaRegistradaDto>.Ok(new AdvertenciaRegistradaDto(
            MapeadorDisciplina.Advertencia(nova, nomes), MapeadorDisciplina.Semaforo(semaforo), MapeadorDisciplina.Regua(regua)));
```

```csharp
            funcionario.Codigo, funcionario.Nome, semaforo, regua,
            advertencias.Select(a => MapeadorDisciplina.Advertencia(a, nomes)).ToList()));
    }
```

```csharp
    private readonly HrAutoDbContext _db;

    public async Task<ResultadoPaginado<Funcionario>> ListarAsync(FiltroFuncionarios filtro, CancellationToken cancelamento = default)
    {
        var consulta = _db.Funcionarios.AsNoTracking().AsQueryable();
        if (filtro.IdFuncionario is { } id) consulta = consulta.Where(f => f.Id == id);
        if (filtro.IdCargo is { } cargo) consulta = consulta.Where(f => f.IdCargo == cargo);
        if (filtro.IdDepartamento is { } departamento) consulta = consulta.Where(f => f.IdDepartamento == departamento);
        if (filtro.Ativo is { } ativo) consulta = consulta.Where(f => f.Ativo == ativo);
        if (filtro.IdGestor is { } gestor) consulta = consulta.Where(f => f.IdGestor == gestor);

        var total = await consulta.CountAsync(cancelamento);
        var itens = await consulta
            .Include(f => f.Cargo).Include(f => f.Departamento).Include(f => f.Gestor)
            .OrderBy(f => f.Id)
            .Skip(filtro.Paginacao.Pular)
            .Take(filtro.Paginacao.Tamanho)
            .ToListAsync(cancelamento);

        return new ResultadoPaginado<Funcionario>(itens, total);
```

```csharp
using HRAuto.Application.Abstractions;
using HRAuto.Application.Common;
using Microsoft.Data.Sqlite;
using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Storage;

namespace HRAuto.Infrastructure.Persistencia;

/// <summary>
/// Unidade de trabalho sobre o <see cref="HrAutoDbContext"/>. Os repositórios, a auditoria e esta classe
/// compartilham o mesmo contexto (escopo da requisição), então um SaveChanges grava tudo junto.
/// </summary>
public sealed class UnidadeDeTrabalho : IUnidadeDeTrabalho
{
    // Códigos estendidos do SQLite para violação de chave única e de chave estrangeira.
    private const int SqliteConstraintPrimaryKey = 1555;
    private const int SqliteConstraintUnique = 2067;
    private const int SqliteConstraintForeignKey = 787;

    private readonly HrAutoDbContext _db;

    public UnidadeDeTrabalho(HrAutoDbContext db) => _db = db;

    public async Task SalvarAsync(CancellationToken cancelamento = default)
    {
        try
        {
            await _db.SaveChangesAsync(cancelamento);
        }
        catch (DbUpdateException e) when (e.InnerException is SqliteException
        {
            SqliteErrorCode: 19,
            SqliteExtendedErrorCode: SqliteConstraintUnique or SqliteConstraintPrimaryKey or SqliteConstraintForeignKey
        })
        {
            // Ex.: dois toques simultâneos tentando criar o PONTO do mesmo dia (índice único ux_ponto_func_data, RNF09),
            // dois cadastros com o mesmo e-mail, ou a exclusão de um cargo que acabou de ser usado (chave estrangeira).
            throw new ConflitoDePersistenciaException("Violação de chave única ou estrangeira ao gravar.", e);
        }
    }

    public async Task<ITransacao> IniciarTransacaoAsync(CancellationToken cancelamento = default) =>
        new TransacaoEf(await _db.Database.BeginTransactionAsync(cancelamento));

    private sealed class TransacaoEf : ITransacao
    {
        private readonly IDbContextTransaction _transacao;

        public TransacaoEf(IDbContextTransaction transacao) => _transacao = transacao;

        public Task ConfirmarAsync(CancellationToken cancelamento = default) => _transacao.CommitAsync(cancelamento);

        // Descartar sem confirmar desfaz tudo (rollback).
        public ValueTask DisposeAsync() => _transacao.DisposeAsync();
    }
}
```

```csharp
    private static string CpfValido()
    {
        var d = new int[11];
        var n = Interlocked.Increment(ref _semente);
        for (var i = 0; i < 9; i++) d[i] = (n / (int)Math.Pow(10, 8 - i)) % 10;
        if (d.Take(9).Distinct().Count() == 1) d[8] = (d[8] + 1) % 10;

        int Digito(int quantos)
        {
            var soma = 0;
            for (var i = 0; i < quantos; i++) soma += d[i] * (quantos + 1 - i);
            var resto = soma * 10 % 11;
            return resto == 10 ? 0 : resto;
        }

        d[9] = Digito(9);
        d[10] = Digito(10);
        return string.Concat(d);
    }
```
