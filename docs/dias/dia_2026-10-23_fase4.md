## Suposições (confirmar com o grupo)
- Ninguém ajusta o próprio banco de horas nem decide as próprias férias; só Admin mexe em quem tem acesso RH/Admin (mesma hierarquia de `RegrasAcesso`). O gestor direto pode decidir férias mesmo sendo RH.
- Solicitação de férias avisa o gestor ativo; sem gestor (ou desligado), avisa os RH ativos. Cancelar = apagar a solicitação ainda Pendente (fica na auditoria), pois a tabela não tem status "Cancelada".
- Extrato do banco de horas: padrão 90 dias, máximo 366; o saldo é sempre o total.
- Nome do feriado 3 a 100 caracteres (limite assumido; o SQL não impõe). Feriado excluído é apagado de verdade (auditado).
- Notificação de outra pessoa responde 404 (igual a inexistente). Ordem 403 antes de 404 nas consultas por código.
- Fora de escopo e ainda PENDENTE: manutenção de ponto (endpoint de ajuste de batida) e holerites.

### backend/src/HRAuto.Application/Afastamentos/ConsultarFerias.cs

```csharp
using HRAuto.Application.Abstractions;
using HRAuto.Application.Common;
using HRAuto.Domain.Enums;
using HRAuto.Domain.ValueObjects;

namespace HRAuto.Application.Afastamentos;

/// <summary>
/// RF26, RF27: lista de solicitações de férias. "minhas" (padrão) é de qualquer pessoa; "equipe" mostra os subordinados
/// diretos e só vale para quem é gestor; "todas" é do RH/Admin. Da mais recente para a mais antiga.
/// </summary>
public sealed class ListarFerias
{
    private readonly IFuncionarioRepositorio _funcionarios;
    private readonly IFuncionarioCadastroRepositorio _cadastro;
    private readonly IFeriasRepositorio _ferias;

    public ListarFerias(IFuncionarioRepositorio funcionarios, IFuncionarioCadastroRepositorio cadastro, IFeriasRepositorio ferias)
    {
        _funcionarios = funcionarios;
        _cadastro = cadastro;
        _ferias = ferias;
    }

    public async Task<Resultado<PaginaDto<FeriasDto>>> ExecutarAsync(
        Solicitante solicitante, FiltroListaFeriasDto? filtro, CancellationToken cancelamento = default)
    {
        filtro ??= new FiltroListaFeriasDto(null, null, null, null, null);

        var escopo = (filtro.Escopo ?? "minhas").Trim().ToLowerInvariant();
        if (escopo is not ("minhas" or "equipe" or "todas"))
            return Resultado<PaginaDto<FeriasDto>>.Falha("Escopo inválido. Use minhas, equipe ou todas.");

        string? status = null;
        if (!string.IsNullOrWhiteSpace(filtro.Status))
        {
            status = filtro.Status.Trim().ToLowerInvariant() switch
            {
                "pendente" => StatusSolicitacao.Pendente,
                "aprovada" => StatusSolicitacao.Aprovada,
                "rejeitada" => StatusSolicitacao.Rejeitada,
                _ => null
            };
            if (status is null)
                return Resultado<PaginaDto<FeriasDto>>.Falha("Status inválido. Use Pendente, Aprovada ou Rejeitada.");
        }

        int? idCodigo = null;
        if (!string.IsNullOrWhiteSpace(filtro.Codigo))
        {
            if (!CodigoFuncionario.TentarConverter(filtro.Codigo, out var codigo))
                return Resultado<PaginaDto<FeriasDto>>.Falha("Código inválido. Use o formato FUNC-0001.");
            idCodigo = codigo.Id;
        }

        int? idFuncionario = idCodigo;
        int? idGestor = null;
        switch (escopo)
        {
            case "minhas":
                // O código só poderia mostrar outra pessoa; nas "minhas" ele nunca amplia o acesso.
                idFuncionario = solicitante.IdFuncionario;
                if (idCodigo is { } c && c != solicitante.IdFuncionario)
                    return Resultado<PaginaDto<FeriasDto>>.Falha(TipoErro.Proibido, "Use o escopo equipe ou todas para ver outro funcionário.");
                break;
            case "equipe":
                if (!await _funcionarios.EhGestorAsync(solicitante.IdFuncionario, cancelamento))
                    return Resultado<PaginaDto<FeriasDto>>.Falha(TipoErro.Proibido, "Você não é gestor de nenhuma equipe.");
                idGestor = solicitante.IdFuncionario;
                break;
            case "todas":
                if (!solicitante.Perfil.EhRhOuAdmin())
```

```csharp
/// domingo/feriado/2 dias antes). A solicitação nasce Pendente e o gestor é avisado; se o funcionário não tem gestor

public sealed class SolicitarFerias
{
    public const int TamanhoMaximoObservacao = 500;

    private readonly IFuncionarioRepositorio _funcionarios;
    private readonly IFeriadoRepositorio _feriados;
    private readonly IFeriasRepositorio _ferias;

        IFeriasRepositorio ferias,
        IUsuarioGestaoRepositorio usuarios,
        INotificacaoRepositorio notificacoes,
        IAuditoria auditoria,

        _relogio = relogio;
    }

    public async Task<Resultado<FeriasDto>> ExecutarAsync(
        Solicitante solicitante, SolicitarFeriasDto? dto, CancellationToken cancelamento = default)
    {
        if (dto is null || dto.Inicio is null || dto.Fim is null)
            return Resultado<FeriasDto>.Falha("Informe a data de início e a data final das férias.");
        if (dto.Observacao is { Length: > TamanhoMaximoObservacao })
            return Resultado<FeriasDto>.Falha($"A observação deve ter no máximo {TamanhoMaximoObservacao} caracteres.");

        var inicio = dto.Inicio.Value;
        var fim = dto.Fim.Value;

        var funcionario = await _funcionarios.ObterPorIdAsync(solicitante.IdFuncionario, cancelamento);
        if (funcionario is null)
            return Resultado<FeriasDto>.Falha(TipoErro.NaoEncontrado, "Funcionário não encontrado.");

        var agora = _relogio.Agora();
        var hoje = _relogio.Hoje();

        // Só o início e os 2 dias seguintes precisam ser conferidos contra os feriados (CLT, art. 134 §3º).
        var detalhes = await _funcionarios.ObterDetalhesAsync(funcionario.Id, cancelamento);
        var feriados = await _feriados.ListarPeriodoAsync(inicio, inicio.AddDays(2), cancelamento);
        var datasDeFeriado = ResolvedorContextoDia.DatasDeFeriadoAplicaveis(feriados, detalhes?.Cidade, detalhes?.Estado);
        var existentes = await _ferias.ListarAtivasDoFuncionarioAsync(funcionario.Id, cancelamento);

        var dias = dto.Dias ?? (fim.DayNumber - inicio.DayNumber + 1);

        Ferias solicitacao;
        try
        {
            solicitacao = RegrasFerias.Solicitar(
                new DadosFuncionario(funcionario.Id, funcionario.DataAdmissao, funcionario.Ativo),
                inicio, fim, dias, dto.Observacao, hoje, existentes, datasDeFeriado);
        }
        catch (DominioException e)
        {
            return Resultado<FeriasDto>.Falha(e.Message);
        }

        foreach (var idDestinatario in await DestinatariosDoAvisoAsync(funcionario, cancelamento))
            _notificacoes.Adicionar(NotificacoesJornada.NovaSolicitacaoDeFerias(idDestinatario, solicitacao, agora.DateTime));

        await using var transacao = await _unidade.IniciarTransacaoAsync(cancelamento);
        await _unidade.SalvarAsync(cancelamento); // gera o id da solicitação
        _auditoria.Registrar(solicitante.IdUsuario, "SolicitarFerias", "FERIAS", solicitacao.Id,
            $"{funcionario.Codigo}: {inicio:yyyy-MM-dd} a {fim:yyyy-MM-dd} ({dias} dias)");
        await _unidade.SalvarAsync(cancelamento);
        await transacao.ConfirmarAsync(cancelamento);

        var nomes = new Dictionary<int, string> { [funcionario.Id] = funcionario.Nome };
        return Resultado<FeriasDto>.Ok(MapeadorFerias.Ferias(solicitacao, nomes));
    }

    // Gestor ativo; sem gestor (ou gestor desligado), todos os RH ativos. Ninguém é avisado da própria solicitação.

            if (gestor is { Ativo: true }) return new[] { idGestor };
        }

        return (await _usuarios.ListarIdsFuncionariosRhAtivosAsync(cancelamento))
            .Where(id => id != funcionario.Id)
            .Distinct()
            .ToList();
    }
```

```csharp
        if (!solicitante.Perfil.EhRhOuAdmin())
            return Resultado<AjusteRegistradoDto>.Falha(TipoErro.Proibido, "Apenas o RH pode ajustar o banco de horas.");
        if (dto is null)
            return Resultado<AjusteRegistradoDto>.Falha("Envie os dados do ajuste.");
        if (!CodigoFuncionario.TentarConverter(dto.CodigoFuncionario, out var codigo))
```

```csharp
using HRAuto.Application.Abstractions;
using HRAuto.Application.Common;
using HRAuto.Domain.Common;
using HRAuto.Domain.Enums;
using HRAuto.Domain.Jornada;
using HRAuto.Domain.ValueObjects;

namespace HRAuto.Application.BancoDeHoras;

/// <summary>
/// RF24: saldo e extrato do banco de horas. O próprio funcionário vê o seu (sem código = o meu); o gestor vê o da equipe
/// (subordinados diretos); RH e Admin veem de todos. Sem período, mostra os últimos 90 dias; o saldo é sempre o total.
/// </summary>
public sealed class ConsultarBancoHoras
{
    public const int DiasPadrao = 90;
    public const int DiasMaximos = 366;

    private readonly IFuncionarioCadastroRepositorio _funcionarios;
    private readonly IBancoHorasRepositorio _bancoHoras;
    private readonly IRelogio _relogio;

    public ConsultarBancoHoras(IFuncionarioCadastroRepositorio funcionarios, IBancoHorasRepositorio bancoHoras, IRelogio relogio)
    {
        _funcionarios = funcionarios;
        _bancoHoras = bancoHoras;
        _relogio = relogio;
    }

    public async Task<Resultado<BancoHorasDto>> ExecutarAsync(
        Solicitante solicitante, string? codigoTexto, DateOnly? inicio, DateOnly? fim, CancellationToken cancelamento = default)
    {
        var idAlvo = solicitante.IdFuncionario;
        if (!string.IsNullOrWhiteSpace(codigoTexto))
        {
            if (!CodigoFuncionario.TentarConverter(codigoTexto, out var codigo))
                return Resultado<BancoHorasDto>.Falha("Código inválido. Use o formato FUNC-0001.");
            idAlvo = codigo.Id;
        }

        var dataFim = fim ?? _relogio.Hoje();
        var dataInicio = inicio ?? dataFim.AddDays(-(DiasPadrao - 1));
        if (dataInicio > dataFim)
            return Resultado<BancoHorasDto>.Falha("A data inicial não pode ser posterior à data final.");
        if (dataFim.DayNumber - dataInicio.DayNumber + 1 > DiasMaximos)
            return Resultado<BancoHorasDto>.Falha($"O período pode ter no máximo {DiasMaximos} dias.");

        // 403 antes de 404: quem não pode ver o funcionário também não descobre se o código existe.
        if (idAlvo != solicitante.IdFuncionario && !solicitante.Perfil.EhRhOuAdmin()
            && !await _funcionarios.EhSubordinadoDiretoAsync(solicitante.IdFuncionario, idAlvo, cancelamento))
            return Resultado<BancoHorasDto>.Falha(TipoErro.Proibido, "Você não tem acesso a este funcionário.");

        var funcionario = await _funcionarios.ObterAsync(idAlvo, cancelamento);
        if (funcionario is null)
            return Resultado<BancoHorasDto>.Falha(TipoErro.NaoEncontrado, "Funcionário não encontrado.");

        IReadOnlyList<Domain.Entities.BancoHoras> todos;
        decimal saldo;
        try
        {
            todos = await _bancoHoras.ListarTodosAsync(idAlvo, cancelamento);
            saldo = RegrasBancoHoras.Saldo(todos);
        }
        catch (DominioException)
        {
            // Lançamento com tipo desconhecido no banco: erro de dados, não do usuário.
            return Resultado<BancoHorasDto>.Falha(TipoErro.Conflito, "Há um lançamento com tipo inválido no banco de horas. Procure o administrador.");
        }

        var extrato = todos
            .Where(l => l.Data >= dataInicio && l.Data <= dataFim)
            .OrderByDescending(l => l.Data).ThenByDescending(l => l.Id)
            .Select(MapeadorBancoHoras.Lancamento)
            .ToList();

        return Resultado<BancoHorasDto>.Ok(new BancoHorasDto(
            funcionario.Codigo, funcionario.Nome, saldo, MapeadorBancoHoras.FormatarSaldo(saldo), dataInicio, dataFim, extrato));
```

```csharp
using HRAuto.Domain.Common;
using HRAuto.Domain.Entities;
using HRAuto.Domain.Enums;

namespace HRAuto.Domain.Calendario;

/// <summary>
/// Regras do cadastro de feriados (RF28):
/// - Nacional vale para todos e não tem cidade nem estado.
/// - Estadual exige o estado (sigla de 2 letras) e não tem cidade.
/// - Municipal exige cidade e estado.
/// Estado é gravado em maiúsculas e a cidade com os espaços das pontas removidos.
/// </summary>
public static class RegrasFeriado
{
    public const int TamanhoMinimoNome = 3;
    public const int TamanhoMaximoNome = 100;
    public const int TamanhoMaximoCidade = 100;

    /// <summary>Cria o feriado ou lança <see cref="DominioException"/> com o motivo.</summary>
    public static Feriado Criar(DateOnly data, string? nome, string? tipo, string? cidade, string? estado)
    {
        var nomeLimpo = nome?.Trim() ?? string.Empty;
        if (nomeLimpo.Length < TamanhoMinimoNome || nomeLimpo.Length > TamanhoMaximoNome)
            throw new DominioException($"O nome do feriado deve ter de {TamanhoMinimoNome} a {TamanhoMaximoNome} caracteres.");

        var tipoNormalizado = NormalizarTipo(tipo)
            ?? throw new DominioException("Tipo inválido. Use Nacional, Estadual ou Municipal.");

        var cidadeLimpa = string.IsNullOrWhiteSpace(cidade) ? null : cidade.Trim();
        var estadoLimpo = string.IsNullOrWhiteSpace(estado) ? null : estado.Trim().ToUpperInvariant();

        if (estadoLimpo is not null && !EstadoValido(estadoLimpo))
            throw new DominioException("Estado inválido. Use a sigla de 2 letras (ex.: SP).");
        if (cidadeLimpa is { Length: > TamanhoMaximoCidade })
            throw new DominioException($"A cidade deve ter no máximo {TamanhoMaximoCidade} caracteres.");

        switch (tipoNormalizado)
        {
            case TiposFeriado.Nacional:
                if (cidadeLimpa is not null || estadoLimpo is not null)
                    throw new DominioException("Feriado nacional não tem cidade nem estado.");
                break;
            case TiposFeriado.Estadual:
                if (estadoLimpo is null) throw new DominioException("Informe o estado do feriado estadual.");
                if (cidadeLimpa is not null) throw new DominioException("Feriado estadual não tem cidade.");
                break;
            case TiposFeriado.Municipal:
                if (estadoLimpo is null || cidadeLimpa is null)
                    throw new DominioException("Informe a cidade e o estado do feriado municipal.");
                break;
        }

        return new Feriado
        {
            Data = data,
            Nome = nomeLimpo,
            Tipo = tipoNormalizado,
            Cidade = cidadeLimpa,
            Estado = estadoLimpo
        };
    }

    /// <summary>"nacional", "ESTADUAL"... para o texto oficial gravado no banco; nulo se não for um tipo conhecido.</summary>
    public static string? NormalizarTipo(string? tipo) => tipo?.Trim().ToLowerInvariant() switch
    {
        "nacional" => TiposFeriado.Nacional,
        "estadual" => TiposFeriado.Estadual,
        "municipal" => TiposFeriado.Municipal,
        _ => null
    };

    private static bool EstadoValido(string estado) => estado.Length == 2 && estado.All(c => c is >= 'A' and <= 'Z');
}
```
