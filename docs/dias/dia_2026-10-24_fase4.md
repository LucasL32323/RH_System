Domain novo/estendido: `RegrasManutencaoPonto` (justificativa, prazo de 60 dias, `AplicarCorrecao` das batidas), `RegrasBancoHoras.LancamentoDeCorrecao` (reajuste do dia só pela diferença), `RegrasHolerite` (competência, PDF até 5 MB, caminho seguro), `NotificacoesJornada` (novo pedido, holerite publicado). `FechamentoDoDia` ganhou `ReajustarAsync`.

## Como a manutenção de ponto funciona
1. O colaborador pede a manutenção de um dia (até 60 dias atrás, nunca futuro nem antes da admissão) com justificativa de 10 a 500 caracteres. Se o dia não tem nenhuma batida, o ponto vazio é criado na mesma transação (a FK `id_ponto` exige). Só um pedido pendente por dia. O gestor ativo é avisado; sem gestor ativo, os RH ativos.
2. Quem decide: gestor direto, ou RH/Admin respeitando a hierarquia (RH não decide pedido de RH/Admin). Ninguém decide o próprio pedido. Rejeitar exige motivo e não aceita batidas.
3. Ao **aprovar**, o gestor pode enviar `batidas` (`entrada`, `saidaIntervalo`, `retornoIntervalo`, `saida`; campo vazio = não mexer). A correção inteira é validada antes de gravar (batidas contíguas, crescentes, 1 min de folga; nada é apagado, RNF07). Aprovar sem batidas = abono (só registra a decisão).
4. Se o dia ficou com as 4 batidas, o banco de horas é reajustado: não se altera lançamento antigo, entra um crédito/débito só com a diferença (origem "Ponto"). Auditoria: `DecidirManutencaoPonto` (MANUTENCAO_PONTO) e `AjustarBatida` (PONTO, com horário antigo e novo). O colaborador recebe a notificação da decisão.

## Suposições (confirmar com o grupo)
- Como `MANUTENCAO_PONTO` não tem colunas de horários propostos, o colaborador descreve os horários na justificativa e **quem aprova informa as batidas corrigidas**. Não existe ajuste direto de batida sem pedido aprovado.
- Prazo de 60 dias para pedir manutenção e justificativa/observação com no máximo 500 caracteres (limites assumidos; o SQL não impõe).
- Abono de dia sem batidas só registra a decisão; não altera o espelho nem o banco de horas.
- Reajuste do banco de horas não reenvia o alerta de hora extra ao gestor.
- Holerite: só RH/Admin publica; ninguém publica o próprio; competência repetida dá 409 a menos que peça `substituir` (o PDF antigo é mantido no disco por rastreabilidade); competência não pode ser futura nem anterior ao mês de admissão; funcionário desligado pode receber holerite.
- Holerite: o gestor NÃO enxerga holerites da equipe; só o dono e RH/Admin. Para os demais o holerite "não existe" (404). RH/Admin baixando o de outra pessoa gera auditoria `ConsultaHolerite`.
- PDFs ficam em `dados/holerites/FUNC-0001/AAAA/MM-<guid>.pdf` (pasta configurável em `Holerites:Pasta`); o banco guarda só o caminho relativo, nunca exposto na API. Nome do arquivo sorteado pelo sistema.
- O modelo físico não tem índice único de holerite por competência (`idx_holerite_func` não é único): duas publicações simultâneas para o mesmo mês poderiam duplicar; improvável com um RH só, e não foi alterado o esquema.
- Falha de disco ao gravar o PDF fica como erro 500 (tratador de exceções).

```csharp
using System.Text;
using System.Text.RegularExpressions;
using HRAuto.Domain.Common;
using HRAuto.Domain.Entities;
using HRAuto.Domain.ValueObjects;

namespace HRAuto.Domain.Holerites;

/// <summary>
/// Holerites (RF29, RF30, RNF17): o RH publica um PDF por funcionário/mês/ano; o funcionário consulta e baixa o próprio.
/// O arquivo fica em uma pasta do servidor e o banco guarda só o caminho RELATIVO. Aqui ficam as regras que não dependem
/// de disco nem de HTTP: competência válida, tamanho e tipo do arquivo e formato seguro do caminho.
/// </summary>
public static partial class RegrasHolerite
{
    /// <summary>RNF17: no máximo 5 MB por arquivo.</summary>
    public const int TamanhoMaximoBytes = 5 * 1024 * 1024;

    public const int AnoMinimo = 2000;

    private static readonly byte[] AssinaturaPdf = Encoding.ASCII.GetBytes("%PDF-");

    // FUNC-0001/2026/09-<32 hex>.pdf — nada de "..", barras invertidas, unidade de disco ou nome vindo do usuário.
    [GeneratedRegex(@"^FUNC-\d{4,}/\d{4}/\d{2}-[0-9a-f]{32}\.pdf$", RegexOptions.CultureInvariant)]
    private static partial Regex PadraoDeCaminho();

    /// <summary>Mês de 1 a 12, ano a partir de 2000, não pode ser futuro nem anterior ao mês de admissão.</summary>
    public static void ValidarCompetencia(int mes, int ano, DateOnly hoje, DateOnly admissao)
    {
        if (mes is < 1 or > 12) throw new DominioException("O mês do holerite deve estar entre 1 e 12.");
        if (ano < AnoMinimo) throw new DominioException($"O ano do holerite deve ser {AnoMinimo} ou posterior.");

        var competencia = ano * 12 + mes;
        if (competencia > hoje.Year * 12 + hoje.Month)
            throw new DominioException("Não é possível publicar o holerite de um mês que ainda não começou.");
        if (competencia < admissao.Year * 12 + admissao.Month)
            throw new DominioException("O funcionário ainda não havia sido admitido nessa competência.");
    }

    /// <summary>Só PDF (extensão e assinatura "%PDF-"), não vazio e com no máximo 5 MB (RNF17).</summary>
    public static void ValidarArquivo(string? nomeOriginal, ReadOnlySpan<byte> conteudo)
    {
        if (conteudo.IsEmpty) throw new DominioException("O arquivo do holerite está vazio.");
        if (conteudo.Length > TamanhoMaximoBytes)
            throw new DominioException($"O arquivo do holerite deve ter no máximo {TamanhoMaximoBytes / (1024 * 1024)} MB.");

        var extensao = Path.GetExtension(nomeOriginal ?? string.Empty);
        if (!string.Equals(extensao, ".pdf", StringComparison.OrdinalIgnoreCase))
            throw new DominioException("O holerite deve ser um arquivo PDF.");
        if (!conteudo.StartsWith(AssinaturaPdf))
            throw new DominioException("O arquivo enviado não é um PDF válido.");
    }

    /// <summary>Monta o caminho relativo no padrão seguro. O nome do arquivo é sorteado: nunca usa texto do usuário.</summary>
    public static string MontarCaminho(int idFuncionario, int ano, int mes, Guid unico) =>
        $"{CodigoFuncionario.Formatar(idFuncionario)}/{ano:D4}/{mes:D2}-{unico:N}.pdf";

    /// <summary>Confere o formato do caminho gravado no banco antes de tocar no disco (defesa contra path traversal).</summary>
    public static bool CaminhoEhSeguro(string? caminho) =>
        !string.IsNullOrEmpty(caminho) && PadraoDeCaminho().IsMatch(caminho);

    public static Holerite Criar(int idFuncionario, int mes, int ano, DateOnly hoje, string caminho)
    {
        if (idFuncionario <= 0) throw new DominioException("O funcionário do holerite é obrigatório.");
        ExigirCaminhoSeguro(caminho);
        return new Holerite { IdFuncionario = idFuncionario, Mes = mes, Ano = ano, DataPublicacao = hoje, Arquivo = caminho };
    }

    /// <summary>Troca o arquivo da competência por uma nova versão (o RH corrigiu o PDF).</summary>
    public static void Substituir(Holerite existente, DateOnly hoje, string novoCaminho)
    {
        ExigirCaminhoSeguro(novoCaminho);
        existente.Arquivo = novoCaminho;
        existente.DataPublicacao = hoje;
    }

    /// <summary>Nome sugerido ao baixar: sem dados pessoais, por exemplo "holerite-2026-09.pdf".</summary>
    public static string NomeDoArquivoParaDownload(Holerite holerite) => $"holerite-{holerite.Ano:D4}-{holerite.Mes:D2}.pdf";

    private static void ExigirCaminhoSeguro(string caminho)
    {
        if (!CaminhoEhSeguro(caminho)) throw new DominioException("Caminho de arquivo inválido para o holerite.");
    }
}
```

### backend/src/HRAuto.Domain/Jornada/RegrasManutencaoPonto.cs

```csharp
using HRAuto.Domain.Common;
using HRAuto.Domain.Entities;
using HRAuto.Domain.Enums;

namespace HRAuto.Domain.Jornada;

/// <summary>
/// Manutenção de ponto (RF22, RF23): o colaborador pede ajuste de batida ou abono do dia com justificativa;
/// o gestor aprova ou rejeita. Quem pode analisar (gestor do funcionário) é checado na API; aqui ficam as regras
/// que não dependem de perfil. Observação: MANUTENCAO_PONTO.id_ponto é FK de PONTO, então para abonar um dia sem
/// nenhuma batida a API precisa criar antes o registro de ponto vazio do dia (Ponto.Criar).
/// </summary>
public static class RegrasManutencaoPonto
{
    public const int TamanhoMinimoJustificativa = 10;
    public const int TamanhoMaximoJustificativa = 500;
    public const int TamanhoMinimoObservacaoRejeicao = 5;
    public const int TamanhoMaximoObservacao = 500;

    /// <summary>Quantos dias para trás o colaborador ainda pode pedir manutenção (suposição do TCC; ajustável).</summary>
    public const int PrazoMaximoDias = 60;

    private static readonly TipoBatida[] Sequencia =
    {
        TipoBatida.Entrada, TipoBatida.SaidaIntervalo, TipoBatida.RetornoIntervalo, TipoBatida.Saida
    };

    /// <summary>
    /// Confere e limpa a justificativa (obrigatória, RF22). Devolve o texto sem espaços nas pontas.
    /// Como MANUTENCAO_PONTO não tem colunas para horários propostos, o colaborador descreve o que aconteceu aqui.
    /// </summary>
    public static string NormalizarJustificativa(string? justificativa)
    {
        var texto = justificativa?.Trim() ?? string.Empty;
        if (texto.Length < TamanhoMinimoJustificativa)
            throw new DominioException($"A justificativa é obrigatória (mínimo de {TamanhoMinimoJustificativa} caracteres).");
        if (texto.Length > TamanhoMaximoJustificativa)
            throw new DominioException($"A justificativa deve ter no máximo {TamanhoMaximoJustificativa} caracteres.");
        return texto;
    }

    /// <summary>
    /// O dia do pedido precisa existir para o funcionário: não pode ser futuro, anterior à admissão
    /// nem mais antigo que <see cref="PrazoMaximoDias"/> dias.
    /// </summary>
    public static void ValidarDiaDoPedido(DateOnly data, DateOnly admissao, DateOnly hoje)
    {
        if (data > hoje)
            throw new DominioException("Não é possível pedir manutenção de um dia que ainda não aconteceu.");
        if (data < admissao)
            throw new DominioException("Não é possível pedir manutenção de um dia anterior à admissão.");
        if (hoje.DayNumber - data.DayNumber > PrazoMaximoDias)
            throw new DominioException($"O pedido de manutenção só vale para os últimos {PrazoMaximoDias} dias.");
    }

    /// <summary>
    /// Aplica as batidas corrigidas pelo gestor ao aprovar o pedido (RF22/RF23). Campos nulos ficam como estão; uma batida
    /// nunca é apagada (RNF07). A correção inteira é validada antes de qualquer alteração: as batidas finais precisam ser
    /// contíguas (sem "buraco") e crescentes, com pelo menos <see cref="Ponto.IntervaloMinimoEntreBatidas"/> entre elas.
    /// </summary>
    /// <returns>As batidas que de fato mudaram (vazio se a correção repete o que já estava gravado).</returns>
    public static IReadOnlyList<AlteracaoBatida> AplicarCorrecao(Ponto ponto, CorrecaoBatidas correcao)
    {
        if (correcao.Vazia) throw new DominioException("Informe pelo menos um horário para corrigir.");

        var finais = Sequencia.ToDictionary(t => t, t => Proposto(correcao, t) ?? ponto.Obter(t));

        // Contíguas: depois da primeira batida vazia não pode haver nenhuma preenchida.
        var vazioVisto = false;
        foreach (var tipo in Sequencia)
        {
            if (finais[tipo] is null) { vazioVisto = true; continue; }
            if (vazioVisto)
                throw new DominioException(
                    $"Informe também as batidas anteriores à {Ponto.Nome(tipo)}: o dia não pode ter batidas faltando no meio.");
        }

        // Crescentes, com folga mínima entre uma e outra.
        TimeOnly? anterior = null;
        TipoBatida? tipoAnterior = null;
        foreach (var tipo in Sequencia)
        {
            if (finais[tipo] is not { } atual) break;
            if (anterior is { } ant)
            {
                var diferenca = atual.ToTimeSpan() - ant.ToTimeSpan();
                if (diferenca < Ponto.IntervaloMinimoEntreBatidas)
                    throw new DominioException(
                        $"A {Ponto.Nome(tipo)} ({atual:HH\\:mm}) precisa ser depois da {Ponto.Nome(tipoAnterior!.Value)} ({ant:HH\\:mm}).");
            }
            anterior = atual;
            tipoAnterior = tipo;
        }

        var alteracoes = new List<AlteracaoBatida>();
        foreach (var tipo in Sequencia)
        {
            if (Proposto(correcao, tipo) is not { } depois) continue;
            var antes = ponto.Obter(tipo);
            if (antes == depois) continue;

            alteracoes.Add(new AlteracaoBatida(tipo, antes, depois));
            Definir(ponto, tipo, depois);
        }
        return alteracoes;
    }

    private static TimeOnly? Proposto(CorrecaoBatidas c, TipoBatida tipo) => tipo switch
    {
        TipoBatida.Entrada => c.Entrada,
        TipoBatida.SaidaIntervalo => c.SaidaIntervalo,
        TipoBatida.RetornoIntervalo => c.RetornoIntervalo,
        TipoBatida.Saida => c.Saida,
        _ => null
    };

    private static void Definir(Ponto ponto, TipoBatida tipo, TimeOnly hora)
    {
        switch (tipo)
        {
            case TipoBatida.Entrada: ponto.Entrada = hora; break;
            case TipoBatida.SaidaIntervalo: ponto.SaidaIntervalo = hora; break;
            case TipoBatida.RetornoIntervalo: ponto.RetornoIntervalo = hora; break;
            case TipoBatida.Saida: ponto.Saida = hora; break;
        }
    }

    /// <param name="ponto">Registro de ponto que será ajustado (já salvo, com Id).</param>
    /// <param name="idSolicitante">Funcionário que está pedindo; só pode pedir do próprio ponto.</param>
    /// <param name="justificativa">Obrigatória.</param>
    /// <param name="agora">Data e hora de Brasília (do relógio do servidor).</param>
    /// <param name="existentes">Pedidos já feitos para este ponto, para impedir dois pendentes ao mesmo tempo.</param>
    public static ManutencaoPonto Solicitar(
        Ponto ponto, int idSolicitante, string? justificativa, DateTime agora, IEnumerable<ManutencaoPonto> existentes)
    {
        if (ponto.Id <= 0) throw new DominioException("O ponto precisa estar salvo antes de pedir manutenção.");
        if (ponto.IdFuncionario != idSolicitante)
            throw new DominioException("Só é possível pedir manutenção do próprio ponto.");
        if (ponto.Data > DateOnly.FromDateTime(agora))
            throw new DominioException("Não é possível pedir manutenção de um dia que ainda não aconteceu.");

        var texto = NormalizarJustificativa(justificativa);

        if (existentes.Any(m => m.IdPonto == ponto.Id && m.Status == StatusSolicitacao.Pendente))
            throw new DominioException("Já existe um pedido de manutenção pendente para este dia.");

        return new ManutencaoPonto
        {
            IdPonto = ponto.Id,
            IdFuncionario = idSolicitante,
            Justificativa = texto,
            DataSolicitacao = agora,
            Status = StatusSolicitacao.Pendente
        };
    }

    /// <summary>Aprova ou rejeita o pedido, registrando responsável e data da análise (RF23).</summary>
    public static void Analisar(ManutencaoPonto pedido, bool aprovar, int idResponsavel, string? observacao, DateTime agora)
    {
        if (pedido.Status != StatusSolicitacao.Pendente)
            throw new DominioException("Este pedido já foi analisado.");
        if (idResponsavel <= 0) throw new DominioException("O responsável pela análise é obrigatório.");
        if (idResponsavel == pedido.IdFuncionario)
            throw new DominioException("Ninguém pode analisar o próprio pedido de manutenção.");

        var texto = observacao?.Trim();
        if (texto is { Length: > TamanhoMaximoObservacao })
            throw new DominioException($"A observação deve ter no máximo {TamanhoMaximoObservacao} caracteres.");
        if (!aprovar && (texto is null || texto.Length < TamanhoMinimoObservacaoRejeicao))
            throw new DominioException("Informe o motivo da rejeição.");

        pedido.Status = aprovar ? StatusSolicitacao.Aprovada : StatusSolicitacao.Rejeitada;
        pedido.IdResponsavel = idResponsavel;
        pedido.DataAnalise = agora;
        pedido.Observacao = string.IsNullOrEmpty(texto) ? null : texto;
    }
}

/// <summary>
/// Horários corrigidos pelo gestor ao aprovar. Campo nulo = "não mexer". Para completar um dia, informe também as batidas
/// que já existiam se quiser conferi-las; as que não vierem continuam como estão.
/// </summary>
public sealed record CorrecaoBatidas(TimeOnly? Entrada, TimeOnly? SaidaIntervalo, TimeOnly? RetornoIntervalo, TimeOnly? Saida)
{
    public bool Vazia => Entrada is null && SaidaIntervalo is null && RetornoIntervalo is null && Saida is null;
}

/// <summary>Uma batida que mudou: <c>Antes</c> é nulo quando a batida não existia (ex.: esquecimento de saída).</summary>
public sealed record AlteracaoBatida(TipoBatida Tipo, TimeOnly? Antes, TimeOnly Depois);
```

### backend/src/HRAuto.Domain/Jornada/RegrasBancoHoras.cs

```csharp
using HRAuto.Domain.Common;
using HRAuto.Domain.Entities;
using HRAuto.Domain.Enums;

namespace HRAuto.Domain.Jornada;

/// <summary>
/// Banco de horas (RF24): créditos e débitos diários gerados pelo ponto e ajustes manuais do RH.
/// Quantidade é sempre positiva e em horas decimais (4 casas); o sinal vem do tipo (Credito/Debito).
/// </summary>
public static class RegrasBancoHoras
{
    private const int CasasDecimais = 4;
    public const int TamanhoMinimoObservacao = 5;

    /// <summary>
    /// Gera o lançamento do fechamento do dia, ou nulo quando não há o que lançar:
    /// dia que não está completo (em andamento, inconsistente, falta, sem expediente), saldo zero,
    /// ou dia que já tem lançamento de origem "Ponto" (idempotente: rodar o fechamento duas vezes não duplica).
    /// </summary>
    public static BancoHoras? LancamentoDoDia(int idFuncionario, ResumoDia resumo, IEnumerable<BancoHoras> lancamentosExistentes)
    {
        if (idFuncionario <= 0) throw new DominioException("O funcionário do lançamento é obrigatório.");
        if (resumo.Situacao != SituacaoDia.Completo) return null;

        var jaLancado = lancamentosExistentes.Any(l =>
            l.IdFuncionario == idFuncionario && l.Data == resumo.Data && l.Origem == OrigensBancoHoras.Ponto);
        if (jaLancado) return null;

        var horas = Math.Round(resumo.Saldo.TotalHours, CasasDecimais, MidpointRounding.AwayFromZero);
        if (horas == 0) return null;

        return new BancoHoras
        {
            IdFuncionario = idFuncionario,
            Data = resumo.Data,
            Tipo = horas > 0 ? TiposBancoHoras.Credito : TiposBancoHoras.Debito,
            Quantidade = Math.Abs(horas),
            Origem = OrigensBancoHoras.Ponto,
            Observacao = $"Fechamento do ponto de {resumo.Data:dd/MM/yyyy}"
        };
    }

    /// <summary>
    /// Reajusta o banco de horas de um dia que já foi fechado quando uma manutenção de ponto aprovada muda as batidas (RF22, RF24).
    /// Não apaga nem altera lançamentos antigos (RNF07): lança só a DIFERENÇA entre o saldo novo do dia e o que os
    /// lançamentos de origem "Ponto" desse dia já somam. Se o dia não estava fechado, a diferença é o saldo inteiro.
    /// Devolve nulo quando o dia não está completo ou quando não há diferença (idempotente).
    /// </summary>
    public static BancoHoras? LancamentoDeCorrecao(int idFuncionario, ResumoDia resumo, IEnumerable<BancoHoras> lancamentosExistentes)
    {
        if (idFuncionario <= 0) throw new DominioException("O funcionário do lançamento é obrigatório.");
        if (resumo.Situacao != SituacaoDia.Completo) return null;

        var jaLancado = Saldo(lancamentosExistentes.Where(l =>
            l.IdFuncionario == idFuncionario && l.Data == resumo.Data && l.Origem == OrigensBancoHoras.Ponto));
        var alvo = Math.Round((decimal)resumo.Saldo.TotalHours, CasasDecimais, MidpointRounding.AwayFromZero);
        var diferenca = Math.Round(alvo - jaLancado, CasasDecimais, MidpointRounding.AwayFromZero);
        if (diferenca == 0) return null;

        return new BancoHoras
        {
            IdFuncionario = idFuncionario,
            Data = resumo.Data,
            Tipo = diferenca > 0 ? TiposBancoHoras.Credito : TiposBancoHoras.Debito,
            Quantidade = (double)Math.Abs(diferenca),
            Origem = OrigensBancoHoras.Ponto,
            Observacao = $"Correção do ponto de {resumo.Data:dd/MM/yyyy} (manutenção aprovada)"
        };
    }

    /// <summary>
    /// Ajuste manual do RH. Exige observação (rastreabilidade, RNF07) e não aceita data futura.
    /// </summary>
    public static BancoHoras AjusteManual(
        int idFuncionario, DateOnly data, string tipo, double horas, string? observacao, DateOnly hoje)
    {
        if (idFuncionario <= 0) throw new DominioException("O funcionário do ajuste é obrigatório.");
        if (tipo != TiposBancoHoras.Credito && tipo != TiposBancoHoras.Debito)
            throw new DominioException("O tipo do ajuste deve ser Credito ou Debito.");
        if (double.IsNaN(horas) || double.IsInfinity(horas) || horas <= 0)
            throw new DominioException("A quantidade de horas do ajuste deve ser maior que zero.");
        if (data > hoje) throw new DominioException("O ajuste não pode ter data futura.");

        var texto = observacao?.Trim() ?? string.Empty;
        if (texto.Length < TamanhoMinimoObservacao)
            throw new DominioException($"Informe a observação do ajuste (mínimo de {TamanhoMinimoObservacao} caracteres).");

        return new BancoHoras
        {
            IdFuncionario = idFuncionario,
            Data = data,
            Tipo = tipo,
            Quantidade = Math.Round(horas, CasasDecimais, MidpointRounding.AwayFromZero),
            Origem = OrigensBancoHoras.AjusteRH,
            Observacao = texto
        };
    }

    /// <summary>Saldo em horas: soma dos créditos menos a soma dos débitos (pode ser negativo).</summary>
    public static decimal Saldo(IEnumerable<BancoHoras> lancamentos)
    {
        decimal saldo = 0;
        foreach (var l in lancamentos)
        {
            var quantidade = (decimal)l.Quantidade;
            saldo += l.Tipo switch
            {
                TiposBancoHoras.Credito => quantidade,
                TiposBancoHoras.Debito => -quantidade,
                _ => throw new DominioException($"Tipo de lançamento desconhecido no banco de horas: '{l.Tipo}'.")
            };
        }
        return Math.Round(saldo, CasasDecimais, MidpointRounding.AwayFromZero);
    }
}
```

```csharp
using HRAuto.Application.Abstractions;
using HRAuto.Domain.Comunicacao;
using HRAuto.Domain.Entities;
using HRAuto.Domain.Jornada;

namespace HRAuto.Application.RegistroPonto;

/// <summary>
/// Fecha o dia quando a 4ª batida é registrada: calcula horas (RF20), lança o crédito/débito no banco de horas (RF24)
/// e avisa o gestor se as horas extras passaram do limite (RF25). Só marca as mudanças nos repositórios;
/// quem grava é o caso de uso que chamou, dentro da sua transação.
/// </summary>
public sealed class FechamentoDoDia
{
    private readonly IFuncionarioRepositorio _funcionarios;
    private readonly IFeriadoRepositorio _feriados;
    private readonly IFeriasRepositorio _ferias;
    private readonly IBancoHorasRepositorio _bancoHoras;
    private readonly INotificacaoRepositorio _notificacoes;
    private readonly JornadaPadrao _jornada;
    private readonly CalculadoraJornada _calculadora;
    private readonly IRelogio _relogio;

    public FechamentoDoDia(
        IFuncionarioRepositorio funcionarios,
        IFeriadoRepositorio feriados,
        IFeriasRepositorio ferias,
        IBancoHorasRepositorio bancoHoras,
        INotificacaoRepositorio notificacoes,
        JornadaPadrao jornada,
        CalculadoraJornada calculadora,
        IRelogio relogio)
    {
        _funcionarios = funcionarios;
        _feriados = feriados;
        _ferias = ferias;
        _bancoHoras = bancoHoras;
        _notificacoes = notificacoes;
        _jornada = jornada;
        _calculadora = calculadora;
        _relogio = relogio;
    }

    /// <param name="funcionario">Dono do ponto.</param>
    /// <param name="ponto">Ponto do dia, com as quatro batidas.</param>
    public async Task<ResumoDia> ExecutarAsync(Funcionario funcionario, Ponto ponto, CancellationToken cancelamento = default)
    {
        var dia = ponto.Data;
        var resumo = await CalcularAsync(funcionario, ponto, cancelamento);

        // Idempotente: se o dia já tem lançamento de origem "Ponto", não cria outro.
        var existentes = await _bancoHoras.ListarDoDiaAsync(funcionario.Id, dia, cancelamento);
        var lancamento = RegrasBancoHoras.LancamentoDoDia(funcionario.Id, resumo, existentes);
        if (lancamento is not null) _bancoHoras.Adicionar(lancamento);

        if (funcionario.IdGestor is int idGestor && _calculadora.ExcedeLimiteHoraExtra(resumo))
        {
            _notificacoes.Adicionar(FabricaNotificacoes.HoraExtra(
                idGestor, funcionario.Id, resumo.HorasExtras, dia, _relogio.Agora().DateTime));
        }

        return resumo;
    }

    /// <summary>
    /// Refaz o fechamento de um dia cujas batidas foram corrigidas por uma manutenção de ponto aprovada (RF22, RF24).
    /// Os lançamentos antigos ficam como estão (RNF07); entra só a diferença (<see cref="RegrasBancoHoras.LancamentoDeCorrecao"/>).
    /// Não reenvia o alerta de hora extra: o gestor que aprovou a correção já está ciente.
    /// </summary>
    public async Task<ResumoDia> ReajustarAsync(Funcionario funcionario, Ponto ponto, CancellationToken cancelamento = default)
    {
        var resumo = await CalcularAsync(funcionario, ponto, cancelamento);

        var existentes = await _bancoHoras.ListarDoDiaAsync(funcionario.Id, ponto.Data, cancelamento);
        var correcao = RegrasBancoHoras.LancamentoDeCorrecao(funcionario.Id, resumo, existentes);
        if (correcao is not null) _bancoHoras.Adicionar(correcao);

        return resumo;
    }

    // Contexto do dia (feriado conforme a cidade/estado do funcionário, férias aprovadas) e cálculo das horas (RF20).
    private async Task<ResumoDia> CalcularAsync(Funcionario funcionario, Ponto ponto, CancellationToken cancelamento)
    {
        var dia = ponto.Data;

        var detalhes = await _funcionarios.ObterDetalhesAsync(funcionario.Id, cancelamento);
        var feriados = await _feriados.ListarPeriodoAsync(dia, dia, cancelamento);
        var datasDeFeriado = ResolvedorContextoDia.DatasDeFeriadoAplicaveis(feriados, detalhes?.Cidade, detalhes?.Estado);
        var ferias = await _ferias.ListarAprovadasNoPeriodoAsync(funcionario.Id, dia, dia, cancelamento);

        var contexto = ResolvedorContextoDia.Resolver(dia, _jornada, ferias, datasDeFeriado);
        return _calculadora.Calcular(ponto, dia, contexto, diaEncerrado: true);
    }
}
```

### backend/src/HRAuto.Application/ManutencaoDePonto/SolicitarManutencaoPonto.cs

```csharp
using HRAuto.Application.Abstractions;
using HRAuto.Application.Common;
using HRAuto.Domain.Comunicacao;
using HRAuto.Domain.Common;
using HRAuto.Domain.Entities;
using HRAuto.Domain.Jornada;

namespace HRAuto.Application.ManutencaoDePonto;

/// <summary>
/// RF22: o colaborador pede a manutenção do ponto de um dia (ajuste de batida ou abono) com justificativa obrigatória.
/// O pedido nasce Pendente e o gestor é avisado; sem gestor ativo, o aviso vai para o RH. Se o dia não tem nenhuma batida,
/// o registro de ponto vazio é criado antes (MANUTENCAO_PONTO.id_ponto é FK de PONTO), na mesma transação do pedido.
/// </summary>
public sealed class SolicitarManutencaoPonto
{
    private readonly IFuncionarioRepositorio _funcionarios;
    private readonly IPontoRepositorio _pontos;
    private readonly IManutencaoPontoRepositorio _manutencoes;
    private readonly IUsuarioGestaoRepositorio _usuarios;
    private readonly INotificacaoRepositorio _notificacoes;
    private readonly IAuditoria _auditoria;
    private readonly IUnidadeDeTrabalho _unidade;
    private readonly IRelogio _relogio;

    public SolicitarManutencaoPonto(
        IFuncionarioRepositorio funcionarios,
        IPontoRepositorio pontos,
        IManutencaoPontoRepositorio manutencoes,
        IUsuarioGestaoRepositorio usuarios,
        INotificacaoRepositorio notificacoes,
        IAuditoria auditoria,
        IUnidadeDeTrabalho unidade,
        IRelogio relogio)
    {
        _funcionarios = funcionarios;
        _pontos = pontos;
        _manutencoes = manutencoes;
        _usuarios = usuarios;
        _notificacoes = notificacoes;
        _auditoria = auditoria;
        _unidade = unidade;
        _relogio = relogio;
    }

    public async Task<Resultado<ManutencaoPontoDto>> ExecutarAsync(
        Solicitante solicitante, SolicitarManutencaoPontoDto? dto, CancellationToken cancelamento = default)
    {
        if (dto is null || dto.Data is null)
            return Resultado<ManutencaoPontoDto>.Falha("Informe o dia do ponto que precisa de manutenção.");

        var funcionario = await _funcionarios.ObterPorIdAsync(solicitante.IdFuncionario, cancelamento);
        if (funcionario is null || !funcionario.Ativo)
            return Resultado<ManutencaoPontoDto>.Falha(TipoErro.Proibido, "Funcionário inativo ou não encontrado.");

        var data = dto.Data.Value;
        var agora = _relogio.Agora();
        var hoje = _relogio.Hoje();

        // Tudo que dá para conferir sem gravar nada é conferido antes de abrir a transação.
        string justificativa;
        try
        {
            justificativa = RegrasManutencaoPonto.NormalizarJustificativa(dto.Justificativa);
            RegrasManutencaoPonto.ValidarDiaDoPedido(data, funcionario.DataAdmissao, hoje);
        }
        catch (DominioException e)
        {
            return Resultado<ManutencaoPontoDto>.Falha(e.Message);
        }

        var ponto = await _pontos.ObterDoDiaAsync(funcionario.Id, data, cancelamento);
        var existentes = ponto is null
            ? Array.Empty<ManutencaoPonto>()
            : await _manutencoes.ListarDoPontoAsync(ponto.Id, cancelamento);
        var destinatarios = await DestinatariosDeAviso.ObterAsync(funcionario, _funcionarios, _usuarios, cancelamento);

        ManutencaoPonto pedido;
        try
        {
            await using var transacao = await _unidade.IniciarTransacaoAsync(cancelamento);

            if (ponto is null)
            {
                // Dia sem nenhuma batida: cria o ponto vazio e grava para obter o id (a FK exige).
                ponto = Ponto.Criar(funcionario.Id, data);
                _pontos.Adicionar(ponto);
                await _unidade.SalvarAsync(cancelamento);
            }

            try
            {
                pedido = RegrasManutencaoPonto.Solicitar(ponto, funcionario.Id, justificativa, agora.DateTime, existentes);
            }
            catch (DominioException e)
            {
                return Resultado<ManutencaoPontoDto>.Falha(e.Message); // a transação é descartada (rollback)
            }

            _manutencoes.Adicionar(pedido);
            foreach (var idDestinatario in destinatarios)
                _notificacoes.Adicionar(NotificacoesJornada.NovaManutencaoDePonto(idDestinatario, pedido, data, agora.DateTime));
            await _unidade.SalvarAsync(cancelamento); // gera o id do pedido

            _auditoria.Registrar(solicitante.IdUsuario, "SolicitarManutencaoPonto", "MANUTENCAO_PONTO", pedido.Id,
                $"{funcionario.Codigo}: ponto de {data:yyyy-MM-dd}");
            await _unidade.SalvarAsync(cancelamento);

            await transacao.ConfirmarAsync(cancelamento);
        }
        catch (ConflitoDePersistenciaException)
        {
            return Resultado<ManutencaoPontoDto>.Falha(
                TipoErro.Conflito, "Outra operação alterou o ponto deste dia ao mesmo tempo. Tente novamente.");
        }

        var nomes = new Dictionary<int, string> { [funcionario.Id] = funcionario.Nome };
        return Resultado<ManutencaoPontoDto>.Ok(MapeadorManutencao.Pedido(pedido, ponto, nomes));
    }
```

```csharp
    public async Task<Resultado<PaginaDto<ManutencaoPontoDto>>> ExecutarAsync(

    {

        if (escopo is not ("minhas" or "equipe" or "todas"))

                "pendente" => StatusSolicitacao.Pendente,
                "aprovada" => StatusSolicitacao.Aprovada,
                "rejeitada" => StatusSolicitacao.Rejeitada,
                _ => null
            };
            if (status is null)
                return Resultado<PaginaDto<ManutencaoPontoDto>>.Falha("Status inválido. Use Pendente, Aprovada ou Rejeitada.");
        }

        int? idCodigo = null;
        if (!string.IsNullOrWhiteSpace(filtro.Codigo))

            if (!CodigoFuncionario.TentarConverter(filtro.Codigo, out var codigo))
                return Resultado<PaginaDto<ManutencaoPontoDto>>.Falha("Código inválido. Use o formato FUNC-0001.");
            idCodigo = codigo.Id;
        }

        int? idFuncionario = idCodigo;
        int? idGestor = null;
        switch (escopo)

            case "minhas":
                // O código só poderia mostrar outra pessoa; nas "minhas" ele nunca amplia o acesso.

    }
```

```csharp
using HRAuto.Application.Abstractions;
using HRAuto.Application.Common;
using HRAuto.Application.RegistroPonto;
using HRAuto.Domain.Comunicacao;

/// <summary>
/// RF23: aprova ou rejeita um pedido de manutenção de ponto pendente, registrando responsável e data da análise.
/// Quem decide: o gestor direto do funcionário ou o RH/Admin (o RH não decide o pedido de quem tem acesso RH ou Admin,
/// salvo se for o gestor direto). Ninguém decide o próprio pedido.
/// Ao aprovar, o gestor pode informar as batidas corrigidas: elas são validadas (ordem, sem buraco, nada é apagado),
/// gravadas e auditadas com o horário antigo e o novo (RNF07); se o dia ficou com as quatro batidas, o banco de horas é
/// reajustado pela diferença. Aprovar sem batidas é um abono: só registra a decisão. O funcionário é avisado.
/// </summary>
public sealed class DecidirManutencaoPonto
{
    private readonly IManutencaoPontoRepositorio _manutencoes;
    private readonly IFuncionarioRepositorio _funcionarios;
    private readonly IFuncionarioCadastroRepositorio _cadastro;
    private readonly IUsuarioGestaoRepositorio _usuarios;
    private readonly FechamentoDoDia _fechamento;
    private readonly INotificacaoRepositorio _notificacoes;
    private readonly IAuditoria _auditoria;
    private readonly IUnidadeDeTrabalho _unidade;
    private readonly IRelogio _relogio;

    public DecidirManutencaoPonto(
        IManutencaoPontoRepositorio manutencoes,
        IFuncionarioRepositorio funcionarios,
        IFuncionarioCadastroRepositorio cadastro,
        IUsuarioGestaoRepositorio usuarios,
        FechamentoDoDia fechamento,
        INotificacaoRepositorio notificacoes,
        IAuditoria auditoria,
        IUnidadeDeTrabalho unidade,
        IRelogio relogio)
    {
        _manutencoes = manutencoes;
        _funcionarios = funcionarios;
        _cadastro = cadastro;
        _usuarios = usuarios;
        _fechamento = fechamento;
        _notificacoes = notificacoes;
        _auditoria = auditoria;
        _unidade = unidade;
        _relogio = relogio;
    }

    public async Task<Resultado<ManutencaoDecididaDto>> ExecutarAsync(
        Solicitante solicitante, int idManutencao, DecidirManutencaoPontoDto? dto, CancellationToken cancelamento = default)
    {
        if (dto is null || dto.Aprovar is null)
            return Resultado<ManutencaoDecididaDto>.Falha("Informe se o pedido foi aprovado ou rejeitado.");
        var aprovar = dto.Aprovar.Value;
        if (!aprovar && dto.Batidas is not null)
            return Resultado<ManutencaoDecididaDto>.Falha("Não é possível corrigir batidas ao rejeitar o pedido.");

        var pedido = await _manutencoes.ObterParaEdicaoAsync(idManutencao, cancelamento);
        if (pedido is null)
            return Resultado<ManutencaoDecididaDto>.Falha(TipoErro.NaoEncontrado, "Pedido de manutenção de ponto não encontrado.");

        if (pedido.IdFuncionario == solicitante.IdFuncionario)
            return Resultado<ManutencaoDecididaDto>.Falha(TipoErro.Proibido, "Você não pode decidir o seu próprio pedido de manutenção.");
        if (!await PodeDecidirAsync(solicitante, pedido.IdFuncionario, cancelamento))
            return Resultado<ManutencaoDecididaDto>.Falha(TipoErro.Proibido, "Você não pode decidir os pedidos deste funcionário.");
        if (pedido.Status != StatusSolicitacao.Pendente)
            return Resultado<ManutencaoDecididaDto>.Falha(TipoErro.Conflito, "Este pedido de manutenção já foi decidido.");

        var ponto = await _manutencoes.ObterPontoParaEdicaoAsync(pedido.IdPonto, cancelamento);
        var dono = await _funcionarios.ObterPorIdAsync(pedido.IdFuncionario, cancelamento);
        if (ponto is null || dono is null)
            return Resultado<ManutencaoDecididaDto>.Falha(TipoErro.NaoEncontrado, "O ponto deste pedido não foi encontrado.");

        var agora = _relogio.Agora();
        IReadOnlyList<AlteracaoBatida> alteracoes = Array.Empty<AlteracaoBatida>();

        try
        {
            // Primeiro as batidas (a correção é validada inteira antes de mexer em qualquer horário); a análise vem depois
            // e não falha mais neste ponto, então nunca sobra batida alterada com pedido ainda pendente.
            if (aprovar && dto.Batidas is { } b)
            {
                alteracoes = RegrasManutencaoPonto.AplicarCorrecao(
                    ponto, new CorrecaoBatidas(b.Entrada, b.SaidaIntervalo, b.RetornoIntervalo, b.Saida));
            }
            RegrasManutencaoPonto.Analisar(pedido, aprovar, solicitante.IdFuncionario, dto.Observacao, agora.DateTime);
        }
        catch (DominioException e)
        {
            return Resultado<ManutencaoDecididaDto>.Falha(e.Message);
        }

        // Dia com as quatro batidas depois da correção: o banco de horas acompanha (só a diferença é lançada).
        ResumoDia? fechamento = null;
        if (alteracoes.Count > 0 && ponto.TotalBatidas() == 4)
            fechamento = await _fechamento.ReajustarAsync(dono, ponto, cancelamento);

        _notificacoes.Adicionar(FabricaNotificacoes.DecisaoDeManutencao(pedido, agora.DateTime));

        var codigo = CodigoFuncionario.Formatar(pedido.IdFuncionario);
        _auditoria.Registrar(solicitante.IdUsuario, "DecidirManutencaoPonto", "MANUTENCAO_PONTO", pedido.Id,
            $"{codigo}: {pedido.Status} (ponto de {ponto.Data:yyyy-MM-dd})");
        if (alteracoes.Count > 0)
        {
            var detalhes = string.Join("; ", alteracoes.Select(a =>
                $"{Ponto.Nome(a.Tipo)} {(a.Antes is { } antes ? antes.ToString("HH\\:mm\\:ss") : "(vazia)")} -> {a.Depois:HH\\:mm\\:ss}"));
            _auditoria.Registrar(solicitante.IdUsuario, "AjustarBatida", "PONTO", ponto.Id,
                $"{codigo} em {ponto.Data:yyyy-MM-dd}: {detalhes}");
        }

        try
        {
            await _unidade.SalvarAsync(cancelamento);
        }
        catch (ConflitoDePersistenciaException)
        {
            return Resultado<ManutencaoDecididaDto>.Falha(
                TipoErro.Conflito, "O ponto foi alterado ao mesmo tempo. Atualize a tela e tente novamente.");
        }

        var nomes = await _cadastro.ObterNomesAsync(new[] { pedido.IdFuncionario }, cancelamento);
        return Resultado<ManutencaoDecididaDto>.Ok(new ManutencaoDecididaDto(
            MapeadorManutencao.Pedido(pedido, ponto, nomes),
            alteracoes.Count > 0,
            alteracoes.Select(MapeadorManutencao.Alteracao).ToList(),
            fechamento is null ? null : ResumoDiaDto.De(fechamento)));
    }

    // Mesma hierarquia das férias: gestor direto sempre pode; RH/Admin só se puder gerenciar o perfil do dono.
    private async Task<bool> PodeDecidirAsync(Solicitante solicitante, int idDono, CancellationToken cancelamento)
    {
        if (await _cadastro.EhSubordinadoDiretoAsync(solicitante.IdFuncionario, idDono, cancelamento)) return true;
        if (!solicitante.Perfil.EhRhOuAdmin()) return false;

        var usuarioDono = await _usuarios.ObterPorFuncionarioAsync(idDono, cancelamento);
        return usuarioDono is null || RegrasAcesso.PodeGerenciar(solicitante.Perfil, (PerfilAcesso)usuarioDono.IdPerfil);
    }
}
```

```csharp
using HRAuto.Application.Abstractions;
using HRAuto.Application.Common;
using HRAuto.Domain.Comunicacao;
using HRAuto.Domain.Common;
using HRAuto.Domain.Entities;
using HRAuto.Domain.Enums;

using HRAuto.Domain.ValueObjects;

namespace HRAuto.Application.Holerites;

/// <summary>
/// RF29: o RH/Admin publica o holerite (PDF, até 5 MB, RNF17) de um funcionário para um mês/ano. O arquivo vai para a pasta
/// do servidor com nome sorteado e o banco guarda só o caminho relativo. Competência repetida dá conflito, a menos que
/// o RH peça para substituir (o arquivo antigo é mantido no disco por rastreabilidade, RNF07). Ninguém publica o próprio
/// holerite. O funcionário é avisado e a publicação é auditada. Se a gravação no banco falhar, o arquivo novo é apagado.
/// </summary>
public sealed class PublicarHolerite
{
    private readonly IFuncionarioRepositorio _funcionarios;
    private readonly IFuncionarioCadastroRepositorio _cadastro;
    private readonly IHoleriteRepositorio _holerites;
    private readonly IArmazenamentoHolerites _armazenamento;
    private readonly INotificacaoRepositorio _notificacoes;
    private readonly IAuditoria _auditoria;
    private readonly IUnidadeDeTrabalho _unidade;
    private readonly IRelogio _relogio;

    public PublicarHolerite(
        IFuncionarioRepositorio funcionarios,
        IFuncionarioCadastroRepositorio cadastro,
        IHoleriteRepositorio holerites,
        IArmazenamentoHolerites armazenamento,
        INotificacaoRepositorio notificacoes,
        IAuditoria auditoria,
        IUnidadeDeTrabalho unidade,
        IRelogio relogio)
    {
        _funcionarios = funcionarios;
        _cadastro = cadastro;
        _holerites = holerites;
        _armazenamento = armazenamento;
        _notificacoes = notificacoes;
        _auditoria = auditoria;
        _unidade = unidade;
        _relogio = relogio;
    }

    public async Task<Resultado<HoleritePublicadoDto>> ExecutarAsync(
        Solicitante solicitante, PublicarHoleriteDto? dto, CancellationToken cancelamento = default)
    {
        if (!solicitante.Perfil.EhRhOuAdmin())
            return Resultado<HoleritePublicadoDto>.Falha(TipoErro.Proibido, "Apenas o RH pode publicar holerites.");

        if (dto is null || string.IsNullOrWhiteSpace(dto.Codigo))
            return Resultado<HoleritePublicadoDto>.Falha("Informe o código do funcionário (FUNC-0001).");
        if (!CodigoFuncionario.TentarConverter(dto.Codigo, out var codigo))
            return Resultado<HoleritePublicadoDto>.Falha("Código inválido. Use o formato FUNC-0001.");
        if (dto.Mes is null || dto.Ano is null)
            return Resultado<HoleritePublicadoDto>.Falha("Informe o mês e o ano do holerite.");

        if (codigo.Id == solicitante.IdFuncionario)
            return Resultado<HoleritePublicadoDto>.Falha(TipoErro.Proibido, "Você não pode publicar o seu próprio holerite.");

        var funcionario = await _funcionarios.ObterPorIdAsync(codigo.Id, cancelamento);
        if (funcionario is null)
            return Resultado<HoleritePublicadoDto>.Falha(TipoErro.NaoEncontrado, "Funcionário não encontrado.");

        var mes = dto.Mes.Value;
        var ano = dto.Ano.Value;
        var agora = _relogio.Agora();
        var hoje = _relogio.Hoje();

        try
        {
            RegrasHolerite.ValidarCompetencia(mes, ano, hoje, funcionario.DataAdmissao);
            RegrasHolerite.ValidarArquivo(dto.NomeArquivo, dto.Conteudo.Span);
        }
        catch (DominioException e)
        {
            return Resultado<HoleritePublicadoDto>.Falha(e.Message);
        }

        var existente = await _holerites.ObterDaCompetenciaParaEdicaoAsync(funcionario.Id, ano, mes, cancelamento);
        if (existente is not null && !dto.Substituir)
            return Resultado<HoleritePublicadoDto>.Falha(
                TipoErro.Conflito, $"Já existe holerite de {mes:D2}/{ano} para {funcionario.Codigo}. Peça para substituir.");

        // Arquivo primeiro: se o banco falhar, o arquivo novo é apagado e nada fica pendurado.
        var caminho = await _armazenamento.SalvarAsync(funcionario.Id, ano, mes, dto.Conteudo, cancelamento);
        Holerite holerite;
        try
        {
            await using var transacao = await _unidade.IniciarTransacaoAsync(cancelamento);

            if (existente is null)
            {
                holerite = RegrasHolerite.Criar(funcionario.Id, mes, ano, hoje, caminho);
                _holerites.Adicionar(holerite);
            }
            else
            {
                holerite = existente;
                RegrasHolerite.Substituir(holerite, hoje, caminho);
            }

            _notificacoes.Adicionar(NotificacoesJornada.HoleritePublicado(holerite, existente is not null, agora.DateTime));
            await _unidade.SalvarAsync(cancelamento); // gera o id do holerite

            _auditoria.Registrar(solicitante.IdUsuario, existente is null ? "PublicarHolerite" : "SubstituirHolerite", "HOLERITE", holerite.Id,
                $"{funcionario.Codigo}: {mes:D2}/{ano}");
            await _unidade.SalvarAsync(cancelamento);

            await transacao.ConfirmarAsync(cancelamento);
        }
        catch (ConflitoDePersistenciaException)
        {
            await TentarRemoverAsync(caminho);
            return Resultado<HoleritePublicadoDto>.Falha(
                TipoErro.Conflito, "O holerite foi alterado ao mesmo tempo por outra operação. Tente novamente.");
        }
        catch
        {
            await TentarRemoverAsync(caminho);
            throw;
        }

        var nomes = new Dictionary<int, string> { [funcionario.Id] = funcionario.Nome };
        return Resultado<HoleritePublicadoDto>.Ok(new HoleritePublicadoDto(MapeadorHolerite.Holerite(holerite, nomes), existente is not null));
    }

    // Limpeza de erro: nunca esconde a falha original.
    private async Task TentarRemoverAsync(string caminho)
    {
        try { await _armazenamento.RemoverAsync(caminho, CancellationToken.None); }
        catch { /* arquivo órfão fica na pasta; o erro original é o que importa */ }
    }
}
```

```csharp
        _armazenamento = armazenamento;
        _auditoria = auditoria;
        _unidade = unidade;
    }

    public async Task<Resultado<ArquivoHoleriteDto>> ExecutarAsync(
        Solicitante solicitante, int idHolerite, CancellationToken cancelamento = default)
    {
        var holerite = await _holerites.ObterAsync(idHolerite, cancelamento);
        var ehDono = holerite is not null && holerite.IdFuncionario == solicitante.IdFuncionario;
        if (holerite is null || (!ehDono && !solicitante.Perfil.EhRhOuAdmin()))
            return Resultado<ArquivoHoleriteDto>.Falha(TipoErro.NaoEncontrado, "Holerite não encontrado.");

        var conteudo = RegrasHolerite.CaminhoEhSeguro(holerite.Arquivo)
            ? await _armazenamento.LerAsync(holerite.Arquivo, cancelamento)
            : null;
        if (conteudo is null)
            return Resultado<ArquivoHoleriteDto>.Falha(TipoErro.NaoEncontrado, "O arquivo deste holerite não está disponível. Procure o RH.");

        if (!ehDono)
        {
            _auditoria.Registrar(solicitante.IdUsuario, "ConsultaHolerite", "HOLERITE", holerite.Id,
                $"{CodigoFuncionario.Formatar(holerite.IdFuncionario)}: {holerite.Mes:D2}/{holerite.Ano}");
            await _unidade.SalvarAsync(cancelamento);
        }

        return Resultado<ArquivoHoleriteDto>.Ok(
            new ArquivoHoleriteDto(conteudo, RegrasHolerite.NomeDoArquivoParaDownload(holerite), "application/pdf"));
    }
```

```csharp
/// <summary>Seção "Holerites" do appsettings. A pasta é relativa ao diretório de execução (como o banco SQLite).</summary>
public sealed class OpcoesHolerites
{
    public const string Secao = "Holerites";

    /// <summary>Pasta raiz dos PDFs. Fica fora do wwwroot: os arquivos só saem pela API, com autenticação (RNF08).</summary>
    public string Pasta { get; set; } = Path.Combine("dados", "holerites");

/// <summary>
/// Guarda os PDFs dos holerites em disco (RNF17). Os nomes são gerados pelo sistema (<see cref="RegrasHolerite.MontarCaminho"/>),
/// nunca pelo usuário, e todo caminho é conferido duas vezes antes de tocar no disco: formato permitido e dentro da pasta raiz.
/// </summary>
public sealed class ArmazenamentoHoleritesEmDisco : IArmazenamentoHolerites
{
    private readonly string _raiz;

    public ArmazenamentoHoleritesEmDisco(IOptions<OpcoesHolerites> opcoes)
    {
        var pasta = string.IsNullOrWhiteSpace(opcoes.Value.Pasta) ? new OpcoesHolerites().Pasta : opcoes.Value.Pasta;
        _raiz = Path.GetFullPath(pasta);
    }

    public async Task<string> SalvarAsync(
        int idFuncionario, int ano, int mes, ReadOnlyMemory<byte> conteudo, CancellationToken cancelamento = default)
    {
        var relativo = RegrasHolerite.MontarCaminho(idFuncionario, ano, mes, Guid.NewGuid());
        var completo = Resolver(relativo) ?? throw new InvalidOperationException("Caminho de holerite inválido.");

        Directory.CreateDirectory(Path.GetDirectoryName(completo)!);
        try
        {
            // CreateNew: nunca sobrescreve um arquivo existente.
            await using var arquivo = new FileStream(
                completo, FileMode.CreateNew, FileAccess.Write, FileShare.None, 4096, useAsync: true);
            await arquivo.WriteAsync(conteudo, cancelamento);
        }
        catch
        {
            try { File.Delete(completo); } catch (IOException) { /* arquivo parcial fica; o erro original é repassado */ }
            throw;
        }

        return relativo;
    }

    public async Task<byte[]?> LerAsync(string caminhoRelativo, CancellationToken cancelamento = default)
    {
        var completo = Resolver(caminhoRelativo);
        if (completo is null || !File.Exists(completo)) return null;
        return await File.ReadAllBytesAsync(completo, cancelamento);
    }

    public Task RemoverAsync(string caminhoRelativo, CancellationToken cancelamento = default)
    {
        var completo = Resolver(caminhoRelativo);
        if (completo is not null && File.Exists(completo)) File.Delete(completo);
        return Task.CompletedTask;
    }

    // Caminho absoluto, ou nulo se o formato não é o permitido ou se escapa da pasta raiz.
    private string? Resolver(string caminhoRelativo)
    {
        if (!RegrasHolerite.CaminhoEhSeguro(caminhoRelativo)) return null;

        var completo = Path.GetFullPath(Path.Combine(_raiz, caminhoRelativo));
        var raizComSeparador = _raiz.EndsWith(Path.DirectorySeparatorChar) ? _raiz : _raiz + Path.DirectorySeparatorChar;
        return completo.StartsWith(raizComSeparador, StringComparison.Ordinal) ? completo : null;
    }
}
```

```csharp
            return Falha(TipoErro.Validacao, "Envie o arquivo PDF do holerite.");
        if (form.Arquivo.Length > RegrasHolerite.TamanhoMaximoBytes)
            return Falha(TipoErro.Validacao, "O arquivo do holerite deve ter no máximo 5 MB.");

        // O tamanho já foi conferido; o buffer tem exatamente o tamanho do arquivo.
        var conteudo = new byte[form.Arquivo.Length];
        await using (var leitura = form.Arquivo.OpenReadStream())
            await leitura.ReadExactlyAsync(conteudo, ct);

        var resultado = await caso.ExecutarAsync(
            solicitante,
            new PublicarHoleriteDto(form.Codigo, form.Mes, form.Ano, form.Arquivo.FileName, conteudo, form.Substituir),
            ct);
        if (!resultado.Sucesso) return Falha(resultado.TipoDoErro, resultado.Erro);

        return StatusCode(resultado.Valor!.Substituido ? StatusCodes.Status200OK : StatusCodes.Status201Created, resultado.Valor);
    }
}
```
