import 'colaborador_resumo.dart';

/// Batidas na ordem obrigatória (RN04): entrada, saída para o intervalo, retorno e saída.
enum TipoBatida {
  entrada('Entrada'),
  saidaIntervalo('Saída para intervalo'),
  retornoIntervalo('Retorno do intervalo'),
  saida('Saída');

  const TipoBatida(this.rotulo);
  final String rotulo;

  /// Próxima batida liberada, ou null depois da saída.
  TipoBatida? get proxima => index < values.length - 1 ? values[index + 1] : null;
}

class Batida {
  const Batida({required this.tipo, required this.horario, this.offline = false});

  final TipoBatida tipo;
  final DateTime horario; // horário do servidor (RF07)

  /// Feita sem internet: mantém o horário do aparelho e vai para conferência do gestor (RN05).
  final bool offline;
}

/// Um dia do espelho de ponto, com as horas calculadas pela carga do cargo (RF11).
class DiaEspelho {
  const DiaEspelho({required this.data, required this.batidas, required this.cargaHoras});

  final DateTime data;
  final List<Batida> batidas;
  final double cargaHoras;

  Batida? batida(TipoBatida tipo) {
    for (final b in batidas) {
      if (b.tipo == tipo) return b;
    }
    return null;
  }

  /// Soma dos períodos fechados (entrada -> intervalo, retorno -> saída).
  Duration get trabalhado {
    Duration periodo(TipoBatida inicio, TipoBatida fim) {
      final a = batida(inicio);
      final b = batida(fim);
      if (a == null || b == null) return Duration.zero;
      return b.horario.difference(a.horario);
    }

    return periodo(TipoBatida.entrada, TipoBatida.saidaIntervalo) +
        periodo(TipoBatida.retornoIntervalo, TipoBatida.saida);
  }

  bool get completo => batida(TipoBatida.saida) != null;

  /// Crédito (positivo) ou débito (negativo) do banco de horas. Zero enquanto o dia não fecha.
  Duration get saldo =>
      completo ? trabalhado - Duration(minutes: (cargaHoras * 60).round()) : Duration.zero;
}

/// Espelho de ponto de quem está logado: semana atual e saldo do banco de horas.
class EspelhoPonto {
  const EspelhoPonto({
    required this.agora,
    required this.semana,
    required this.saldoBancoHoras,
    required this.minhasSolicitacoes,
  });

  final DateTime agora;

  /// Segunda a sexta da semana atual; o dia de hoje inclui as batidas já feitas.
  final List<DiaEspelho> semana;
  final Duration saldoBancoHoras;
  final List<AjustePonto> minhasSolicitacoes;

  DiaEspelho? get hoje {
    for (final d in semana) {
      if (d.data.year == agora.year && d.data.month == agora.month && d.data.day == agora.day) return d;
    }
    return null;
  }

  /// Próxima batida liberada hoje (null quando o dia já fechou).
  TipoBatida? get proximaBatida {
    final batidas = hoje?.batidas ?? const [];
    return batidas.isEmpty ? TipoBatida.entrada : batidas.last.tipo.proxima;
  }
}

enum SituacaoAjuste { pendente, aprovado, recusado }

/// Pedido de correção de uma batida (MANUTENCAO_PONTO), aprovado ou recusado pelo gestor (RF10).
class AjustePonto {
  const AjustePonto({
    required this.id,
    required this.colaborador,
    required this.data,
    required this.tipoBatida,
    required this.horarioSolicitado,
    required this.justificativa,
    required this.enviadoEm,
    this.horarioRegistrado,
    this.situacao = SituacaoAjuste.pendente,
    this.observacao,
  });

  final String id;
  final ColaboradorResumo colaborador;
  final DateTime data;
  final TipoBatida tipoBatida;

  /// "08:47"; null quando a batida não foi registrada.
  final String? horarioRegistrado;
  final String horarioSolicitado;
  final String justificativa;
  final DateTime enviadoEm;
  final SituacaoAjuste situacao;

  /// Observação de quem decidiu (motivo da recusa ou nota da aprovação). Null enquanto pendente
  /// ou quando o gestor não escreveu nada.
  final String? observacao;

  /// Copia trocando a decisão. Voltar para [SituacaoAjuste.pendente] apaga a observação;
  /// nos demais casos, [observacao] nova vence e, sem ela, a anterior é mantida.
  AjustePonto copyWith({SituacaoAjuste? situacao, String? observacao}) {
    final novaSituacao = situacao ?? this.situacao;
    return AjustePonto(
      id: id,
      colaborador: colaborador,
      data: data,
      tipoBatida: tipoBatida,
      horarioRegistrado: horarioRegistrado,
      horarioSolicitado: horarioSolicitado,
      justificativa: justificativa,
      enviadoEm: enviadoEm,
      situacao: novaSituacao,
      observacao: novaSituacao == SituacaoAjuste.pendente ? null : (observacao ?? this.observacao),
    );
  }
}

  Future<Funcionario> cadastrar(NovoFuncionario dados) async {
    final novo = await ref.read(adminRepositoryProvider).cadastrarFuncionario(dados);
    state = AsyncData([novo, ...state.valueOrNull ?? const []]);
    ref.invalidate(proximoCodigoProvider);
    ref.invalidate(dashboardAdminViewModelProvider); // o quadro ativo mudou
    return novo;
  }

  /// Inativa ou reativa. Lança [Falha] se não gravar.

      for (final f in state.valueOrNull ?? const <Funcionario>[]) f.id == funcionario.id ? f.copyWith(ativo: ativo) : f,

    ref.invalidate(dashboardAdminViewModelProvider);

        f.id == ficha.id ? ficha.copyWith(ativo: f.ativo) : f,
    ]);
    ref.invalidate(consultasDadosPessoaisProvider);
    return ficha;
  }
}

import 'package:flutter_riverpod/flutter_riverpod.dart';

import '../../core/errors/falha.dart';
import '../../domain/entities/dashboard_admin.dart';

/// ViewModel do dashboard do Admin (MVVM): carrega os indicadores e aplica as decisões
/// sobre os ajustes de ponto. A decisão aparece na hora (otimista) e volta se o repositório falhar.
class DashboardAdminViewModel extends AsyncNotifier<DashboardAdmin> {
  String? _ultimoErro;

  /// Mensagem da última decisão que falhou (já em português, vinda da API quando existe).
  /// Null se a última decisão deu certo ou ainda não houve nenhuma.
  String? get ultimoErro => _ultimoErro;

  @override
  Future<DashboardAdmin> build() => ref.read(adminRepositoryProvider).carregarDashboard();

  /// Aprova, recusa ou desfaz ([SituacaoAjuste.pendente]). [observacao] é o motivo da recusa.
  /// Devolve false se não foi gravado; nesse caso, [ultimoErro] explica quando a API disse o motivo.
  Future<bool> decidir(String id, SituacaoAjuste situacao, {String? observacao}) async {
    final anterior = _situacaoDe(id);
    if (anterior == null) return false;

    _ultimoErro = null;
    _alterar(id, situacao, observacao: observacao);
    try {
      await ref.read(adminRepositoryProvider).decidirAjuste(id: id, situacao: situacao, observacao: observacao);
      return true;
    } on Falha catch (e) {
      _ultimoErro = e.mensagem;
      _alterar(id, anterior);
      return false;
    } catch (_) {
      _alterar(id, anterior);
      return false;
    }
  }

  /// Recarrega do repositório (botão "Tentar novamente" e depois de mudanças em outras abas).
  Future<void> recarregar() async {
    state = const AsyncLoading();
    state = await AsyncValue.guard(() => ref.read(adminRepositoryProvider).carregarDashboard());
  }

  SituacaoAjuste? _situacaoDe(String id) {
    for (final a in state.valueOrNull?.ajustes ?? const <AjustePonto>[]) {
      if (a.id == id) return a.situacao;
    }
    return null;
  }

  void _alterar(String id, SituacaoAjuste situacao, {String? observacao}) {
    final atual = state.valueOrNull;
    if (atual == null) return;
    state = AsyncData(atual.copyWith(
      ajustes: [
        for (final a in atual.ajustes) a.id == id ? a.copyWith(situacao: situacao, observacao: observacao) : a,
      ],
    ));
  }
}

final dashboardAdminViewModelProvider =
    AsyncNotifierProvider<DashboardAdminViewModel, DashboardAdmin>(DashboardAdminViewModel.new);

/// Pede o motivo ao recusar um pedido de ajuste. A API exige no mínimo 5 caracteres e guarda o texto
/// na observação do pedido, que o colaborador lê em "Minhas solicitações de ajuste".
class DialogoMotivoRecusa extends StatefulWidget {
  const DialogoMotivoRecusa({super.key, required this.colaborador});

  final String colaborador;

  static const int motivoMinimo = 5;
  static const int motivoMaximo = 500;

  @override
  State<DialogoMotivoRecusa> createState() => _DialogoMotivoRecusaState();

class _DialogoMotivoRecusaState extends State<DialogoMotivoRecusa> {
  final _form = GlobalKey<FormState>();
  final _motivo = TextEditingController();

  @override
  void dispose() {
    _motivo.dispose();
    super.dispose();
  }

  String? _validar(String? valor) {
    final texto = valor?.trim() ?? '';
    if (texto.isEmpty) return 'Informe o motivo da recusa.';
    if (texto.length < DialogoMotivoRecusa.motivoMinimo) {
      return 'Use pelo menos ${DialogoMotivoRecusa.motivoMinimo} caracteres.';
    }
    if (texto.length > DialogoMotivoRecusa.motivoMaximo) {
      return 'Use no máximo ${DialogoMotivoRecusa.motivoMaximo} caracteres.';

    return null;

    if (!_form.currentState!.validate()) return;
    Navigator.of(context).pop(_motivo.text.trim());
  }

  Widget build(BuildContext context) {
    return DialogoAdmin(
      titulo: 'Recusar ajuste de ponto',

      rotuloConfirmar: 'Recusar pedido',

      conteudo: Form(

  static String telefoneFormatado(String bruto) {
    final d = Formatadores.digitos(bruto);
    if (d.length == 11) return '(${d.substring(0, 2)}) ${d.substring(2, 7)}-${d.substring(7)}';
    if (d.length == 10) return '(${d.substring(0, 2)}) ${d.substring(2, 6)}-${d.substring(6)}';
    return bruto.trim().isEmpty ? _vazio : bruto;
