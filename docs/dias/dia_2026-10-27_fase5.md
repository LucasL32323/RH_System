- `ApiClient` ganhou `getBytes` (download binário) e `postMultipart` (upload) com timeout de 60 s e mensagem para o status 413.
- Os plugins (`file_picker`, `file_saver`) ficam isolados em `data/files/arquivos_plataforma.dart`; o resto do app usa as interfaces `SeletorArquivo` e `SalvadorArquivo` (por isso os testes não dependem de plugin).

```dart
import 'dart:async';
import 'dart:convert';
import 'dart:typed_data';

import 'package:http/http.dart' as http;
import 'package:http_parser/http_parser.dart' show MediaType;

import '../config/app_config.dart';
import '../errors/falha.dart';

/// Erro de comunicação com a API. Herda de [Falha]: a mensagem já serve para o usuário.
class ApiException extends Falha {
  const ApiException(super.mensagem, {this.statusCode});

  final int? statusCode;

  @override
  String toString() => 'ApiException($statusCode): $mensagem';
}

/// Cliente HTTP único do app.
/// A API .NET responde erros em ProblemDetails (application/problem+json): o campo
/// "detail" já vem em português e é o que mostramos ao usuário.
class ApiClient {
  ApiClient({http.Client? cliente, String? baseUrl})
      : _cliente = cliente ?? http.Client(),
        _baseUrl = baseUrl ?? AppConfig.apiBaseUrl;

  final http.Client _cliente;
  final String _baseUrl;

  /// Tempo máximo para enviar ou baixar um arquivo (PDF de até 5 MB, RNF17).
  /// É maior que o [AppConfig.timeoutApi] das chamadas JSON.
  static const Duration timeoutArquivos = Duration(seconds: 60);

  /// POST com corpo JSON. Devolve o objeto JSON da resposta, ou null quando não há corpo (204).
  Future<Map<String, dynamic>?> postJson(
    String caminho,
    Map<String, dynamic> corpo, {
    String? token,
  }) async =>
      _comoObjeto(await _enviar('POST', caminho, corpo: corpo, token: token));

  /// PUT com corpo JSON (atualização completa). Devolve o objeto da resposta ou null (204).
  Future<Map<String, dynamic>?> putJson(
    String caminho,
    Map<String, dynamic> corpo, {
    String? token,
  }) async =>
      _comoObjeto(await _enviar('PUT', caminho, corpo: corpo, token: token));

  /// PATCH com corpo JSON (atualização parcial). Devolve o objeto da resposta ou null (204).
  Future<Map<String, dynamic>?> patchJson(
    String caminho,
    Map<String, dynamic> corpo, {
    String? token,
  }) async =>
      _comoObjeto(await _enviar('PATCH', caminho, corpo: corpo, token: token));

  /// DELETE. A API responde 204 sem corpo.
  Future<void> delete(String caminho, {String? token}) async {
    await _enviar('DELETE', caminho, token: token);
  }

  /// GET que espera um objeto JSON na resposta. [query] vira a query string (?de=...&ate=...).
  Future<Map<String, dynamic>> getJson(
    String caminho, {
    String? token,
    Map<String, String>? query,
  }) async {
    final json = _comoObjeto(await _enviar('GET', caminho, token: token, query: query));
    if (json == null) throw const ApiException('Resposta da API em formato inesperado.');
    return json;
  }

  /// GET que espera um array JSON na resposta (ex.: /cargos, /departamentos e /holerites).
  Future<List<dynamic>> getLista(
    String caminho, {
    String? token,
    Map<String, String>? query,
  }) async {
    final json = await _enviar('GET', caminho, token: token, query: query);
    if (json is! List) throw const ApiException('Resposta da API em formato inesperado.');
    return json;
  }

  /// GET de um arquivo (ex.: PDF do holerite). Devolve os bytes da resposta sem interpretá-los.
  /// Erros chegam como [ApiException] com a mensagem do ProblemDetails (que vem em JSON, não em binário).
  Future<Uint8List> getBytes(
    String caminho, {
    String? token,
    Map<String, String>? query,
  }) =>
      _protegido(() async {
        final resposta = await _cliente
            .get(_uri(caminho, query), headers: _cabecalhos(token: token, aceita: 'application/pdf, application/json'))
            .timeout(timeoutArquivos);

        if (!_sucesso(resposta.statusCode)) {
          throw ApiException(
            _mensagemDeErro(
              resposta.statusCode,
              utf8.decode(resposta.bodyBytes, allowMalformed: true),
              autenticado: token != null,
            ),
            statusCode: resposta.statusCode,
          );
        }
        return resposta.bodyBytes;
      });

  /// POST multipart/form-data com UM arquivo e campos de texto (ex.: publicar holerite).
  /// O Content-Type (com o boundary) é montado pelo próprio pacote http. Devolve o objeto JSON da resposta
  /// ou null (204).
  Future<Map<String, dynamic>?> postMultipart(
    String caminho, {
    required Map<String, String> campos,
    required String campoArquivo,
    required String nomeArquivo,
    required List<int> bytes,
    String tipoConteudo = 'application/pdf',
    String? token,
  }) async {
    final json = await _protegido(() async {
      final requisicao = http.MultipartRequest('POST', _uri(caminho))
        ..headers.addAll(_cabecalhos(token: token))
        ..fields.addAll(campos)
        ..files.add(http.MultipartFile.fromBytes(
          campoArquivo,
          bytes,
          filename: nomeArquivo,
          contentType: MediaType.parse(tipoConteudo),
        ));

      final resposta = await (() async {
        final enviada = await _cliente.send(requisicao);
        return http.Response.fromStream(enviada);
      })()
          .timeout(timeoutArquivos);

      return _lerJson(resposta, autenticado: token != null);
    });
    return _comoObjeto(json);
  }

  /// Aceita só objeto JSON (ou vazio). Qualquer outra forma é resposta inesperada.
  static Map<String, dynamic>? _comoObjeto(Object? json) {
    if (json == null) return null;
    if (json is Map<String, dynamic>) return json;
    throw const ApiException('Resposta da API em formato inesperado.');
  }

  static bool _sucesso(int status) => status >= 200 && status < 300;

  Uri _uri(String caminho, [Map<String, String>? query]) {
    final base = Uri.parse('$_baseUrl$caminho');
    return (query == null || query.isEmpty)
        ? base
        : base.replace(queryParameters: {...base.queryParameters, ...query});
  }

  static Map<String, String> _cabecalhos({String? token, bool corpoJson = false, String aceita = 'application/json'}) => {
        'Accept': aceita,
        if (corpoJson) 'Content-Type': 'application/json; charset=utf-8',
        if (token != null) 'Authorization': 'Bearer $token',
      };

  /// Converte as falhas de rede em [ApiException] com mensagem para o usuário.
  Future<T> _protegido<T>(Future<T> Function() acao) async {
    try {
      return await acao();
    } on ApiException {
      rethrow;
    } on TimeoutException {
      throw const ApiException('O servidor demorou demais para responder. Tente novamente.');
    } on http.ClientException {
      throw const ApiException('Não foi possível conectar ao servidor. Verifique sua conexão.');
    } on FormatException {
      throw const ApiException('Resposta da API em formato inesperado.');
    }
  }

  /// Confere o status e decodifica o JSON (objeto ou array); null se o corpo vier vazio.
  static Object? _lerJson(http.Response resposta, {required bool autenticado}) {
    final conteudo = utf8.decode(resposta.bodyBytes);

    if (!_sucesso(resposta.statusCode)) {
      throw ApiException(
        _mensagemDeErro(resposta.statusCode, conteudo, autenticado: autenticado),
        statusCode: resposta.statusCode,
      );
    }

    if (conteudo.trim().isEmpty) return null;

    final json = jsonDecode(conteudo);
    if (json is! Map<String, dynamic> && json is! List) {
      throw const ApiException('Resposta da API em formato inesperado.');
    }
    return json;
  }

  /// Envia a requisição JSON e devolve o JSON decodificado (objeto ou array), ou null se o corpo vier vazio.
  Future<Object?> _enviar(
    String metodo,
    String caminho, {
    Map<String, dynamic>? corpo,
    String? token,
    Map<String, String>? query,
  }) =>
      _protegido(() async {
        final uri = _uri(caminho, query);
        final temCorpo = metodo == 'POST' || metodo == 'PUT' || metodo == 'PATCH';
        final cabecalhos = _cabecalhos(token: token, corpoJson: temCorpo);
        final texto = jsonEncode(corpo ?? const <String, dynamic>{});

        final requisicao = switch (metodo) {
          'GET' => _cliente.get(uri, headers: cabecalhos),
          'POST' => _cliente.post(uri, headers: cabecalhos, body: texto),
          'PUT' => _cliente.put(uri, headers: cabecalhos, body: texto),
          'PATCH' => _cliente.patch(uri, headers: cabecalhos, body: texto),
          'DELETE' => _cliente.delete(uri, headers: cabecalhos),
          _ => throw ArgumentError.value(metodo, 'metodo', 'Método HTTP não suportado'),
        };
        final resposta = await requisicao.timeout(AppConfig.timeoutApi);
        return _lerJson(resposta, autenticado: token != null);
      });

  /// Usa o "detail" do ProblemDetails; sem ele, uma mensagem padrão por status.
  /// [autenticado] = a chamada levava token: um 401 aí significa sessão expirada, não senha errada.
  static String _mensagemDeErro(int status, String corpo, {required bool autenticado}) {
    try {
      final json = jsonDecode(corpo);
      if (json is Map<String, dynamic>) {
        final detalhe = json['detail'] ?? json['title'];
        if (detalhe is String && detalhe.trim().isNotEmpty) return detalhe;
      }
    } on FormatException {
      // Corpo vazio ou não-JSON: cai na mensagem padrão.
    }

    return switch (status) {
      401 when autenticado => 'Sua sessão expirou. Entre novamente.',
      401 => 'E-mail ou senha inválidos.',
      403 => 'Você não tem permissão para esta operação.',
      413 => 'O arquivo enviado é grande demais.',
      429 => 'Muitas tentativas em pouco tempo. Aguarde um instante e tente de novo.',
      >= 500 => 'O servidor está indisponível. Tente novamente em instantes.',
      _ => 'Não foi possível concluir a operação (erro $status).',
    };
  }

  void dispose() => _cliente.close();
}
```

### hr_auto_login/lib/domain/regras/regras_holerite.dart

```dart
import 'dart:typed_data';

import '../../core/errors/falha.dart';

/// Regras de holerite que o app confere ANTES de chamar a API (as mesmas do back-end, RNF17):
/// código FUNC válido, mês e ano possíveis, arquivo PDF não vazio e de até 5 MB.
/// O servidor repete todas as conferências e ainda valida a competência contra a data real e a admissão.
class RegrasHolerite {
  RegrasHolerite._();

  /// RNF17: no máximo 5 MB por arquivo.
  static const int tamanhoMaximoBytes = 5 * 1024 * 1024;

  static const int anoMinimo = 2000;

  static final RegExp _codigo = RegExp(r'^FUNC-(\d{4,})$');
  static final RegExp _soDigitos = RegExp(r'^\d{1,9}$');

  /// "func-0004" -> "FUNC-0004"; "4" -> "FUNC-0004". Devolve null se não der para formar um código válido.
  static String? normalizarCodigo(String? texto) {
    final t = (texto ?? '').trim().toUpperCase();
    if (_codigo.hasMatch(t)) return t;
    if (_soDigitos.hasMatch(t)) {
      final numero = int.parse(t);
      if (numero <= 0) return null;
      return 'FUNC-${numero.toString().padLeft(4, '0')}';
    }
    return null;
  }

  /// O arquivo começa com a assinatura de PDF ("%PDF-").
  static bool pareceUmPdf(Uint8List bytes) =>
      bytes.length >= 5 && bytes[0] == 0x25 && bytes[1] == 0x50 && bytes[2] == 0x44 && bytes[3] == 0x46 && bytes[4] == 0x2D;

  /// Lança [Falha] com a mensagem para o usuário se o arquivo não puder ser publicado.
  static void validarArquivo(String nome, Uint8List bytes) {
    if (bytes.isEmpty) throw const Falha('O arquivo do holerite está vazio.');
    if (bytes.length > tamanhoMaximoBytes) {
      throw Falha('O arquivo do holerite deve ter no máximo ${tamanhoMaximoBytes ~/ (1024 * 1024)} MB.');
    }
    if (!nome.toLowerCase().endsWith('.pdf')) throw const Falha('O holerite deve ser um arquivo PDF.');
    if (!pareceUmPdf(bytes)) throw const Falha('O arquivo escolhido não é um PDF válido.');
  }

  /// Confere tudo da publicação e devolve o código normalizado ("FUNC-0004").

    if (ano < anoMinimo) throw const Falha('O ano do holerite deve ser $anoMinimo ou posterior.');
    validarArquivo(nomeArquivo, conteudo);
    return codigoOk;
  }

  /// "1,2 MB", "340 KB" ou "512 B" (para mostrar o tamanho do arquivo escolhido).
  static String tamanhoLegivel(int bytes) {
    if (bytes >= 1024 * 1024) return '${(bytes / (1024 * 1024)).toStringAsFixed(1).replaceAll('.', ',')} MB';
    if (bytes >= 1024) return '${(bytes / 1024).round()} KB';
    return '$bytes B';
```

```dart
import 'dart:typed_data';

import '../../core/errors/falha.dart';
import '../../core/network/api_client.dart';
import '../../domain/entities/holerite.dart';
import '../../domain/regras/regras_holerite.dart';

/// Holerites na API .NET (RF29, RF30; ver `dia_2026-10-24_fase4.md`):
///  - GET  /api/v1/holerites?codigo=&ano=   -> [ HoleriteDto ]  (sem código = os de quem está logado)
///  - GET  /api/v1/holerites/{id}/arquivo   -> PDF (application/pdf)
///  - POST /api/v1/holerites (multipart)    -> 201 novo / 200 substituído { holerite, substituido }
/// O token vem da sessão na hora da chamada. O caminho do PDF no servidor nunca é exposto (RNF08).
class HoleriteRepositoryApi implements HoleriteRepository {
  HoleriteRepositoryApi(this._api, {required String? Function() token}) : _token = token;

  final ApiClient _api;
  final String? Function() _token;

  String _tokenObrigatorio() {
    final token = _token();
    if (token == null || token.isEmpty) throw const Falha('Sua sessão expirou. Entre novamente.');
    return token;
  }

  @override
  Future<List<Holerite>> listar({String? codigo, int? ano}) async {
    String? codigoOk;
    if (codigo != null && codigo.trim().isNotEmpty) {
      codigoOk = RegrasHolerite.normalizarCodigo(codigo);
      if (codigoOk == null) throw const Falha('Código inválido. Use o formato FUNC-0001.');
    }

    final itens = await _api.getLista(
      '/api/v1/holerites',
      token: _tokenObrigatorio(),
      query: {if (codigoOk != null) 'codigo': codigoOk, if (ano != null) 'ano': '$ano'},
    );
    return _protegido(() => [for (final j in itens) holeriteDeJson(j as Map<String, dynamic>)]);
  }

  @override
  Future<ArquivoHolerite> baixar(Holerite holerite) async {
    final bytes = await _api.getBytes('/api/v1/holerites/${holerite.id}/arquivo', token: _tokenObrigatorio());
    // Defesa: a API devolve PDF; qualquer outra coisa (página de erro de proxy, por exemplo) não é salva como holerite.
    if (!RegrasHolerite.pareceUmPdf(bytes)) {
      throw const ApiException('O servidor não devolveu um PDF válido. Procure o RH.');
    }
    return ArquivoHolerite(nome: holerite.nomeArquivo, bytes: bytes);
  }

  @override
  Future<HoleritePublicado> publicar({
    required String codigo,
    required int mes,
    required int ano,
    required String nomeArquivo,
    required Uint8List conteudo,
    bool substituir = false,
  }) async {
    // Tudo que dá para conferir sem rede é conferido antes de enviar até 5 MB.
    final codigoOk = RegrasHolerite.validarPublicacao(
      codigo: codigo,
      mes: mes,
      ano: ano,
      nomeArquivo: nomeArquivo,
      conteudo: conteudo,
    );

    final json = await _api.postMultipart(
      '/api/v1/holerites',
      campos: {
        'Codigo': codigoOk,
        'Mes': '$mes',
        'Ano': '$ano',
        if (substituir) 'Substituir': 'true',
      },
      campoArquivo: 'Arquivo',
      nomeArquivo: nomeArquivo,
      bytes: conteudo,
      token: _tokenObrigatorio(),
    );
    if (json == null) throw const ApiException('Resposta da API em formato inesperado.');

    return _protegido(() => HoleritePublicado(
          holerite: holeriteDeJson(json['holerite'] as Map<String, dynamic>),
          substituido: json['substituido'] as bool? ?? false,
        ));
  }

  /// Converte erro de leitura do JSON em mensagem para o usuário (a API mudou ou veio incompleta).
  static T _protegido<T>(T Function() leitura) {
    try {
      return leitura();
    } on TypeError {
      throw const ApiException('Resposta da API em formato inesperado.');
    } on FormatException {
      throw const ApiException('Resposta da API em formato inesperado.');
    }
  }

  /// HoleriteDto -> [Holerite] (público e estático para os testes).
  static Holerite holeriteDeJson(Map<String, dynamic> j) => Holerite(
        id: (j['id'] as num).toInt(),
        codigoFuncionario: j['codigoFuncionario'] as String,
        nomeFuncionario: j['nomeFuncionario'] as String? ?? '',
        mes: (j['mes'] as num).toInt(),
        ano: (j['ano'] as num).toInt(),
        dataPublicacao: _lerData(j['dataPublicacao'] as String),
        nomeArquivo: j['nomeArquivo'] as String? ?? 'holerite.pdf',
      );

  /// "2026-10-21" -> DateTime(2026, 10, 21)
  static DateTime _lerData(String texto) {
    final m = RegExp(r'^(\d{4})-(\d{2})-(\d{2})').firstMatch(texto);
    if (m == null) throw FormatException('Data inválida: $texto');
    return DateTime(int.parse(m.group(1)!), int.parse(m.group(2)!), int.parse(m.group(3)!));
  }
}
```

```dart
/// Holerites em memória para validar as telas sem subir a API (PROTOTIPO=true).
/// Simula as regras do servidor: só RH/Admin consulta ou publica para outra pessoa, ninguém publica o próprio,
/// competência repetida dá conflito sem "substituir". Os PDFs de exemplo são gerados na hora.
class HoleriteRepositoryPrototipo implements HoleriteRepository {
  HoleriteRepositoryPrototipo({
    this.codigoUsuario = 'FUNC-0001',
    this.nomeUsuario = 'Administrador do Sistema',
    this.perfilUsuario = 'Admin',
    this.latencia = Duration.zero,
    DateTime Function()? relogio,
  }) : _relogio = relogio ?? DateTime.now {
    // Três meses anteriores já publicados para o usuário do protótipo.
    final hoje = _relogio();
    for (var i = 1; i <= 3; i++) {
      final competencia = DateTime(hoje.year, hoje.month - i);
      var publicadoEm = DateTime(competencia.year, competencia.month + 1, 5);
      if (publicadoEm.isAfter(hoje)) publicadoEm = hoje; // nunca publicado no futuro
      _adicionar(codigoUsuario, nomeUsuario, competencia.month, competencia.year, publicadoEm);
    }
  }

  /// Quem está logado (o protótipo não tem sessão própria; o provider informa).
  final String codigoUsuario;
  final String nomeUsuario;
  final String perfilUsuario;

  /// Atraso artificial para ver o carregamento (zero nos testes).
  final Duration latencia;

  final DateTime Function() _relogio;

  final List<_Registro> _registros = [];
  int _proximoId = 1;

  /// Funcionários que o protótipo conhece (para consultar/publicar por código).
  static const Map<String, String> _conhecidos = {
    'FUNC-0002': 'Renata Lopes',
    'FUNC-0003': 'Bruno Lima',
    'FUNC-0004': 'Ana Souza',
  };

  bool get _rhOuAdmin => perfilUsuario == 'RH' || perfilUsuario == 'Admin';

  String? _nomeDe(String codigo) => codigo == codigoUsuario ? nomeUsuario : _conhecidos[codigo];

  /// Guarda o holerite com o [pdf] informado (ou um PDF de exemplo) e devolve a entidade criada.
  Holerite _adicionar(String codigo, String nome, int mes, int ano, DateTime publicadoEm, {Uint8List? pdf}) {
    final holerite = Holerite(
      id: _proximoId++,
      codigoFuncionario: codigo,
      nomeFuncionario: nome,
      mes: mes,
      ano: ano,
      dataPublicacao: DateTime(publicadoEm.year, publicadoEm.month, publicadoEm.day),
      nomeArquivo: 'holerite-$ano-${mes.toString().padLeft(2, '0')}.pdf',
    );
    _registros.add(_Registro(holerite, pdf ?? gerarPdfDeExemplo('Holerite ${holerite.competencia} - $codigo')));
    return holerite;
  }

  @override
  Future<List<Holerite>> listar({String? codigo, int? ano}) async {
    await Future<void>.delayed(latencia);

    var alvo = codigoUsuario;
    if (codigo != null && codigo.trim().isNotEmpty) {
      final normalizado = RegrasHolerite.normalizarCodigo(codigo);
      if (normalizado == null) throw const Falha('Código inválido. Use o formato FUNC-0001.');
      alvo = normalizado;
    }
    if (alvo != codigoUsuario) {
      if (!_rhOuAdmin) throw const Falha('Você só pode ver os seus próprios holerites.');
      if (_nomeDe(alvo) == null) throw const Falha('Funcionário não encontrado.');
    }

    final lista = [
      for (final r in _registros)
        if (r.holerite.codigoFuncionario == alvo && (ano == null || r.holerite.ano == ano)) r.holerite,
    ]..sort((a, b) => b.ano != a.ano ? b.ano.compareTo(a.ano) : (b.mes != a.mes ? b.mes.compareTo(a.mes) : b.id.compareTo(a.id)));
    return lista;
  }

  @override
  Future<ArquivoHolerite> baixar(Holerite holerite) async {
    await Future<void>.delayed(latencia);
    final registro = _registros.where((r) => r.holerite.id == holerite.id).firstOrNull;
    final permitido = registro != null && (registro.holerite.codigoFuncionario == codigoUsuario || _rhOuAdmin);
    if (!permitido) throw const Falha('Holerite não encontrado.');
    return ArquivoHolerite(nome: registro.holerite.nomeArquivo, bytes: registro.pdf);
  }

  @override
  Future<HoleritePublicado> publicar({
    required String codigo,
    required int mes,
    required int ano,
    required String nomeArquivo,
    required Uint8List conteudo,
    bool substituir = false,
  }) async {
    await Future<void>.delayed(latencia);

    if (!_rhOuAdmin) throw const Falha('Apenas o RH pode publicar holerites.');
    final codigoOk = RegrasHolerite.validarPublicacao(
      codigo: codigo,
      mes: mes,
      ano: ano,
      nomeArquivo: nomeArquivo,
      conteudo: conteudo,
    );
    if (codigoOk == codigoUsuario) throw const Falha('Você não pode publicar o seu próprio holerite.');
    final nome = _nomeDe(codigoOk);
    if (nome == null) throw const Falha('Funcionário não encontrado.');

    final hoje = _relogio();
    if (ano * 12 + mes > hoje.year * 12 + hoje.month) {
      throw const Falha('Não é possível publicar o holerite de um mês que ainda não começou.');
    }

    final indice = _registros.indexWhere(
      (r) => r.holerite.codigoFuncionario == codigoOk && r.holerite.ano == ano && r.holerite.mes == mes,
    );
    if (indice >= 0 && !substituir) {
      throw Falha('Já existe holerite de ${mes.toString().padLeft(2, '0')}/$ano para $codigoOk. Peça para substituir.');
    }

    if (indice >= 0) {
      final antigo = _registros[indice].holerite;
      final novo = Holerite(
        id: antigo.id,
        codigoFuncionario: antigo.codigoFuncionario,
        nomeFuncionario: antigo.nomeFuncionario,
        mes: mes,
        ano: ano,
        dataPublicacao: DateTime(hoje.year, hoje.month, hoje.day),
        nomeArquivo: antigo.nomeArquivo,
      );
      _registros[indice] = _Registro(novo, Uint8List.fromList(conteudo));
      return HoleritePublicado(holerite: novo, substituido: true);
    }

    final criado = _adicionar(codigoOk, nome, mes, ano, hoje, pdf: Uint8List.fromList(conteudo));
    return HoleritePublicado(holerite: criado, substituido: false);
  }

  /// PDF de uma página com [titulo], montado à mão (tabela xref com os deslocamentos reais).
  /// Serve para o botão "Baixar PDF" abrir um arquivo de verdade no protótipo.
  static Uint8List gerarPdfDeExemplo(String titulo) {
    // Só caracteres ASCII seguros dentro do texto do PDF.
    final seguro = titulo.replaceAll(RegExp(r'[^A-Za-z0-9 .:/-]'), '');
    final conteudo = 'BT /F1 18 Tf 72 720 Td ($seguro) Tj ET';
    final objetos = <String>[
      '<< /Type /Catalog /Pages 2 0 R >>',
      '<< /Type /Pages /Kids [3 0 R] /Count 1 >>',
      '<< /Type /Page /Parent 2 0 R /MediaBox [0 0 595 842] /Contents 4 0 R /Resources << /Font << /F1 5 0 R >> >> >>',
      '<< /Length ${conteudo.length} >>\nstream\n$conteudo\nendstream',
      '<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica >>',
    ];

    final saida = StringBuffer('%PDF-1.4\n');
    final deslocamentos = <int>[];
    for (var i = 0; i < objetos.length; i++) {
      deslocamentos.add(saida.length);
      saida.write('${i + 1} 0 obj\n${objetos[i]}\nendobj\n');
    }
    final inicioXref = saida.length;
    saida.write('xref\n0 ${objetos.length + 1}\n0000000000 65535 f \n');
    for (final d in deslocamentos) {
      saida.write('${d.toString().padLeft(10, '0')} 00000 n \n');
    }
    saida.write('trailer\n<< /Size ${objetos.length + 1} /Root 1 0 R >>\nstartxref\n$inicioXref\n%%EOF\n');

    // Tudo ASCII: os deslocamentos em caracteres são iguais aos em bytes.
    return Uint8List.fromList(ascii.encode(saida.toString()));
  }
}
```

```dart
  Future<HoleritePublicado> publicar({
    required String codigo,
    required int mes,
    required int ano,
    required ArquivoSelecionado arquivo,
    bool substituir = false,
  }) async {
    if (state) throw const Falha('Já existe uma publicação em andamento. Aguarde um instante.');
    state = true;
    try {
      final resultado = await ref.read(holeriteRepositoryProvider).publicar(
            codigo: codigo,
            mes: mes,
            ano: ano,
            nomeArquivo: arquivo.nome,
            conteudo: arquivo.bytes,
            substituir: substituir,
          );

      // Se a consulta aberta é desse funcionário, a lista passa a mostrar o holerite novo.
      final consulta = ref.read(consultaHoleritesViewModelProvider).valueOrNull;
      if (consulta != null && consulta.codigo == resultado.holerite.codigoFuncionario) {
        await ref.read(consultaHoleritesViewModelProvider.notifier).atualizar();
      }
      return resultado;
    } finally {
      state = false;
    }
  }
}
```

```dart
      if (escolhido == null || !mounted) return; // cancelou

      setState(() {
        _arquivo = escolhido;
        _erro = null;
        _conflito = false;

      if (mounted) setState(() => _erro = e.mensagem);

    return null;
```

```dart
/// Holerites (RF29, RF30): regras, repositório da API (MockClient), repositório em memória,
/// ViewModels e a aba. Sem rede, sem plugins (seletor e salvador de arquivos são falsos) e com relógio fixo.
void main() {
  const token = 'jwt-de-teste';
  final agora = DateTime(2026, 10, 27, 12); // terça-feira
  final pdf = HoleriteRepositoryPrototipo.gerarPdfDeExemplo('Teste');

  UsuarioAutenticado usuario(String perfil, {String codigo = 'FUNC-0002', String nome = 'Renata Lopes'}) =>
      UsuarioAutenticado(
        token: token,
        expiraEm: DateTime(2030),
        codigo: codigo,
        nome: nome,
        email: 'usuario@hrauto.local',
        perfil: perfil,
      );

  HoleriteRepositoryPrototipo repositorio(UsuarioAutenticado u) => HoleriteRepositoryPrototipo(
        codigoUsuario: u.codigo,
        nomeUsuario: u.nome,
        perfilUsuario: u.perfil,
        relogio: () => agora,
      );

  http.Response json(Object corpo, [int status = 200]) => http.Response(
        jsonEncode(corpo),
        status,
        headers: {'content-type': 'application/json; charset=utf-8'},
      );

  Map<String, dynamic> holeriteJson({int id = 7, String codigo = 'FUNC-0004', int mes = 9, int ano = 2026}) => {
        'id': id,
        'codigoFuncionario': codigo,
        'nomeFuncionario': 'Ana Souza',
        'mes': mes,
        'ano': ano,
        'dataPublicacao': '2026-10-05',
        'nomeArquivo': 'holerite-$ano-${mes.toString().padLeft(2, '0')}.pdf',
      };

  HoleriteRepositoryApi repoApi(MockClient cliente, {String? Function()? sessao}) => HoleriteRepositoryApi(
        ApiClient(cliente: cliente, baseUrl: 'http://api'),
        token: sessao ?? () => token,
      );

  group('RegrasHolerite', () {
    test('normalizarCodigo aceita FUNC-0004 em qualquer caixa ou só o número', () {
      expect(RegrasHolerite.normalizarCodigo('func-0004'), 'FUNC-0004');
      expect(RegrasHolerite.normalizarCodigo(' FUNC-0004 '), 'FUNC-0004');
      expect(RegrasHolerite.normalizarCodigo('4'), 'FUNC-0004');
      expect(RegrasHolerite.normalizarCodigo(' 12 '), 'FUNC-0012');
      expect(RegrasHolerite.normalizarCodigo('0'), isNull);
      expect(RegrasHolerite.normalizarCodigo('FUNC-12'), isNull);
      expect(RegrasHolerite.normalizarCodigo('abc'), isNull);
      expect(RegrasHolerite.normalizarCodigo(''), isNull);
      expect(RegrasHolerite.normalizarCodigo(null), isNull);
    });

    test('pareceUmPdf confere a assinatura %PDF-', () {
      expect(RegrasHolerite.pareceUmPdf(pdf), isTrue);
      expect(RegrasHolerite.pareceUmPdf(Uint8List.fromList(utf8.encode('<html>erro</html>'))), isFalse);
      expect(RegrasHolerite.pareceUmPdf(Uint8List(3)), isFalse);
    });

    test('validarArquivo recusa vazio, grande demais, não-PDF e extensão errada', () {
      expect(() => RegrasHolerite.validarArquivo('h.pdf', pdf), returnsNormally);
      expect(
        () => RegrasHolerite.validarArquivo('h.pdf', Uint8List(0)),
        throwsA(isA<Falha>().having((f) => f.mensagem, 'mensagem', 'O arquivo do holerite está vazio.')),
      );
      expect(
        () => RegrasHolerite.validarArquivo('h.pdf', Uint8List(RegrasHolerite.tamanhoMaximoBytes + 1)),
        throwsA(isA<Falha>().having((f) => f.mensagem, 'mensagem', contains('5 MB'))),
      );
      expect(
        () => RegrasHolerite.validarArquivo('h.txt', pdf),
        throwsA(isA<Falha>().having((f) => f.mensagem, 'mensagem', 'O holerite deve ser um arquivo PDF.')),
      );
      expect(
        () => RegrasHolerite.validarArquivo('h.pdf', Uint8List.fromList(utf8.encode('não sou um pdf'))),
        throwsA(isA<Falha>().having((f) => f.mensagem, 'mensagem', 'O arquivo escolhido não é um PDF válido.')),
      );
    });

    test('validarPublicacao devolve o código normalizado e confere mês e ano', () {
      String validar({String codigo = '4', int mes = 9, int ano = 2026}) => RegrasHolerite.validarPublicacao(
            codigo: codigo,
            mes: mes,
            ano: ano,
            nomeArquivo: 'h.pdf',
            conteudo: pdf,
          );

      expect(validar(), 'FUNC-0004');
      expect(() => validar(codigo: 'xx'), throwsA(isA<Falha>()));
      expect(() => validar(mes: 13), throwsA(isA<Falha>()));
      expect(() => validar(mes: 0), throwsA(isA<Falha>()));
      expect(() => validar(ano: 1999), throwsA(isA<Falha>()));
    });

    test('tamanhoLegivel', () {
      expect(RegrasHolerite.tamanhoLegivel(512), '512 B');
      expect(RegrasHolerite.tamanhoLegivel(340 * 1024), '340 KB');
      expect(RegrasHolerite.tamanhoLegivel(1572864), '1,5 MB');
    });
  });

  group('ApiClient (arquivos)', () {
    test('getBytes devolve o corpo sem interpretar e manda o token', () async {
      late http.Request enviado;
      final cliente = MockClient((req) async {
        enviado = req;
        return http.Response.bytes(pdf, 200, headers: {'content-type': 'application/pdf'});
      });

      final bytes = await ApiClient(cliente: cliente, baseUrl: 'http://api').getBytes('/api/v1/holerites/7/arquivo', token: token);

      expect(bytes, pdf);
      expect(enviado.headers['Authorization'], 'Bearer $token');
      expect(enviado.headers['Accept'], contains('application/pdf'));
    });

    test('getBytes mostra a mensagem do ProblemDetails quando o servidor recusa', () async {
      final cliente = MockClient((_) async => json({'title': 'Não encontrado', 'detail': 'Holerite não encontrado.'}, 404));

      await expectLater(
        ApiClient(cliente: cliente, baseUrl: 'http://api').getBytes('/api/v1/holerites/9/arquivo', token: token),
        throwsA(isA<ApiException>()
            .having((e) => e.statusCode, 'status', 404)
            .having((e) => e.mensagem, 'mensagem', 'Holerite não encontrado.')),
      );
    });

    test('postMultipart envia campos e arquivo e devolve o JSON', () async {
      late http.Request enviado;
      final cliente = MockClient((req) async {
        enviado = req;
        return json({'ok': true}, 201);
      });

      final resposta = await ApiClient(cliente: cliente, baseUrl: 'http://api').postMultipart(
        '/api/v1/holerites',
        campos: {'Codigo': 'FUNC-0004'},
        campoArquivo: 'Arquivo',
        nomeArquivo: 'h.pdf',
        bytes: pdf,
        token: token,
      );

      expect(resposta, {'ok': true});
      expect(enviado.method, 'POST');
      expect(enviado.headers['content-type'], startsWith('multipart/form-data; boundary='));
      expect(enviado.headers['Authorization'], 'Bearer $token');
      expect(enviado.body, contains('name="Codigo"'));
      expect(enviado.body, contains('name="Arquivo"; filename="h.pdf"'));
      expect(enviado.body, contains('%PDF-'));
    });

    test('postMultipart: 413 vira mensagem de arquivo grande demais', () async {
      final cliente = MockClient((_) async => http.Response('', 413));

      await expectLater(
        ApiClient(cliente: cliente, baseUrl: 'http://api').postMultipart(
          '/api/v1/holerites',
          campos: const {},
          campoArquivo: 'Arquivo',
          nomeArquivo: 'h.pdf',
          bytes: pdf,
          token: token,
        ),
        throwsA(isA<ApiException>().having((e) => e.mensagem, 'mensagem', 'O arquivo enviado é grande demais.')),
      );
    });
  });

  group('HoleriteRepositoryApi', () {
    test('listar sem código traz os holerites de quem está logado', () async {
      late http.Request enviado;
      final cliente = MockClient((req) async {
        enviado = req;
        return json([holeriteJson(), holeriteJson(id: 6, mes: 8)]);
      });

      final lista = await repoApi(cliente).listar();

      expect(enviado.url.path, '/api/v1/holerites');
      expect(enviado.url.queryParameters, isEmpty);
      expect(enviado.headers['Authorization'], 'Bearer $token');
      expect(lista.length, 2);
      expect(lista.first.competencia, '09/2026');
      expect(lista.first.dataPublicacao, DateTime(2026, 10, 5));
      expect(lista.first.nomeFuncionario, 'Ana Souza');
    });

    test('listar por código normaliza e manda o ano', () async {
      late http.Request enviado;
      final cliente = MockClient((req) async {
        enviado = req;
        return json(<Object>[]);
      });

      final lista = await repoApi(cliente).listar(codigo: '4', ano: 2026);

      expect(enviado.url.queryParameters, {'codigo': 'FUNC-0004', 'ano': '2026'});
      expect(lista, isEmpty);
    });

    test('código inválido e sessão ausente não chamam a API', () async {
      var chamou = false;
      final cliente = MockClient((_) async {
        chamou = true;
        return json(<Object>[]);
      });

      await expectLater(repoApi(cliente).listar(codigo: 'abc'), throwsA(isA<Falha>()));
      await expectLater(repoApi(cliente, sessao: () => null).listar(), throwsA(isA<Falha>()));
      expect(chamou, isFalse);
    });

    test('resposta fora do formato vira mensagem para o usuário', () async {
      final cliente = MockClient((_) async => json([{'id': 'não é número'}]));

      await expectLater(
        repoApi(cliente).listar(),
        throwsA(isA<ApiException>().having((e) => e.mensagem, 'mensagem', 'Resposta da API em formato inesperado.')),
      );
    });

    test('baixar devolve o PDF com o nome do holerite', () async {
      late http.Request enviado;
      final cliente = MockClient((req) async {
        enviado = req;
        return http.Response.bytes(pdf, 200, headers: {'content-type': 'application/pdf'});
      });
      final repo = repoApi(cliente);
      final holerite = HoleriteRepositoryApi.holeriteDeJson(holeriteJson());

      final arquivo = await repo.baixar(holerite);

      expect(enviado.url.path, '/api/v1/holerites/7/arquivo');
      expect(arquivo.nome, 'holerite-2026-09.pdf');
      expect(arquivo.bytes, pdf);
    });

    test('baixar recusa resposta que não é PDF (página de erro de proxy, por exemplo)', () async {
      final cliente = MockClient((_) async => http.Response('<html>Bad gateway</html>', 200));

      await expectLater(
        repoApi(cliente).baixar(HoleriteRepositoryApi.holeriteDeJson(holeriteJson())),
        throwsA(isA<ApiException>().having((e) => e.mensagem, 'mensagem', contains('PDF válido'))),
      );
    });

    test('publicar envia os campos do formulário e lê o resultado (201)', () async {
      late http.Request enviado;
      final cliente = MockClient((req) async {
        enviado = req;
        return json({'holerite': holeriteJson(), 'substituido': false}, 201);
      });

      final resultado = await repoApi(cliente).publicar(
        codigo: 'func-0004',
        mes: 9,
        ano: 2026,
        nomeArquivo: 'holerite.pdf',
        conteudo: pdf,
      );

      expect(resultado.substituido, isFalse);
      expect(resultado.holerite.codigoFuncionario, 'FUNC-0004');
      expect(enviado.url.path, '/api/v1/holerites');
      expect(enviado.body, contains('name="Codigo"'));
      expect(enviado.body, contains('FUNC-0004'));
      expect(enviado.body, contains('name="Mes"'));
      expect(enviado.body, contains('name="Ano"'));
      expect(enviado.body, contains('name="Arquivo"; filename="holerite.pdf"'));
      expect(enviado.body, isNot(contains('name="Substituir"')));
    });

    test('publicar com substituir manda o campo Substituir e lê o 200', () async {
      late http.Request enviado;
      final cliente = MockClient((req) async {
        enviado = req;
        return json({'holerite': holeriteJson(), 'substituido': true});
      });

      final resultado = await repoApi(cliente).publicar(
        codigo: 'FUNC-0004',
        mes: 9,
        ano: 2026,
        nomeArquivo: 'holerite.pdf',
        conteudo: pdf,
        substituir: true,
      );

      expect(resultado.substituido, isTrue);
      expect(enviado.body, contains('name="Substituir"'));
    });

    test('publicar confere o arquivo antes de enviar e repassa o conflito da API', () async {
      var chamadas = 0;
      final cliente = MockClient((_) async {
        chamadas++;
        return json({'title': 'Conflito', 'detail': 'Já existe holerite de 09/2026 para FUNC-0004.'}, 409);
      });
      final repo = repoApi(cliente);

      await expectLater(
        repo.publicar(
          codigo: 'FUNC-0004',
          mes: 9,
          ano: 2026,
          nomeArquivo: 'holerite.pdf',
          conteudo: Uint8List.fromList(utf8.encode('não é pdf')),
        ),
        throwsA(isA<Falha>()),
      );
      expect(chamadas, 0);

      await expectLater(
        repo.publicar(codigo: 'FUNC-0004', mes: 9, ano: 2026, nomeArquivo: 'holerite.pdf', conteudo: pdf),
        throwsA(isA<ApiException>()
            .having((e) => e.statusCode, 'status', 409)
            .having((e) => e.mensagem, 'mensagem', startsWith('Já existe'))),
      );
      expect(chamadas, 1);
    });
  });

  group('HoleriteRepositoryPrototipo', () {
    test('o usuário começa com os três meses anteriores, do mais recente ao mais antigo', () async {
      final repo = repositorio(usuario('Colaborador'));

      final lista = await repo.listar();

      expect(lista.map((h) => h.competencia), ['09/2026', '08/2026', '07/2026']);
      expect(lista.every((h) => h.codigoFuncionario == 'FUNC-0002'), isTrue);
      expect(lista.every((h) => !h.dataPublicacao.isAfter(agora)), isTrue);
      expect(await repo.listar(ano: 2025), isEmpty);
    });

    test('colaborador só vê os próprios; RH consulta por código e o inexistente dá erro', () async {
      final colaborador = repositorio(usuario('Colaborador'));
      await expectLater(
        colaborador.listar(codigo: 'FUNC-0004'),
        throwsA(isA<Falha>().having((f) => f.mensagem, 'mensagem', 'Você só pode ver os seus próprios holerites.')),
      );

      final rh = repositorio(usuario('RH'));
      expect(await rh.listar(codigo: '4'), isEmpty); // Ana Souza existe e ainda não tem holerite
      await expectLater(
        rh.listar(codigo: 'FUNC-0099'),
        throwsA(isA<Falha>().having((f) => f.mensagem, 'mensagem', 'Funcionário não encontrado.')),
      );
    });

    test('publicar: novo, conflito sem substituir e substituição mantêm a mesma competência única', () async {
      final repo = repositorio(usuario('RH'));

      final novo = await repo.publicar(codigo: '4', mes: 9, ano: 2026, nomeArquivo: 'h.pdf', conteudo: pdf);
      expect(novo.substituido, isFalse);
      expect(novo.holerite.nomeFuncionario, 'Ana Souza');

      await expectLater(
        repo.publicar(codigo: 'FUNC-0004', mes: 9, ano: 2026, nomeArquivo: 'h.pdf', conteudo: pdf),
        throwsA(isA<Falha>().having((f) => f.mensagem, 'mensagem', startsWith('Já existe'))),
      );

      final troca = await repo.publicar(
        codigo: 'FUNC-0004',
        mes: 9,
        ano: 2026,
        nomeArquivo: 'h.pdf',
        conteudo: pdf,
        substituir: true,
      );
      expect(troca.substituido, isTrue);
      expect(troca.holerite.id, novo.holerite.id);
      expect((await repo.listar(codigo: 'FUNC-0004')).length, 1);
    });

    test('regras do servidor: perfil, próprio holerite e mês futuro', () async {
      await expectLater(
        repositorio(usuario('Colaborador')).publicar(codigo: '4', mes: 9, ano: 2026, nomeArquivo: 'h.pdf', conteudo: pdf),
        throwsA(isA<Falha>().having((f) => f.mensagem, 'mensagem', 'Apenas o RH pode publicar holerites.')),
      );

      final rh = repositorio(usuario('RH'));
      await expectLater(
        rh.publicar(codigo: 'FUNC-0002', mes: 9, ano: 2026, nomeArquivo: 'h.pdf', conteudo: pdf),
        throwsA(isA<Falha>().having((f) => f.mensagem, 'mensagem', 'Você não pode publicar o seu próprio holerite.')),
      );
      await expectLater(
        rh.publicar(codigo: 'FUNC-0004', mes: 11, ano: 2026, nomeArquivo: 'h.pdf', conteudo: pdf),
        throwsA(isA<Falha>().having((f) => f.mensagem, 'mensagem', contains('ainda não começou'))),
      );
    });

    test('baixar entrega o PDF publicado e recusa holerite que não existe', () async {
      final repo = repositorio(usuario('RH'));
      final publicado = await repo.publicar(codigo: '4', mes: 9, ano: 2026, nomeArquivo: 'h.pdf', conteudo: pdf);

      final arquivo = await repo.baixar(publicado.holerite);
      expect(arquivo.bytes, pdf);
      expect(arquivo.nome, 'holerite-2026-09.pdf');

      final inexistente = HoleriteRepositoryApi.holeriteDeJson(holeriteJson(id: 999));
      await expectLater(repo.baixar(inexistente), throwsA(isA<Falha>()));
    });

    test('o PDF de exemplo tem xref coerente (abre em qualquer leitor)', () {
      final texto = ascii.decode(HoleriteRepositoryPrototipo.gerarPdfDeExemplo('Holerite 09/2026 - FUNC-0002'));
      final inicio = int.parse(RegExp(r'startxref\n(\d+)\n').firstMatch(texto)!.group(1)!);

      expect(texto.startsWith('%PDF-1.4'), isTrue);
      expect(texto.substring(inicio).startsWith('xref'), isTrue);
      expect(texto.trimRight().endsWith('%%EOF'), isTrue);
    });
  });

        holeriteRepositoryProvider.overrideWithValue(repositorio(u)),
        salvadorArquivoProvider.overrideWithValue(salvador ?? _SalvadorFalso()),

      ]);
      addTearDown(container.dispose);
      return container;

      expect(lista.map((h) => h.competencia), ['09/2026', '08/2026', '07/2026']);

      final consulta = c.read(consultaHoleritesViewModelProvider.notifier);

      await consulta.consultar('abc');
      expect(c.read(consultaHoleritesViewModelProvider).error, isA<Falha>());

      await consulta.consultar('4');

      final resultado = await c.read(publicarHoleriteViewModelProvider.notifier).publicar(
            codigo: '4',
            mes: 9,
            ano: 2026,
            arquivo: ArquivoSelecionado(nome: 'h.pdf', bytes: pdf),

      expect(resultado.substituido, isFalse);
      expect(c.read(publicarHoleriteViewModelProvider), isFalse);
      final consulta = c.read(consultaHoleritesViewModelProvider).value!;
      expect(consulta.holerites.map((h) => h.competencia), ['09/2026']);
      expect(consulta.nome, 'Ana Souza');
    });

    test('publicar repetido sem substituir falha; com substituir troca o PDF', () async {
      final c = criar(usuario('RH'));
      final publicar = c.read(publicarHoleriteViewModelProvider.notifier);
      final arquivo = ArquivoSelecionado(nome: 'h.pdf', bytes: pdf);

      await publicar.publicar(codigo: '4', mes: 9, ano: 2026, arquivo: arquivo);
      await expectLater(
        publicar.publicar(codigo: '4', mes: 9, ano: 2026, arquivo: arquivo),
        throwsA(isA<Falha>().having((f) => f.mensagem, 'mensagem', startsWith('Já existe'))),
      );
      expect(c.read(publicarHoleriteViewModelProvider), isFalse); // liberou o botão mesmo com erro

      final troca = await publicar.publicar(codigo: '4', mes: 9, ano: 2026, arquivo: arquivo, substituir: true);
      expect(troca.substituido, isTrue);
    });

    test('segundo clique durante o envio é recusado', () async {
      final c = criar(usuario('RH'));
      final publicar = c.read(publicarHoleriteViewModelProvider.notifier);
      final arquivo = ArquivoSelecionado(nome: 'h.pdf', bytes: pdf);

      final primeiro = publicar.publicar(codigo: '4', mes: 9, ano: 2026, arquivo: arquivo);
      await expectLater(
        publicar.publicar(codigo: '4', mes: 8, ano: 2026, arquivo: arquivo),
        throwsA(isA<Falha>().having((f) => f.mensagem, 'mensagem', contains('em andamento'))),
      );
      await primeiro;
      expect(c.read(publicarHoleriteViewModelProvider), isFalse);

  group('AbaHolerites', () {
    Future<_Cena> abrir(
      WidgetTester tester,
      String perfil, {
      ArquivoSelecionado? arquivo,
      Falha? erroSeletor,
    }) async {
      tester.view.physicalSize = const Size(1440, 1800);
      tester.view.devicePixelRatio = 1.0;
      addTearDown(tester.view.reset);

      final u = usuario(perfil);
      final salvador = _SalvadorFalso();
      final seletor = _SeletorFalso(arquivo: arquivo, erro: erroSeletor);

      await tester.pumpWidget(
        ProviderScope(
          overrides: [

/// Seletor de arquivos falso: devolve [arquivo] (ou null = cancelou) ou lança [erro].
class _SeletorFalso implements SeletorArquivo {
  _SeletorFalso({this.arquivo, this.erro});

  final ArquivoSelecionado? arquivo;
  final Falha? erro;
  int chamadas = 0;

  @override
  Future<ArquivoSelecionado?> escolherPdf() async {
    chamadas++;
    if (erro != null) throw erro!;
    return arquivo;
  }
}

/// Salvador falso: guarda o que seria baixado.
class _SalvadorFalso implements SalvadorArquivo {
  final List<String> nomes = [];
  final List<Uint8List> conteudos = [];

  @override
  Future<String> salvarPdf({required String nome, required Uint8List bytes}) async {
    nomes.add(nome);
    conteudos.add(bytes);
    return nome;
  }
}

class _Cena {
  const _Cena(this.salvador, this.seletor);

  final _SalvadorFalso salvador;
  final _SeletorFalso seletor;
}
```
