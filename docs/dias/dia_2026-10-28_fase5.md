## O que esta parte entrega
Requisito de negócio: o funcionário tem código próprio (FUNC-0001) e a busca é pelo código, não pelo nome. Até hoje o app só entrava por e-mail (o Flutter recusava o código antes de chamar a API). Agora:
- **API**: `POST /api/v1/auth/login` aceita `{ "login": "FUNC-0007", "senha": "..." }` (e-mail ou código) e continua aceitando o campo antigo `email` (que também aceita o código). Se vierem os dois, vale `login`.
- Código com minúsculas e espaços nas pontas é aceito (`func-0007`). Formato inválido (`FUNC-7`, `FUNC-0000`, nome, `ana`) = mesma resposta 401 de qualquer credencial inválida, sem consultar o banco e sem log (não dá para descobrir quem tem conta). Código inexistente, funcionário sem login, senha errada e conta inativa também recebem a mesma mensagem; inexistente gasta um BCrypt (tempo igual).
- Auditoria (RF08): login pelo código grava `Login` com a descrição "Entrada pelo código FUNC."; falhas seguem como `LoginFalha`/`LoginNegado`.
- **Flutter**: o campo "E-mail ou usuário" passa a aceitar e-mail **ou** código (`FUNC-0001`, `func-1`, `FUNC1` → normalizado para `FUNC-0001`); nome/usuário solto é barrado no formulário (poupa o limite de 10 tentativas por minuto). Protótipo em memória também entra pelo código (Admin de demonstração = `FUNC-0001`).

### backend/src/HRAuto.Application/Autenticacao/AutenticarUsuario.cs

```csharp
using HRAuto.Application.Abstractions;
using HRAuto.Application.Common;
using HRAuto.Domain.Entities;
using HRAuto.Domain.Enums;
using HRAuto.Domain.ValueObjects;

namespace HRAuto.Application.Autenticacao;

/// <summary>
/// Corpo de POST /auth/login. Campos opcionais para que a falta deles vire mensagem clara, não erro genérico.
/// "Login" aceita o e-mail OU o código do funcionário (FUNC-0001); "Email" continua aceito por compatibilidade
/// com o app antigo. Se os dois vierem preenchidos, vale "Login".
/// </summary>
public sealed record CredenciaisDto(string? Email, string? Senha, string? Login = null);

public sealed record UsuarioAutenticadoDto(string Codigo, string Nome, string Email, string Perfil);

public sealed record LoginRespostaDto(string Token, DateTimeOffset ExpiraEm, UsuarioAutenticadoDto Usuario);

/// <summary>
/// Login com e-mail ou código FUNC e senha (RF01, RNF04, RNF05). Devolve o token JWT com o perfil.
/// Cuidados de segurança:
/// - a mesma mensagem para identificador inexistente, mal formado, senha errada e conta inativa (não revela quem tem conta);
/// - conta inexistente também gasta um BCrypt, para o tempo de resposta não denunciar o identificador;
/// - a conta inativa só é reconhecida depois de a senha estar correta;
/// - login e falha de login ficam no log de auditoria (RF08, RNF07).
/// O código FUNC-0001 é derivado de FUNCIONARIO.id_funcionario (o modelo físico não tem coluna de código).
/// </summary>
public sealed class AutenticarUsuario
{
    public const string MensagemCredenciaisInvalidas = "E-mail ou senha inválidos.";
    public const string MensagemCamposObrigatorios = "Informe o e-mail (ou o código FUNC) e a senha.";

    // Hash descartável para gastar o mesmo tempo do BCrypt quando o identificador não existe. Criado uma vez por processo.
    private static string? _hashFalso;

    private readonly IUsuarioRepositorio _usuarios;
    private readonly IHashSenha _hash;
    private readonly IServicoToken _tokens;
    private readonly IAuditoria _auditoria;
    private readonly IUnidadeDeTrabalho _unidade;
    private readonly IRelogio _relogio;
    private readonly IUsuarioGestaoRepositorio _acessos;

    public AutenticarUsuario(
        IUsuarioRepositorio usuarios,
        IHashSenha hash,
        IServicoToken tokens,
        IAuditoria auditoria,
        IUnidadeDeTrabalho unidade,
        IRelogio relogio,
        IUsuarioGestaoRepositorio acessos)
    {
        _usuarios = usuarios;
        _hash = hash;
        _tokens = tokens;
        _auditoria = auditoria;
        _unidade = unidade;
        _relogio = relogio;
        _acessos = acessos;
    }

    public async Task<Resultado<LoginRespostaDto>> ExecutarAsync(CredenciaisDto? credenciais, CancellationToken cancelamento = default)
    {
        // "Login" (e-mail ou FUNC) tem prioridade; "Email" é o campo antigo.
        var digitado = credenciais is null
            ? null
            : string.IsNullOrWhiteSpace(credenciais.Login) ? credenciais.Email : credenciais.Login;

        if (credenciais is null || string.IsNullOrWhiteSpace(digitado) || string.IsNullOrEmpty(credenciais.Senha))
            return Resultado<LoginRespostaDto>.Falha(MensagemCamposObrigatorios);

        // Tem "@" = e-mail; senão tem de ser um código FUNC-0001. A busca é sempre pelo código, nunca pelo nome.
        var identificador = digitado.Trim();
        var porCodigo = !identificador.Contains('@');

        Usuario? usuario;
        if (porCodigo)
        {
            // Código fora do formato nunca existe no banco: mesma resposta de qualquer outra credencial inválida.
            if (!CodigoFuncionario.TentarConverter(identificador, out var codigo))
                return CredenciaisInvalidas();

            usuario = await _acessos.ObterPorFuncionarioAsync(codigo.Id, cancelamento);
        }
        else
        {
            if (!EmailUsuario.TentarNormalizar(identificador, out var email))
                return CredenciaisInvalidas();

            usuario = await _usuarios.ObterPorEmailAsync(email, cancelamento);
        }

        if (usuario is null)
        {
            _hash.Verificar(credenciais.Senha, ObterHashFalso());
            return CredenciaisInvalidas();
        }

        if (!_hash.Verificar(credenciais.Senha, usuario.Senha))
        {
            _auditoria.Registrar(usuario.Id, "LoginFalha", "USUARIO", usuario.Id, "Senha incorreta.");
            await _unidade.SalvarAsync(cancelamento);
            return CredenciaisInvalidas();
        }

        var funcionario = usuario.Funcionario;
        var perfilValido = Enum.IsDefined(typeof(PerfilAcesso), usuario.IdPerfil);
        if (!usuario.Ativo || funcionario is null || !funcionario.Ativo || !perfilValido)
        {
            _auditoria.Registrar(usuario.Id, "LoginNegado", "USUARIO", usuario.Id, "Usuário ou funcionário inativo.");
            await _unidade.SalvarAsync(cancelamento);
            return CredenciaisInvalidas();
        }

        var perfil = (PerfilAcesso)usuario.IdPerfil;
        usuario.UltimoLogin = _relogio.Agora().DateTime;
        _auditoria.Registrar(usuario.Id, "Login", "USUARIO", usuario.Id, porCodigo ? "Entrada pelo código FUNC." : null);
        await _unidade.SalvarAsync(cancelamento);

        var codigoFunc = CodigoFuncionario.Formatar(funcionario.Id);
        var token = _tokens.GerarAcesso(new DadosToken(usuario.Id, funcionario.Id, codigoFunc, funcionario.Nome, usuario.Email, perfil));

        return Resultado<LoginRespostaDto>.Ok(new LoginRespostaDto(
            token.Token,
            token.ExpiraEm,
            new UsuarioAutenticadoDto(codigoFunc, funcionario.Nome, usuario.Email, NomesPerfil.De(perfil))));
    }

    private static Resultado<LoginRespostaDto> CredenciaisInvalidas() =>
        Resultado<LoginRespostaDto>.Falha(TipoErro.NaoAutenticado, MensagemCredenciaisInvalidas);

    private string ObterHashFalso() => _hashFalso ??= _hash.Gerar(Guid.NewGuid().ToString("N"));
}
```

### hr_auto_login/lib/core/validation/identificador_login.dart

```dart
/// Como o usuário se identifica no login: e-mail corporativo ou código do funcionário (FUNC-0001).
/// Espelha o back-end (AutenticarUsuario.cs): se tem "@" é e-mail; senão tem de ser um código.
/// A busca é sempre pelo código, nunca pelo nome (regra de negócio do HR-Auto).
enum TipoLogin { email, codigoFunc }

class IdentificadorLogin {
  const IdentificadorLogin._(this.tipo, this.valor);

  final TipoLogin tipo;

  /// E-mail sem espaços nas pontas, ou código normalizado ("func-1" -> "FUNC-0001").
  final String valor;

  static const int _tamanhoMaximoEmail = 254; // mesmo limite de EmailUsuario.cs

  // Mesmo padrão de EmailUsuario.cs (algo@dominio.ext).
  static final RegExp _email = RegExp(r'^[^@\s]+@[^@\s]+\.[^@\s]+$');

  // Aceita "FUNC-0001", "func-1", "FUNC1" e "FUNC 0001"; o código final sempre tem 4 dígitos.
  static final RegExp _codigo = RegExp(r'^FUNC[-\s]?(\d{1,4})$', caseSensitive: false);

  /// Interpreta o texto digitado. Devolve null se não for e-mail nem código válidos.
  static IdentificadorLogin? tentarLer(String? texto) {
    final limpo = (texto ?? '').trim();
    if (limpo.isEmpty) return null;

    if (limpo.contains('@')) {
      if (limpo.length > _tamanhoMaximoEmail || !_email.hasMatch(limpo)) return null;
      return IdentificadorLogin._(TipoLogin.email, limpo);
    }

    final m = _codigo.firstMatch(limpo);
    if (m == null) return null;
    final numero = int.parse(m.group(1)!);
    if (numero == 0) return null; // não existe funcionário 0
    return IdentificadorLogin._(TipoLogin.codigoFunc, formatarCodigo(numero));
  }

  /// 7 -> "FUNC-0007" (mesmo formato de CodigoFuncionario.Formatar no back-end).
  static String formatarCodigo(int id) => 'FUNC-${id.toString().padLeft(4, '0')}';

  bool get ehCodigo => tipo == TipoLogin.codigoFunc;
}
```

### hr_auto_login/lib/data/repositories/auth_repository_api.dart

```dart
import '../../core/errors/falha.dart';
import '../../core/network/api_client.dart';
import '../../core/validation/identificador_login.dart';
import '../../domain/entities/usuario_autenticado.dart';
import '../../domain/repositories/auth_repository.dart';

/// Implementação real, sobre os endpoints da Fase 4:
///  - POST /api/v1/auth/login          -> 200 { token, expiraEm, usuario }
///  - POST /api/v1/auth/esqueci-senha  -> 202 { mensagem }
/// O login aceita o e-mail (campo "email") ou o código do funcionário (campo "login", ex.: FUNC-0001).
/// O cadastro ainda NÃO tem endpoint: no HR-Auto quem cria a conta do colaborador é o RH.
class AuthRepositoryApi implements AuthRepository {
  AuthRepositoryApi(this._api);

  final ApiClient _api;

  static const String _formatoInvalido = 'Entre com o e-mail corporativo ou com o código FUNC-0001.';

  @override
  Future<UsuarioAutenticado> entrar({required String login, required String senha}) async {
    // Texto fora do formato nunca existe no servidor: avisa antes de gastar uma tentativa
    // (o limite de tentativas do login conta cada chamada).
    final identificador = IdentificadorLogin.tentarLer(login);
    if (identificador == null) throw const Falha(_formatoInvalido);

    final json = await _api.postJson('/api/v1/auth/login', {
      // E-mail segue no campo "email" (compatível com a API antiga); código vai em "login".
      if (identificador.ehCodigo) 'login': identificador.valor else 'email': identificador.valor,
      'senha': senha,
    });
    if (json == null) throw const ApiException('Resposta da API em formato inesperado.');

    try {
      return UsuarioAutenticado.fromJson(json);
    } on TypeError {
      throw const ApiException('Resposta da API em formato inesperado.');
    } on FormatException {
      throw const ApiException('Resposta da API em formato inesperado.');
    }
  }

  @override
  Future<void> cadastrar({required String nome, required String email, required String senha}) async {
    // Sem endpoint público de autocadastro na API (decisão de segurança da Fase 4).
    // Quando o grupo definir o fluxo (ex.: convite do RH ou POST /api/v1/auth/cadastro), troque aqui.
    throw const Falha(
      'O autocadastro ainda não está disponível no servidor. Peça ao RH para criar sua conta.',
    );
  }

  @override
  Future<String> solicitarRecuperacaoSenha(String email) async {
    final json = await _api.postJson('/api/v1/auth/esqueci-senha', {'email': email.trim()});
    final mensagem = json?['mensagem'];
    return mensagem is String && mensagem.isNotEmpty
        ? mensagem
        : 'Se o e-mail estiver cadastrado, você receberá as instruções em instantes.';
  }
}
```

### hr_auto_login/lib/data/repositories/auth_repository_prototipo.dart

```dart
import '../../core/config/app_config.dart';
import '../../core/errors/falha.dart';
import '../../core/validation/identificador_login.dart';
import '../../domain/entities/usuario_autenticado.dart';
import '../../domain/repositories/auth_repository.dart';

/// Repositório em memória para validar as telas sem subir a API.
/// Simula o comportamento do back-end: mesma conta de teste, mesma mensagem genérica
/// para credencial inválida, login por e-mail ou por código FUNC e mesma resposta do
/// "esqueci minha senha". Os dados somem ao fechar o app (nada é gravado em disco).
class AuthRepositoryPrototipo implements AuthRepository {
  AuthRepositoryPrototipo({this.latencia = const Duration(milliseconds: 700)}) {
    _contas[AppConfig.emailDemo] = const _Conta(
      id: 1,
      nome: 'Administrador do Sistema',
      email: AppConfig.emailDemo,
      senha: AppConfig.senhaDemo,
      perfil: 'Admin',
    );
  }

  /// Atraso artificial para ver os estados de carregamento (zero nos testes).
  final Duration latencia;

  final Map<String, _Conta> _contas = {};

  static const String _credenciaisInvalidas = 'E-mail ou senha inválidos.';

  @override
  Future<UsuarioAutenticado> entrar({required String login, required String senha}) async {
    await Future<void>.delayed(latencia);

    final conta = _buscarConta(login);
    if (conta == null || conta.senha != senha) throw const Falha(_credenciaisInvalidas);

    return UsuarioAutenticado(
      token: 'prototipo-${DateTime.now().millisecondsSinceEpoch}',
      expiraEm: DateTime.now().add(const Duration(hours: 8)),
      codigo: IdentificadorLogin.formatarCodigo(conta.id),
      nome: conta.nome,
      email: conta.email,
      perfil: conta.perfil,
    );
  }

  /// Procura pelo e-mail ou pelo código FUNC (nunca pelo nome), como a API.
  _Conta? _buscarConta(String login) {
    final identificador = IdentificadorLogin.tentarLer(login);
    if (identificador == null) return null;

    if (!identificador.ehCodigo) return _contas[identificador.valor.toLowerCase()];

    for (final conta in _contas.values) {
      if (IdentificadorLogin.formatarCodigo(conta.id) == identificador.valor) return conta;
    }
    return null;
  }

  @override
  Future<void> cadastrar({required String nome, required String email, required String senha}) async {
    await Future<void>.delayed(latencia);

    final chave = email.trim().toLowerCase();
    if (_contas.containsKey(chave)) throw const Falha('Este e-mail já está cadastrado.');

    _contas[chave] = _Conta(
      id: _contas.length + 1,
      nome: nome,
      email: chave,
      senha: senha,
      perfil: 'Colaborador', // perfil inicial; RH/Admin são atribuídos pelo administrador
    );
  }

  @override
  Future<String> solicitarRecuperacaoSenha(String email) async {
    await Future<void>.delayed(latencia);
    return 'Se o e-mail estiver cadastrado, você receberá as instruções em instantes.';
  }
}

class _Conta {
  const _Conta({
    required this.id,
    required this.nome,
    required this.email,
    required this.senha,
    required this.perfil,
  });

  final int id;
  final String nome;
  final String email;
  final String senha; // em memória e só no protótipo; a API real guarda hash BCrypt
  final String perfil;
}
```
